\section{Hard Problems}

\begin{frame}{Hard problems}
	\begin{itemize}[<+->]
		\item Cryptography is largely about equating the security of a system to the
		      difficulty of solving a math problem that is thought to be computationally
		      very expensive.
		\item With cryptography, we get security systems that we can literally
		      mathematically prove as secure (under assumptions).
		\item Also, this allows for actual magic.
		      \begin{itemize}[<+->]
			      \item Alice and Bob meet for the first time in the same room as you.
			      \item You are listening to everything they are saying.
			      \item Can they exchange a secret without you learning it?
		      \end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{No known feasible computation}
	\begin{itemize}[<+->]
		\item The discrete logarithm problem:
		      \begin{itemize}
			      \item Given a finite cyclic group $G$, a generator $g \in G$, and an element
			            $h \in G$, find the integer $x$ such that $g^{x}=h$
		      \end{itemize}
		\item In more concrete terms:
		      \begin{itemize}
			      \item Let $p$ be a large prime and let $g$ be a generator of the multiplicative
			            group $\mathbb{Z}_{p}^{*}$ (all nonzero integers modulo $p$).

			      \item Given:
			            \begin{itemize}
				            \item $g \in \mathbb{Z}_{p}^{*}$, $h \in \mathbb{Z}_{p}^{*}$

				            \item Find $x \in \{0, 1, \ldots, p-2\}$ such that $g^{x} \equiv h \pmod
					                  {p}$
			            \end{itemize}

			      \item This problem is believed to be computationally hard when $p$ is large
			            and $g$ is a primitive root modulo $p$.
			            \begin{itemize}
				            \item ``Believed to be'' = we don't know of any way to do it that doesn't
				                  take forever, unless we have a strong, stable quantum computer (Shor's
				                  algorithm)
			            \end{itemize}
		      \end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{Hard problems}
	\begin{columns}[c]
		\begin{column}{0.5\textwidth}
			\textbf{Asymmetric Primitives}
			\begin{itemize}[<+->]
				\item Diffie-Hellman, RSA, ML-KEM, etc.
				\item ``Asymmetric'' because there is a ``public key'' and a ``private
				      key'' for each party.
				\item Algebraic, assume the hardness of mathematical problems (as seen
				      just now).
			\end{itemize}
		\end{column}

		\begin{column}{0.5\textwidth}
			\textbf{Symmetric Primitives}
			\begin{itemize}[<+->]
				\item AES, SHA-2, ChaCha20, HMAC\ldots
				\item ``Symmetric'' because there is one secret key.
				\item Not algebraic but unstructured, relying on their understood
				      resistance to $n$ years of cryptanalysis.
				\item Can act as substitutes for assumptions in security proofs!
				      \begin{itemize}
					      \item Example: hash function assumed to be a ``random oracle''
				      \end{itemize}
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Complexity theory: the plan for this section}
	\definitionbox{Computational Complexity Theory}{The mathematical framework for what makes a problem ``hard'': it classifies problems by the time and memory any algorithm must spend on them.}
	\begin{itemize}[<+->]
		\item Born in the 1960s--70s (Hartmanis--Stearns 1965; Cook 1971, Karp 1972 and Levin 1973 for NP-completeness).
		\item Our route, in four stops:
		      \begin{enumerate}
			      \item What a computational problem is, and how we measure ``hard''.
			      \item The classes P and NP: solving vs.\ checking.
			      \item NP-completeness, reductions, and the P vs.\ NP question.
			      \item \textbf{The punchline}: why NP-complete is the \emph{wrong} kind of hard for cryptography.
		      \end{enumerate}
	\end{itemize}
\end{frame}

\begin{frame}{Computational problems}
	\definitionbox{Computational Problem}{
		A question that can be answered by performing a computation.
		\begin{itemize}
			\item \textbf{Decision problems}: Questions with ``yes'' or ``no'' answers
			      \begin{itemize}
				      \item Example: ``Is 217 a prime number?'' (No: $217 = 7 \times 31$.)
			      \end{itemize}
			\item \textbf{Search problems}: Questions that require finding a specific value
			      \begin{itemize}
				      \item Example: ``How many instances of the letter \textit{`i'} appear in \textit{`incomprehensibilities'}?''
			      \end{itemize}
		\end{itemize}
	}
	\begin{itemize}[<+->]
		\item Cryptography cares about \textbf{search} problems (find the key!); complexity theory prefers \textbf{decision} problems. Usually the two are equivalent: an oracle for ``is $x < k$?'' lets you binary-search for $x$.
	\end{itemize}
\end{frame}

\begin{frame}{Computational hardness}
	\definitionbox{Computational Hardness}{
		The property of computational problems for which no algorithm exists that can solve the problem in a reasonable amount of time.
		\begin{itemize}
			\item Also called \textbf{intractable problems}.
			\item Hardness is independent of the computing device used: all standard computing models are equivalent in what they can compute efficiently.
			\item \textbf{Exception}: Quantum computers, for certain problems.
		\end{itemize}
	}
	\begin{itemize}[<+->]
		\item ``Hard'' means hard for \textbf{every} algorithm, not just the ones we have thought of --- which is what makes hardness so difficult to prove.
		\item Faster hardware only shifts constants: what is hard today should stay hard after ten more years of Moore's law.
	\end{itemize}
\end{frame}

\begin{frame}{Measuring algorithm complexity}
	\begin{columns}[c]
		\begin{column}{0.4\textwidth}
			\begin{itemize}[<+->]
				\item To evaluate hardness, we measure an algorithm's running time using \textbf{asymptotic analysis}.
				\item Common notation:
				      \begin{itemize}
					      \item $\mathcal{O}(n)$: Linear time.
					      \item $\mathcal{O}(n^2)$: Quadratic time.
					      \item $\mathcal{O}(2^n)$: Exponential time.
				      \end{itemize}
				\item What matters is the \textbf{growth} with the input size $n$, not the constants.
			\end{itemize}
		\end{column}

\begin{frame}{Categorizing computational hardness}
	\begin{columns}[c]
		\begin{column}{0.5\textwidth}
			\textbf{Easy Problems}
			\begin{itemize}[<+->]
				\item Solvable in polynomial time.
				\item \textbf{Examples}: Sorting, searching.
				\item Running time: $\mathcal{O}(n^c)$ for some constant $c$.
				\item Generally scales reasonably with input size.
				\item Class P (Polynomial time).
			\end{itemize}
		\end{column}
		\begin{column}{0.5\textwidth}
			\textbf{Hard Problems}
			\begin{itemize}[<+->]
				\item No known polynomial-time solution.
				\item \textbf{Example}: Factoring the product of two large primes.
				\item Best known running time: exponential or subexponential in the input size.
				\item Becomes impractical quickly as input grows.
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Why hard problems matter}
	\begin{itemize}[<+->]
		\item Hard problems \textbf{create asymmetry between legitimate users and attackers}: easy in one direction, hard in reverse.
		      \begin{itemize}
			      \item Easy to multiply two large primes; hard to factor the product.
			      \item Easy to compute $g^x \bmod p$; hard to recover $x$.
		      \end{itemize}
		\item Public-key cryptography is built on exactly these two problems:
		      \begin{itemize}
			      \item \textbf{RSA}: integer factorization.
			      \item \textbf{Diffie-Hellman}: discrete logarithm.
		      \end{itemize}
		\item The security of these schemes depends on the \textbf{continued} hardness of these problems: an algorithmic breakthrough breaks the crypto, no bug required.
	\end{itemize}
\end{frame}

\begin{frame}{Quantum vulnerability of hard problems}
	\begin{itemize}[<+->]
		\item The hard problems we rely on today (factoring, discrete logarithm) are vulnerable to quantum computers.
		\item Shor's algorithm (1994) can efficiently solve both problems on a sufficiently powerful quantum computer.
		\item This has motivated the search for \textbf{``post-quantum''} hard problems:
		      \begin{itemize}[<+->]
			      \item Lattice-based cryptography (e.g., ML-KEM, formerly CRYSTALS-Kyber).
			      \item Hash-based cryptography.
			      \item Code-based cryptography.
			      \item Multivariate cryptography.
			      \item Isogeny-based cryptography.
		      \end{itemize}
		\item In 2024, NIST standardized the first post-quantum algorithms (FIPS 203--205: ML-KEM, ML-DSA, SLH-DSA). More in topic 2.6.
	\end{itemize}
\end{frame}

\begin{frame}{What are complexity classes?}
	\definitionbox{Complexity Class}{
		A group of computational problems that share similar resource requirements (time, memory, etc.).
	}
	\begin{itemize}[<+->]
		\item \textbf{Example}: All problems solvable in $\mathcal{O}(n^2)$ time form one class.
		\item Different classes represent different levels of computational difficulty.
		\item Understanding these classes helps us categorize cryptographic problems.
	\end{itemize}
\end{frame}

\begin{frame}{TIME complexity classes}
	\begin{itemize}[<+->]
		\item \textbf{TIME}$(f(n))$ = class of problems solvable in time $\mathcal{O}(f(n))$.
		\item Examples:
		      \begin{itemize}
			      \item \textbf{TIME}$(n^2)$ = problems solvable in $\mathcal{O}(n^2)$ time
			      \item \textbf{TIME}$(n^3)$ = problems solvable in $\mathcal{O}(n^3)$ time
			      \item \textbf{TIME}$(2^n)$ = problems solvable in $\mathcal{O}(2^n)$ time
		      \end{itemize}
		\item \textbf{Key insight}: If you can solve a problem in $\mathcal{O}(n^2)$ time, you can also solve it in $\mathcal{O}(n^3)$ time.
		\item Therefore: \textbf{TIME}$(n^2) \subseteq$ \textbf{TIME}$(n^3) \subseteq$ \textbf{TIME}$(n^4) \subseteq \ldots$
	\end{itemize}
\end{frame}

\begin{frame}{The class P (Polynomial time)}
	\definitionbox{Class P}{
		The union of all \textbf{TIME}$(n^k)$ classes for all constants $k$.
		\begin{itemize}
			\item P = $\textbf{TIME}(n) \cup \textbf{TIME}(n^2) \cup \textbf{TIME}(n^3) \cup \ldots$
			\item Contains all problems solvable in polynomial time.
			\item Generally considered ``efficiently solvable''.
		\end{itemize}
	}
	\begin{itemize}[<+->]
		\item Most practical algorithms we use daily are in class P.
		\item Examples: Sorting, searching, basic arithmetic.
		\item Cryptography relies on problems \textbf{believed} not to be in P --- nobody can prove it!
	\end{itemize}
\end{frame}

\begin{frame}{SPACE complexity classes: memory matters too}
	\begin{itemize}[<+->]
		\item \textbf{SPACE}$(f(n))$ = class of problems solvable using $\mathcal{O}(f(n))$ bits of memory; \textbf{PSPACE} = union of all \textbf{SPACE}$(n^k)$ for constants $k$.
		\item An algorithm running in time $f(n)$ can touch at most $f(n)$ memory cells, so \textbf{TIME}$(f(n)) \subseteq$ \textbf{SPACE}$(f(n))$, and hence $P \subseteq PSPACE$.
		\item The converse fails: brute-force key search uses almost no memory but takes forever.
		\item Why a cryptographer cares: memory is the expensive resource on real hardware.
		      \begin{itemize}
			      \item Reminder from 1.6: \textbf{scrypt} is \textbf{memory-hard} on purpose, so that GPU and ASIC password crackers lose their advantage.
			      \item Baby-step giant-step (coming up) trades memory for time.
		      \end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{The class NP (Nondeterministic Polynomial time)}
	\definitionbox{Class NP}{
		The class of decision problems for which you can \textbf{verify} a solution in polynomial time, even if finding the solution is hard.
	}
	\begin{itemize}[<+->]
		\item \textbf{Key insight}: Easy to check, hard to find!
		\item Given a potential solution, you can run a polynomial-time algorithm to verify if it's correct.
		\item You don't need to find the solution efficiently --- only verify it efficiently.
		\item \textbf{Relationship}: $P \subseteq NP$ (if you can solve it quickly, you can certainly verify it quickly).
	\end{itemize}
\end{frame}

\begin{frame}{NP: two cryptographic examples}
	\begin{columns}[t]
		\begin{column}{0.5\textwidth}
			\textbf{Hash preimage}\\
			Given a hash value $H$, does there exist an input $M$ such that $\func{hash}{M} = H$?
			\begin{itemize}[<+->]
				\item \textbf{Finding} $M$: brute force over inputs, exponential time.
				\item \textbf{Checking} a candidate $M_0$: compute $\func{hash}{M_0}$ once and compare it with $H$.
			\end{itemize}
		\end{column}
		\begin{column}{0.5\textwidth}
			\textbf{Key recovery}\\
			Given plaintext $P$ and ciphertext $C$, does there exist a key $K$ such that $C = E(K, P)$?
			\begin{itemize}[<+->]
				\item \textbf{Finding} $K$: brute-force key search, exponential time.
				\item \textbf{Checking} a candidate $K_0$: compute $E(K_0, P)$ once and compare it with $C$.
			\end{itemize}
		\end{column}
	\end{columns}
	\vspace{0.4cm}
	\begin{itemize}[<+->]
		\item In both cases verification is one hash or one encryption: polynomial time. Both problems are in NP.
		\item This is the whole reason brute force ``works'' at all: you can tell when you have found the answer.
	\end{itemize}
\end{frame}

\begin{frame}{What's (probably) NOT in NP?}
	\begin{itemize}[<+->]
		\item Flip the question: ``Is there \textbf{no} key $K$ with $E(K, P) = C$?''
		      \begin{itemize}
			      \item A ``yes'' answer has no short certificate: to convince someone, you seem to need to rule out all $2^{128}$ keys.
			      \item This is the complement of the NP problem on the previous slide. The class of such complements is \textbf{coNP}.
			      \item Whether NP $=$ coNP is open; almost everyone believes they differ.
		      \end{itemize}
		\item Ciphertext-only key recovery, with truly random plaintexts, isn't even a well-posed problem:
		      \begin{itemize}
			      \item There is nothing to check a candidate key against, so there is no certificate at all.
			      \item Every key ``decrypts'' to \emph{some} plaintext, and all of them look equally plausible.
		      \end{itemize}
		\item \textbf{Moral}: ``in NP'' is a statement about \emph{checking}, and it silently assumes you know what a correct answer looks like.
	\end{itemize}
\end{frame}

\begin{frame}{NP-complete problems}
	\definitionbox{NP-Complete Problems}{
		The hardest decision problems in the class NP.
		\begin{itemize}
			\item No known polynomial-time algorithms exist for worst-case instances.
			\item If any NP-complete problem can be solved efficiently, then \textbf{all} problems in NP can be solved efficiently.
		\end{itemize}
	}
	\begin{itemize}[<+->]
		\item Discovered in the 1970s during the development of complexity theory.
		\item \textbf{Remarkable discovery}: All NP-complete problems are fundamentally equally hard!
		\item Examples: Boolean satisfiability (SAT), traveling salesman problem, graph coloring.
	\end{itemize}
\end{frame}

\begin{frame}{The magic of reductions}
	\definitionbox{Problem Reduction}{
		A polynomial-time transformation of any instance of problem A into an instance of problem B with the same yes/no answer. Solve B efficiently and you solve A efficiently.
	}
	\begin{itemize}[<+->]
		\item You can transform \textbf{any} (generalized) Sudoku puzzle into a Boolean formula: the Sudoku has a solution $\Leftrightarrow$ the formula is satisfiable.
		\item NP-complete means: \textbf{every} problem in NP reduces to it. So all NP-complete problems reduce to each other --- the same problem in different costumes.
		\item \textbf{Consequence}: a polynomial-time algorithm for Sudoku would solve \textbf{every} problem in NP efficiently, and prove $P = NP$.
		\item \textbf{The practical upside}: progress on one is progress on all --- better SAT solvers mean better chip verification and planning.
	\end{itemize}
\end{frame}

					      \item All classic Legend of Zelda games
					      \item All classic Metroid games
					      \item All classic Pok\'emon role-playing games
				      \end{itemize}
				\item \textbf{The catch}: ``Generalized versions'' with arbitrarily large levels.
				      \begin{itemize}
					      \item Real Nintendo levels are designed to be solvable by humans.
					      \item But the \textbf{mathematical structure} of these games is inherently complex.
				      \end{itemize}
				\item \textbf{Bonus}: the Donkey Kong Country games and several Zelda games are even \textbf{PSPACE-complete}: as hard as anything solvable with polynomial memory.
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{The P vs. NP problem}
	\definitionbox{The P vs. NP Problem}{
		One of the most important unsolved problems in computer science and mathematics.
		\begin{itemize}
			\item \textbf{Question}: Does P = NP?
			\item \textbf{Translation}: Are there problems that are easy to verify but fundamentally hard to solve?
		\end{itemize}
	}
	\begin{itemize}[<+->]
		\item If you could solve \textbf{any} NP-complete problem in polynomial time, then you could solve \textbf{all} NP problems in polynomial time.
		\item This would mean P = NP.
		\item \textbf{Intuition says}: Surely some problems are easy to check but hard to solve!
		\item \textbf{Example}: Brute-force key recovery seems inherently exponential-time\ldots
		\item \textbf{Reality}: No one has proved this mathematically!
	\end{itemize}
\end{frame}

\begin{frame}{What if P = NP?}
	\begin{center}
		\Large\textbf{The cryptographic apocalypse scenario}
	\end{center}
	\begin{itemize}[<+->]
		\item If P = NP, then \textbf{any easily checked solution would also be easy to find} (assuming a practical algorithm, not $n^{1000}$).
		\item \textbf{Symmetric cryptography} would be completely broken:
		      \begin{itemize}
			      \item Key recovery becomes polynomial-time.
			      \item AES, ChaCha20, all symmetric ciphers become useless.
		      \end{itemize}
		\item \textbf{Hash functions} would be invertible in polynomial time:
		      \begin{itemize}
			      \item Finding preimages becomes easy.
			      \item Digital signatures, password storage, all broken.
		      \end{itemize}
		\item \textbf{All of modern cryptography} would collapse overnight --- except the one-time pad and other information-theoretic schemes, which never relied on hardness.
		\item \textbf{But also}: protein folding, logistics, chip design and finding mathematical proofs would all become easy\ldots
	\end{itemize}
\end{frame}

\begin{frame}{The million-dollar question: why we don't panic}
	\begin{itemize}[<+->]
		\item The \textbf{Clay Mathematics Institute} offers \$1,000,000 for settling P vs.\ NP --- one of its seven ``Millennium Prize Problems'', unclaimed since 2000.
		\item Scott Aaronson calls it \textit{``one of the deepest questions that human beings have ever asked''}.
		\item \textbf{The consensus}: most complexity theorists believe $P \neq NP$.
		      \begin{itemize}
			      \item 50+ years of attacks on thousands of NP-complete problems, with enormous practical incentives, and not one polynomial-time algorithm.
		      \end{itemize}
	\end{itemize}
	\definitionbox{Why the question is so hard}{
		\begin{itemize}
			\item \textbf{Proving $P = NP$}: exhibit one polynomial-time algorithm for one NP-complete problem.
			\item \textbf{Proving $P \neq NP$}: show that no such algorithm can \textbf{ever} exist, for any of them --- a statement about all possible programs.
		\end{itemize}
	}
\end{frame}

\begin{frame}{Why NP-complete problems don't work for cryptography}
	\begin{itemize}[<+->]
		\item \textbf{Tempting idea}: Base cryptography on NP-complete problems for provable security!
		\item \textbf{The dream}: Prove that breaking some cipher is NP-hard.
		      \begin{itemize}
			      \item Security would be guaranteed as long as $P \neq NP$.
		      \end{itemize}
		\item \textbf{Reality is disappointing}: NP-complete problems are hard in the \textbf{worst case}, not the \textbf{average case}.
		      \begin{itemize}
			      \item The structure that makes them hard can make specific instances easy.
			      \item Cryptography needs problems that are hard for \textbf{random} instances.
		      \end{itemize}
		\item \textbf{What we actually use}: Problems that are probably \textbf{not} NP-hard.
		      \begin{itemize}
			      \item Factoring, discrete logarithm, lattice problems: believed hard on average, not known to be NP-complete.
			      \item Factoring even lies in NP $\cap$ coNP; if it were NP-complete then NP $=$ coNP, which almost nobody believes.
		      \end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{Average-case vs. worst-case hardness}
	\begin{columns}[c]
		\begin{column}{0.5\textwidth}
			\textbf{Worst-case hardness (NP-complete)}
			\begin{itemize}[<+->]
				\item Some instances of the problem are very hard.
				\item Other instances might be easy.
				\item \textbf{Example}: 3-SAT has hard instances, but also trivial ones.
				\item Not suitable for cryptography.
			\end{itemize}
		\end{column}
		\begin{column}{0.5\textwidth}
			\textbf{Average-case hardness (Crypto)}
			\begin{itemize}[<+->]
				\item Random instances are typically hard.
				      \begin{itemize}
					      \item \textbf{Example}: given $g^a \bmod p$ for random $a$, find $a$ (Diffie-Hellman).
					      \item \textbf{Example}: factoring $N = pq$ for random primes (RSA).
				      \end{itemize}
				\item Exception: lattices (Ajtai, 1996) --- random instances are as hard as the worst case.
			\end{itemize}
		\end{column}
	\end{columns}
	\definitionbox{Hard Problems for Cryptography}{
		We need problems where almost every instance is hard, not just the worst ones.
	}
\end{frame}

\begin{frame}{Discussion time}
	\begin{enumerate}
		\item \textbf{Sudoku is NP-complete. Why would a cipher whose key is ``the solution to this Sudoku'' be a terrible idea?}
		      \begin{itemize}
			      \item Think about which Sudokus you would actually be handing out.
		      \end{itemize}
		\item \textbf{``This AES key maps this plaintext to this ciphertext'' is easy to check. What about ``no AES key does''?}
		      \begin{itemize}
			      \item Which class is each statement in?
		      \end{itemize}
		\item \textbf{Suppose someone proves $P \neq NP$ tomorrow. Is Diffie-Hellman now proven secure?}
		      \begin{itemize}
			      \item What exactly would that proof say about the discrete logarithm?
		      \end{itemize}
	\end{enumerate}
\end{frame}

\begin{frame}{The key exchange problem}
	\begin{itemize}[<+->]
		\item Alice and Bob want to communicate securely over the internet.
		\item They've never met before and share no secrets.
		\item How can they establish a shared secret key for encryption?
		\item Traditional approach: meet in person, exchange keys physically.
		\item \textbf{Problem}: This doesn't scale for the internet!
	\end{itemize}
	\definitionbox{The Challenge}{
		Create a shared secret between two parties who have never communicated before, even when an eavesdropper can see everything they send to each other.
	}
\end{frame}

\begin{frame}{The magic of Diffie-Hellman}
	\begin{itemize}[<+->]
		\item Whitfield Diffie and Martin Hellman solved this ``impossible'' problem.
		\item Their solution (1976) came one year \textbf{before} RSA (1977).
		      \begin{itemize}
			      \item Hellman himself says it should be called \emph{Diffie-Hellman-Merkle}: Ralph Merkle had the idea of public-key distribution first.
			      \item GCHQ's Malcolm Williamson had found the same scheme in 1974; it stayed classified until 1997.
		      \end{itemize}
		\item Uses the \textbf{discrete logarithm problem} as its foundation.
		\item Allows two strangers to create a shared secret in public!
	\end{itemize}
	\vspace{0.3cm}
	\begin{center}
		\Large\textbf{This was the birth of modern cryptography}
	\end{center}
\end{frame}

\begin{frame}{What makes the discrete logarithm hard?}
	Take $p = 17$ and $g = 3$ (a generator of $\mathbb{Z}_{17}^*$). Easy one way, hard the other:
	\vspace{0.15cm}
	\begin{columns}[t]
		\begin{column}{0.5\textwidth}
			\textbf{Easy direction: computing powers}
			\begin{itemize}[<+->]
				\item $3^1 \bmod 17 = 3$
				\item $3^2 \bmod 17 = 9$
				\item $3^3 \bmod 17 = 27 \bmod 17 = 10$
				\item $3^4 \bmod 17 = 81 \bmod 17 = 13$
				\item $3^5 \bmod 17 = 243 \bmod 17 = 5$
			\end{itemize}
		\end{column}
		\begin{column}{0.5\textwidth}
			\textbf{Hard direction: finding the exponent}
			\begin{itemize}[<+->]
				\item Given $g = 3$, $p = 17$ and the result $5$: which exponent was it?
				\item With small numbers: easy, try them all (answer: $x = 5$, the only one below $16$).
				\item With a 2048-bit $p$: practically impossible, even with every computer on Earth.
			\end{itemize}
		\end{column}
	\end{columns}
	\vspace{0.2cm}
	\pause
	Computing $g^x \bmod p$ takes about $\log_2 x$ multiplications (square-and-multiply); going back has no known shortcut.
\end{frame}

\begin{frame}{Mathematical groups: the foundation}
	\definitionbox{What is a Mathematical Group?}{
		A set of elements with an operation that follows specific rules.
		\begin{itemize}
			\item Think of it as a \textbf{mathematical playground} with consistent rules.
			\item For cryptography, we use $\mathbb{Z}_p^*$: numbers $\{1, 2, 3, \ldots, p-1\}$ with multiplication mod $p$.
		\end{itemize}
	}
	\begin{itemize}[<+->]
		\item \textbf{Example}: $\mathbb{Z}_5^* = \{1, 2, 3, 4\}$ with multiplication mod 5
		      \begin{itemize}
			      \item $3 \times 4 = 12 \equiv 2 \pmod{5}$
			      \item $2 \times 3 = 6 \equiv 1 \pmod{5}$
		      \end{itemize}
		\item The ``rules'' ensure the math works consistently for cryptography.
	\end{itemize}
\end{frame}

\begin{frame}{Group rules (simplified)}
	For our cryptographic group $\mathbb{Z}_p^*$, these rules always hold:
	\begin{itemize}[<+->]
		\item \textbf{Closure}: Multiplying any two elements gives another element in the group
		      \begin{itemize}
			      \item In $\mathbb{Z}_5^*$: $2 \times 3 = 1$ (still in the group!)
		      \end{itemize}
		\item \textbf{Identity}: There's a special element (1) that doesn't change others
		      \begin{itemize}
			      \item $1 \times 4 = 4$, $1 \times 2 = 2$, etc.
		      \end{itemize}
		\item \textbf{Inverses}: Every element has a ``partner'' that multiplies to 1
		      \begin{itemize}
			      \item In $\mathbb{Z}_5^*$: $2 \times 3 = 1$, so 2 and 3 are inverses
		      \end{itemize}
		\item \textbf{Associativity}: $(a \times b) \times c = a \times (b \times c)$
	\end{itemize}
	\vspace{0.5cm}
	\textbf{Why care?} These rules guarantee that our cryptographic operations will behave predictably!
\end{frame}

\begin{frame}{Generators: reaching every element in the group}
	\definitionbox{Generator}{
		An element $g$ whose powers $g^1, g^2, g^3, \ldots$ produce \textbf{every element in the group}.
	}
	\begin{itemize}[<+->]
		\item In $\mathbb{Z}_5^*$, let's try $g = 2$:
		      \begin{itemize}
			      \item $2^1 \bmod 5 = 2$
			      \item $2^2 \bmod 5 = 4$
			      \item $2^3 \bmod 5 = 3$
			      \item $2^4 \bmod 5 = 1$
		      \end{itemize}
		\item We got $\{2, 4, 3, 1\}$ - that's all the elements! So $g = 2$ is a generator.
		\item \textbf{Generators are crucial}: They let us express every group element as a power of $g$.
	\end{itemize}
\end{frame}

\begin{frame}{The discrete logarithm problem (DLP)}
	\definitionbox{Discrete Logarithm Problem}{
		Given $g$, $p$, and $h = g^x \bmod p$, find the secret exponent $x$.
	}
	\begin{itemize}[<+->]
		\item \textbf{``Discrete''} because we work with integers, not real numbers.
		\item \textbf{``Logarithm''} because we're finding the exponent (like $\log_2(8) = 3$).
		\item \textbf{Example}: Given $g = 3$, $p = 17$, $h = 10$, find $x$ such that $3^x \equiv 10 \pmod{17}$.
		      \begin{itemize}
			      \item Answer: $x = 3$ (since $3^3 = 27 \equiv 10$)
			      \item Easy with small numbers, hard with large ones!
		      \end{itemize}
		\item For cryptographic-sized numbers (2048+ bits), no efficient algorithm is known.
	\end{itemize}
\end{frame}

\begin{frame}{DLP vs. factoring: similarly hard}
	\begin{columns}[c]
		\begin{column}{0.5\textwidth}
			\textbf{Factoring Problem}
			\begin{itemize}[<+->]
				\item Given $N = p \times q$, find $p$ and $q$
				\item Used in RSA (1977)
				\item Well-known, intuitive
			\end{itemize}
		\end{column}
		\begin{column}{0.5\textwidth}
			\textbf{Discrete Logarithm}
			\begin{itemize}[<+->]
				\item Given $g^x \bmod p$, find $x$
				\item Used in Diffie-Hellman (1976)
				\item Less intuitive, more mathematical
			\end{itemize}
		\end{column}
	\end{columns}
	\vspace{1cm}
	\begin{itemize}[<+->]
		\item \textbf{Security equivalence}: $n$-bit factoring $\approx$ $n$-bit discrete logarithm
		\item Both are vulnerable to Shor's quantum algorithm
		\item Both are \textbf{not} known to be NP-hard
		\item Algorithms for both problems share similar techniques
	\end{itemize}
\end{frame}

\begin{frame}{Attacking the discrete logarithm: can we beat brute force?}
	\begin{itemize}[<+->]
		\item \textbf{Brute force}: try every $x \in \{0, 1, \ldots, p-2\}$ until $g^x \equiv h \pmod{p}$.
		      \begin{itemize}
			      \item Running time: $\mathcal{O}(p)$ --- for a 2048-bit prime, longer than the age of the universe.
		      \end{itemize}
		\item But brute force is the \textit{dumbest} possible algorithm. Surely we can be smarter?
		\item Two classic \textbf{generic} algorithms beat brute force in \textbf{any} group:
		      \begin{itemize}
			      \item \textbf{Baby-step giant-step}: trade memory for time.
			      \item \textbf{Pollard's rho}: same speed, (almost) no memory.
		      \end{itemize}
		\item Both bring the running time down from $\mathcal{O}(p)$ to $\mathcal{O}(\sqrt{p})$.
	\end{itemize}
	\alertbox{Spoiler}{$\sqrt{p}$ is a colossal improvement over $p$, but for a 2048-bit prime, $\sqrt{p} \approx 2^{1024}$. Still hopeless.}
\end{frame}

\begin{frame}{Diffie-Hellman: the mathematical version}
	\textbf{Setup}: Alice and Bob agree on public values $g$ (generator) and $p$ (large prime)
	\vspace{0.5cm}
	\begin{enumerate}[<+->]
		\item \textbf{Alice}: Chooses secret $a$, computes $A = g^a \bmod p$, sends $A$ to Bob
		\item \textbf{Bob}: Chooses secret $b$, computes $B = g^b \bmod p$, sends $B$ to Alice
		\item \textbf{Alice}: Computes shared secret $S = B^a \bmod p = (g^b)^a \bmod p = g^{ab} \bmod p$
		\item \textbf{Bob}: Computes shared secret $S = A^b \bmod p = (g^a)^b \bmod p = g^{ab} \bmod p$
	\end{enumerate}
	\vspace{0.5cm}
	\textbf{Result}: Alice and Bob both have $S = g^{ab} \bmod p$ without ever sharing $a$ or $b$!
\end{frame}

\begin{frame}{Diffie-Hellman example with small numbers}
	\begin{columns}[c]
		\begin{column}{0.5\textwidth}
			\textbf{Setup}
			\begin{itemize}[<+->]
				\item Public parameters: $p = 13$, $g = 2$
				\item Alice picks secret: \textcolor{red}{$a = 5$}
				\item Bob picks secret: \textcolor{red}{$b = 7$}
			\end{itemize}
			\pause
			\vspace{0.3cm}
			\textbf{Public Exchange}
			\begin{itemize}[<+->]
				\item Alice computes: $A = g^{\textcolor{red}{a}} \bmod p = 6$
				\item Bob computes: $B = g^{\textcolor{red}{b}} \bmod p = 11$
				\item Alice sends $A = 6$ to Bob
				\item Bob sends $B = 11$ to Alice
			\end{itemize}
		\end{column}
		\begin{column}{0.5\textwidth}
			\pause
			\textbf{Shared Secret Computation}
			\begin{itemize}[<+->]
				\item Alice computes: $S = B^{\textcolor{red}{a}} \bmod p = 11^{\textcolor{red}{5}} \bmod 13$
				      \begin{itemize}
					      \item $= 161051 \bmod 13 = \textcolor{red}{7}$
				      \end{itemize}
				\item Bob computes: $S = A^{\textcolor{red}{b}} \bmod p = 6^{\textcolor{red}{7}} \bmod 13$
				      \begin{itemize}
					      \item $= 279936 \bmod 13 = \textcolor{red}{7}$
				      \end{itemize}
				\item \textcolor{red}{\textbf{Shared secret: $S = 7$}}
			\end{itemize}
			\pause
			\vspace{0.3cm}
			\textbf{Eavesdropper sees only:}
			\begin{itemize}[<+->]
				\item $p = 13$, $g = 2$, $A = 6$, $B = 11$
				\item (In the real world, $p$, \textcolor{red}{$a$}, and \textcolor{red}{$b$} are much larger numbers)
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Your turn: run Diffie-Hellman by hand}
	Public parameters: $p = 23$, $g = 5$. Alice picks $a = 6$, Bob picks $b = 15$. Work in pairs and reduce mod $23$ after every multiplication.
	\begin{enumerate}
		\item What does Alice send? ($A = 5^6 \bmod 23$)
		\item What does Bob send? ($B = 5^{15} \bmod 23$)
		\item What shared secret do they both end up with?
		\item You are Eve and you see $p, g, A, B$. How many exponents might you have to try to find $a$?
	\end{enumerate}
	\pause
	\vspace{0.1cm}
	\definitionbox{Answers}{
		$A = 8$ (since $5^2 = 25 \equiv 2$, so $5^6 = 2^3 = 8$); \quad $B = 19$; \quad $S = B^6 \bmod 23 = A^{15} \bmod 23 = 2$.\\
		Eve: at most $22$ tries here; $2^{2048}$ for a 2048-bit prime, still $2^{1024}$ with baby-step giant-step. And Eve doesn't actually need $a$ or $b$, only $S$: that is the next slide's problem.
	}
\end{frame}

\begin{frame}{The computational Diffie-Hellman (CDH) problem}
	\definitionbox{Computational Diffie-Hellman (CDH) Problem}{
		Given $g^a \bmod p$ and $g^b \bmod p$, compute the shared secret $g^{ab} \bmod p$ without knowing the secret values $a$ and $b$.
	}
	\begin{itemize}[<+->]
		\item \textbf{Motivation}: Even if an eavesdropper captures the public values $g^a$ and $g^b$, they shouldn't be able to determine the shared secret $g^{ab}$.
		\item \textbf{Example}: Given $A = 6$ and $B = 11$ from our earlier example, can you compute $S = 7$?
		      \begin{itemize}
			      \item With $p = 13$ you can brute-force it in seconds; with a 2048-bit $p$, nobody knows how.
		      \end{itemize}
		\item \textbf{Real-world relevance}: This is exactly what an attacker faces when trying to break Diffie-Hellman.
	\end{itemize}
\end{frame}

\begin{frame}{CDH vs. DLP: the relationship}
	\begin{itemize}[<+->]
		\item \textbf{Key insight}: If you can solve DLP, then you can also solve CDH.
		      \begin{itemize}
			      \item Given $g^a$ and $g^b$, use DLP to find $a$ and $b$
			      \item Then compute $g^{ab}$ directly
		      \end{itemize}
		\item \textbf{Mathematical relationship}: DLP is \textbf{at least as hard} as CDH.
		      \begin{itemize}
			      \item CDH $\leq$ DLP (CDH reduces to DLP)
		      \end{itemize}
		\item \textbf{Open question}: Is CDH at least as hard as DLP?
		      \begin{itemize}
			      \item We don't know if solving CDH allows you to solve DLP!
			      \item Maybe there's a clever way to compute $g^{ab}$ without finding $a$ and $b$
		      \end{itemize}
		\item \textbf{Security assumption}: We assume CDH is hard even if it turns out to be easier than DLP.
	\end{itemize}
\end{frame}

\begin{frame}{The decisional Diffie-Hellman (DDH) problem}
	\definitionbox{Decisional Diffie-Hellman (DDH) Problem}{
		Given $g^a \bmod p$, $g^b \bmod p$, and a value $X$ that is either $g^{ab} \bmod p$ (the real shared secret) or $g^c \bmod p$ for a random $c$, each with probability $1/2$: determine which one $X$ is.
	}
	\begin{itemize}[<+->]
		\item \textbf{Why do we need this?} Indistinguishability!
		      \begin{itemize}
			      \item What if an attacker can compute the first 32 bits of $g^{ab}$?
			      \item CDH isn't broken, but the attacker learns something --- and that partial leak may be all an application needs to fail.
		      \end{itemize}
		\item \textbf{DDH ensures}: The shared secret $g^{ab}$ is \textbf{indistinguishable} from a random group element.
		\item \textbf{Where it's used}: security proofs of ElGamal, Cramer-Shoup and many key exchange protocols. In our library notation, it's what lets us swap the real key for a uniform one.
	\end{itemize}
\end{frame}

\begin{frame}{DDH vs. CDH: the hierarchy}
	\begin{itemize}[<+->]
		\item \textbf{Key relationship}: If you can solve CDH, then you can solve DDH.
		      \begin{itemize}
			      \item Given $(g^a, g^b, X)$, use CDH to compute $g^{ab}$
			      \item Check if $X = g^{ab}$; if yes, then $X$ is the real shared secret
		      \end{itemize}
		\item \textbf{Hardness hierarchy}: DDH $\leq$ CDH $\leq$ DLP.
		      \begin{itemize}
			      \item DDH is \textbf{no harder} than CDH.
			      \item CDH is, in turn, no harder than DLP.
		      \end{itemize}
		\item \textbf{Surprising fact}: DDH is \textbf{not hard} in certain groups!
		      \begin{itemize}
			      \item In $\mathbb{Z}_p^*$, DDH can be broken using the Legendre symbol (quadratic residuosity): $g^{ab}$ is a square exactly when $g^a$ or $g^b$ is, while a random $g^c$ is a square only half the time.
			      \item But CDH remains hard in the same group.
		      \end{itemize}
		\item \textbf{Solution}: work in a group of large \textbf{prime order} $q$, where this leak disappears: the order-$q$ subgroup of $\mathbb{Z}_p^*$ for $p = 2q + 1$, or a prime-order elliptic curve. There DDH is believed hard.
	\end{itemize}
\end{frame}

\begin{frame}{Diffie-Hellman key exchange in practice}
	\begin{center}
		\Large\textbf{How does this actually work in the real world?}
	\end{center}
	\vspace{0.5cm}
	\begin{enumerate}[<+->]
		\item \textbf{Parameter generation}: Choose secure values for $p$ and $g$
		      \begin{itemize}
			      \item $p$ must be a large prime (2048+ bits)
			      \item $g$ must be a generator of a large subgroup
		      \end{itemize}
		\item \textbf{Key generation}: Each party picks a random secret
		      \begin{itemize}
			      \item Alice picks $a$ randomly from $\{1, 2, \ldots, p-2\}$
			      \item Bob picks $b$ randomly from $\{1, 2, \ldots, p-2\}$
		      \end{itemize}
		\item \textbf{Public key computation}: Each party computes their public value
		\item \textbf{Key exchange}: Public values are sent over the network
		\item \textbf{Shared secret derivation}: Each party computes the final shared secret
	\end{enumerate}
\end{frame}

\begin{frame}{Don't use $g^{ab}$ directly as a key}
	\begin{itemize}[<+->]
		\item Alice and Bob share $S = g^{ab} \bmod p$. \textbf{Tempting}: just use $S$ as an AES key!
		\item \textbf{Problem}: $S$ is \textbf{not} a uniformly random bitstring --- it's a specific element of a specific algebraic group.
		      \begin{itemize}
			      \item It always lies in the subgroup generated by $g$, and depending on $g$ may always be a \textbf{quadratic residue}.
			      \item Its byte-length can vary (leading zero bytes), a classic source of real bugs: the \textbf{Raccoon attack} (2020) on TLS 1.2 exploited exactly this.
		      \end{itemize}
		\item Structured, biased input makes a \textbf{terrible} symmetric key --- it hands an attacker a head start.
	\end{itemize}
	\alertbox{Rule of thumb}{Never feed a shared secret straight into a cipher --- always pass it through a \textbf{KDF} first.}
\end{frame}

\begin{frame}{TLS handshake: Diffie-Hellman in action}
	\textbf{When you visit https://gmail.com, here's what happens} (TLS 1.2 flavour, simplified):
	\vspace{0.2cm}
	\begin{enumerate}[<+->]
		\item \textbf{Client Hello}: Your browser says ``I want to talk securely''
		\item \textbf{Server Hello}: Gmail's server responds with its certificate and its DH share
		      \begin{itemize}
			      \item Includes $p$, $g$, and the server's public value $g^b \bmod p$
		      \end{itemize}
		\item \textbf{Client Key Exchange}: Your browser generates its own secret $a$ and sends $g^a \bmod p$
		\item \textbf{Secret computation}: Both sides compute $g^{ab} \bmod p$
		\item \textbf{Key derivation}: The shared secret goes through a KDF to produce the session keys
		\item \textbf{Secure communication}: All further messages are encrypted with these keys
	\end{enumerate}
	\vspace{0.2cm}
	\textbf{Result}: Your password is encrypted before leaving your computer! (TLS 1.3 sends the browser's share in its first message and uses named groups such as X25519: topic 2.1.)
\end{frame}

\subsection{Attacks on Diffie-Hellman}

\begin{frame}{Weak groups in the wild: the Logjam findings}
	\begin{center}
		\Large\textbf{Not all Diffie-Hellman deployments are created equal}
	\end{center}
	\vspace{0.2cm}
	\begin{itemize}[<+->]
		\item In 2015, the \emph{Imperfect Forward Secrecy} paper\footnote{\url{https://appliedcryptography.page/paper/\#imperfect-dh}} scanned millions of TLS, SSH and VPN servers to see how DH was actually deployed.
		\item \textbf{Headline result}: an academic team can break a 768-bit prime and a nation-state a 1024-bit one. Breaking the single most common 1024-bit group would let a passive attacker read 18\% of the top million HTTPS sites; a second group covers 66\% of VPN and 26\% of SSH servers. (The Logjam downgrade attack itself: topic 2.1.)
		\item \textbf{Our interest today}: the misconfigurations they found along the way.
		      \begin{itemize}
			      \item Servers using 512-bit primes for ordinary, non-export DHE.
			      \item Non-safe primes with composite-order subgroups.
		      \end{itemize}
		\item \textbf{Observation}: a correct algorithm still fails when deployed with bad parameters.
	\end{itemize}
\end{frame}

\begin{frame}{Logjam finding: 512-bit primes still in use}
	2,631 servers with browser-trusted certificates used 512-bit primes --- no export downgrade needed!
	\begin{itemize}[<+->]
		\item \textbf{Why this is bad}:
		      \begin{itemize}
			      \item A 512-bit discrete logarithm was within academic reach in 2015: about a week of precomputation per group, then individual logs in about a minute (median 70 seconds).
			      \item A passive eavesdropper can record now and decrypt later. No active attack needed.
		      \end{itemize}
		\item \textbf{Real-world proof}: the researchers decrypted test connections to www.fbi.gov
		      \begin{itemize}
			      \item The site used OpenSSL's default 512-bit group until April 2015.
			      \item Precompute once for that group, and every connection to it falls.
		      \end{itemize}
		\item \textbf{The lesson}: default configurations can be dangerously weak, and many admins never change defaults.
	\end{itemize}
\end{frame}

\begin{frame}{Logjam finding: composite-order subgroups}
	\begin{itemize}[<+->]
		\item Not all servers use \textbf{safe primes}, where $(p-1)/2$ is also prime.
		      \begin{itemize}
			      \item 4,800 of the 70,000 distinct primes seen were not safe.
			      \item If $p-1$ has many small factors, $\mathbb{Z}_p^*$ has many small subgroups.
		      \end{itemize}
		\item The dangerous combination the paper found:
		      \begin{itemize}
			      \item a generator $g$ whose order has small factors, \textbf{and}
			      \item short exponents (160--256 bits), chosen ``for efficiency''.
			      \item Then an attacker recovers the exponent piece by piece, one small factor at a time. (How? Next slide.)
		      \end{itemize}
		\item \textbf{Safe usage}: $g$ generates a subgroup of one large prime order $q$, and exponents live in $\{1, \ldots, q-1\}$. Then there are no small pieces to recover.
		\item \textbf{Real impact}: 159 hosts fully compromised, including VPN devices and web-conferencing servers.
	\end{itemize}
\end{frame}

\begin{frame}{The Pohlig-Hellman algorithm: divide and conquer}
	\begin{columns}[c]
		\begin{column}{0.5\textwidth}
			\begin{itemize}[<+->]
				\item Generic attacks take $\mathcal{O}(\sqrt{n})$, where $n$ is the group's \textbf{order} (here $n = p - 1$).
				\item \textbf{Notice}: if $n = q_1 \cdot q_2 \cdots q_k$ factors into small primes, we don't need $\sqrt{n}$ at all!
				\item Solve the log separately mod each small $q_i$, then recombine with the \textbf{Chinese Remainder Theorem}.
			\end{itemize}
			{\small A discrete log is only as hard as the \textbf{largest prime factor} of the group order, not the order itself!}
		\end{column}
		\begin{column}{0.46\textwidth}
			\begin{center}

\begin{frame}{Pohlig-Hellman: a tiny worked example}
	Let $p = 41$: the group order $n = 40 = 2^3 \cdot 5$ is \textbf{smooth}. Brute force: up to $40$ checks.
	\begin{itemize}[<+->]
		\item Pohlig-Hellman instead solves $x \bmod 8$ (8 checks) and $x \bmod 5$ (5 checks) separately.
		      \begin{itemize}
			      \item How? $h^5 = (g^5)^x$ and $g^5$ has order $8$: a discrete log in a group of $8$ elements gives $x \bmod 8$. Likewise $h^8 = (g^8)^x$ gives $x \bmod 5$.
		      \end{itemize}
		\item Combine with the \textbf{Chinese Remainder Theorem}: one $x \bmod 40$ matches both.
		\item Total: $\sim 13$ checks instead of $40$ --- and the saving \textbf{explodes} for large primes.
	\end{itemize}
	\definitionbox{The general rule}{If the largest prime factor of $p - 1$ has $b$ bits, Pohlig-Hellman runs in $\mathcal{O}(\sqrt{2^b})$ time --- regardless of how big $p$ is!}
\end{frame}

\begin{frame}{Safe primes: closing the door on Pohlig-Hellman}
	\definitionbox{Safe Prime}{A prime $p = 2q + 1$ where $q$ is \textbf{also prime} (a ``Sophie Germain prime'').}
	\begin{itemize}[<+->]
		\item If $p$ is safe, the group order $p - 1 = 2q$ has \textbf{only} the factors $2$ and $q$.
		\item $q$ is huge, so Pohlig-Hellman gains \textbf{nothing}: the best generic attack stays $\mathcal{O}(\sqrt{q}) \approx \mathcal{O}(\sqrt{p})$.
		\item Bonus: a $g$ of order $q$ also kills the Legendre-symbol leak in DDH.
	\end{itemize}
	\alertbox{Back to Logjam}{Logjam's ``composite-order subgroup attacks'' are Pohlig-Hellman in the wild: non-safe primes handed attackers exactly the smooth structure it needs.}
\end{frame}

\begin{frame}{The dark side: unauthenticated Diffie-Hellman}
	\begin{center}
		\Large\textbf{Even with fully correct Diffie-Hellman implementations, there's a serious problem\ldots}
	\end{center}
	\vspace{0.5cm}
	\begin{itemize}[<+->]
		\item \textbf{The vulnerability}: Basic DH has no authentication
		      \begin{itemize}
			      \item Alice can't verify she's talking to Bob
			      \item Bob can't verify he's talking to Alice
		      \end{itemize}
		\item \textbf{The attack}: Man-in-the-middle (MITM)
		      \begin{itemize}
			      \item Mallory sits between Alice and Bob
			      \item Alice does DH with Mallory, thinking it's Bob
			      \item Bob does DH with Mallory, thinking it's Alice
			      \item Mallory can read and modify everything!
		      \end{itemize}
		\item \textbf{Real-world impact}: This attack is practical and devastating!
	\end{itemize}
\end{frame}

\begin{frame}{Man-in-the-middle attack on DH}
	\textbf{How Mallory breaks ``secure'' communication:}
	\vspace{0.2cm}
	\begin{enumerate}[<+->]
		\item \textbf{Alice $\rightarrow$ Mallory}: Alice sends $g^a$ (thinking it goes to Bob)
		\item \textbf{Mallory $\rightarrow$ Bob}: Mallory sends $g^m$ (Bob thinks it's from Alice)
		\item \textbf{Bob $\rightarrow$ Mallory}: Bob sends $g^b$ (thinking it goes to Alice)
		\item \textbf{Mallory $\rightarrow$ Alice}: Mallory sends $g^m$ (Alice thinks it's from Bob)
		\item \textbf{Result}:
		      \begin{itemize}
			      \item Alice and Mallory share secret $g^{am}$
			      \item Bob and Mallory share secret $g^{bm}$
			      \item Alice and Bob don't share any secret!
		      \end{itemize}
		\item \textbf{Communication}: Alice encrypts with $g^{am}$, Mallory decrypts, reads/modifies, re-encrypts with $g^{bm}$ for Bob
	\end{enumerate}
	\vspace{0.2cm}
	\textbf{Alice and Bob never know they've been compromised!}
\end{frame}

\begin{frame}{Why MITM attacks succeed}
	\begin{itemize}[<+->]
		\item \textbf{Public values look random}: $g^a$ and $g^m$ are indistinguishable.
		      \begin{itemize}
			      \item Both appear to be random group elements.
			      \item No way to tell if they come from the intended party.
		      \end{itemize}
		\item \textbf{No identity verification}: DH only establishes a shared secret.
		      \begin{itemize}
			      \item Doesn't prove who you're sharing it with!
			      \item Like agreeing on a secret handshake with someone wearing a mask.
		      \end{itemize}
		\item \textbf{Active vs. passive attacks}:
		      \begin{itemize}
			      \item DH protects against \textbf{passive} eavesdropping.
			      \item Does nothing against \textbf{active} manipulation.
		      \end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{Solution: authenticated key exchange}
	\definitionbox{Authenticated Key Exchange (AKE)}{
		Key exchange that verifies the identity of the parties involved, preventing man-in-the-middle attacks.
	}
	\begin{itemize}[<+->]
		\item \textbf{Core idea}: Combine DH with authentication mechanisms.
		\item \textbf{Common approaches}:
		      \begin{itemize}
			      \item \textbf{Digital signatures}: Sign the DH public values (TLS).
			      \item \textbf{Pre-shared keys}: Use existing shared secrets (IPsec).
			      \item \textbf{Certificates}: Use a trusted third party (Certificate Authority in HTTPS).
			      \item \textbf{Password-based}: Derive authentication from passwords (PAKE protocols like SRP).
		      \end{itemize}
		\item \textbf{Goal}: Ensure that Alice and Bob can verify they're really talking to each other.
	\end{itemize}
\end{frame}

\begin{frame}{TLS: authenticated DH with certificates}
	\textbf{How HTTPS prevents MITM attacks:}
	\begin{enumerate}[<+->]
		\item \textbf{Server authentication}: Gmail sends its certificate and a \textbf{signature} on $g^b$
		      \begin{itemize}
			      \item The certificate binds the signing key to ``gmail.com''; the signature binds $g^b$ to that key
			      \item The certificate is signed by a trusted Certificate Authority (CA)
		      \end{itemize}
		\item \textbf{Certificate verification}: Your browser checks:
		      \begin{itemize}
			      \item Is the signature valid?
			      \item Is the CA trusted?
			      \item Does the certificate match ``gmail.com''?
			      \item Has the certificate expired?
		      \end{itemize}
		\item \textbf{If verification passes}: You know you're really talking to Gmail
		\item \textbf{If verification fails}: Browser shows scary warnings!
	\end{enumerate}
	\textbf{Result}: MITM attacks become much harder (but not impossible!)
\end{frame}

\begin{frame}{Signal: authenticated DH with fingerprints}

				      \end{itemize}

\begin{frame}{SSH: authenticated DH with host keys}
	\textbf{How SSH prevents server impersonation:}
	\begin{itemize}[<+->]
		\item \textbf{First connection}: Server presents its ``host key'' along with its DH public value
		      \begin{itemize}
			      \item SSH shows you a fingerprint: \texttt{SHA256:ABC123\ldots}
			      \item You're supposed to verify this out-of-band (but nobody does!)
		      \end{itemize}
		\item \textbf{Trust on first use (TOFU)}: Client remembers the host key
		      \begin{itemize}
			      \item Stored in \texttt{\~{}/.ssh/known\_hosts}
		      \end{itemize}
		\item \textbf{Subsequent connections}: Client checks if the host key matches
		      \begin{itemize}
			      \item If different, SSH gives you a heart attack: \texttt{WARNING: REMOTE HOST IDENTIFICATION HAS CHANGED!}
			      \item If same: Connection proceeds normally
		      \end{itemize}
		\item \textbf{User authentication}: Usually with passwords or public keys
	\end{itemize}
	\textbf{Weakness}: TOFU is vulnerable on the very first connection!
\end{frame}

\subsection{The Future of Diffie-Hellman}

\begin{frame}{Modern implementations: elliptic curves}
	\begin{itemize}[<+->]
		\item Baby-step giant-step and Pollard's rho are \textbf{generic}: the same $\mathcal{O}(\sqrt{n})$ attack works in \textbf{any} group of order $n$.
		\item But $\mathbb{Z}_p^*$ has \textbf{extra structure} that elliptic curves lack: it also falls to \textbf{index calculus} and the number field sieve, a faster, \textbf{subexponential} attack. That is why $p$ needs 2048+ bits.
	\end{itemize}
	\vspace{0.2cm}
	\begin{columns}[c]
		\begin{column}{0.5\textwidth}
			\textbf{Finite-field DH ($\mathbb{Z}_p^*$)}
			\begin{itemize}[<+->]
				\item Falls to index calculus.
				\item Needs $p \geq 2048$ bits.
				\item Slower, larger public keys.
			\end{itemize}
		\end{column}
		\begin{column}{0.5\textwidth}
			\textbf{Elliptic-curve DH (ECDH)}
			\begin{itemize}[<+->]
				\item Only the generic $\mathcal{O}(\sqrt{n})$ attack applies.
				\item $\sim$256-bit keys $\approx$ 3072-bit $\mathbb{Z}_p^*$.
				\item Much faster, smaller keys: the standard in TLS, Signal, SSH.
			\end{itemize}
		\end{column}
	\end{columns}
	\vspace{0.2cm}
	\pause
	\textbf{Popular curves}: P-256, P-384, X25519, X448. All of topic 1.8 is about them.
\end{frame}

\begin{frame}{The quantum threat to Diffie-Hellman}
	\begin{center}
		\Large\textbf{All DH variants are doomed\ldots}
	\end{center}
	\vspace{0.1cm}
	\begin{itemize}[<+->]
		\item \textbf{Shor's algorithm} (1994) can break DH on quantum computers.
		      \begin{itemize}
			      \item Solves discrete logarithm in polynomial time.
			      \item Works for both traditional DH and ECDH.
		      \end{itemize}
		\item \textbf{Timeline concerns}:
		      \begin{itemize}
			      \item Cryptographically relevant quantum computers don't exist yet.
			      \item But adversaries might store encrypted data now and decrypt it later.
			      \item ``Harvest now, decrypt later'' attacks.
		      \end{itemize}
		\item \textbf{Post-quantum key exchange}: the replacements are standardized and already deployed.
		      \begin{itemize}
			      \item ML-KEM (lattices): NIST standard since 2024, already in TLS and Signal
			      \item SIDH/SIKE (isogenies): broken in 2022 by Castryck and Decru
			      \item Code-based alternatives (topic 2.6)
		      \end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{Lessons from 50 years of Diffie-Hellman}
	\begin{itemize}[<+->]
		\item \textbf{The math is three lines; the deployments still broke.} Logjam's victims ran a correct algorithm with 512-bit defaults, non-safe primes and short exponents.
		\item \textbf{A shared secret is not an identity.} DH alone falls to the man in the middle; every real system bolts on authentication: certificates (TLS), fingerprints and key transparency (Signal, WhatsApp), trust on first use (SSH).
		\item \textbf{Never use $g^{ab}$ raw.} It is a structured group element, not a uniform key: always run it through a KDF.
		\item \textbf{The group is part of the design.} Prime-order subgroups defeat Pohlig-Hellman and the DDH leak; elliptic curves defeat index calculus and shrink keys by an order of magnitude.
		\item \textbf{Shor changes the primitive, not the idea.} ML-KEM replaces the exponentiation, but ``agree on a secret over a public channel'' remains the job description.
	\end{itemize}
\end{frame}

\section{RSA}

\begin{frame}{RSA: the other public-key pioneer}
	\begin{center}
		\Large\textbf{One year after Diffie-Hellman came RSA (1977)}
	\end{center}
	\vspace{0.15cm}
	\begin{itemize}[<+->]
		\item \textbf{Named after}: Rivest, Shamir, and Adleman (MIT researchers)
		\item \textbf{Different approach}: Based on the hardness of factoring large numbers
		\item \textbf{Key innovation}: One trapdoor gives both public-key encryption AND digital signatures
		\item \textbf{Secret history}: GCHQ's Clifford Cocks found the same scheme in 1973 (classified until 1997)
	\end{itemize}
	\vspace{0.15cm}
	\definitionbox{RSA's Core Idea}{
		Anyone can encrypt messages to you using your public key, but only you can decrypt them with your private key.
	}
\end{frame}

\begin{frame}{How RSA works (simplified)}
	\textbf{The mathematical foundation}:
	\begin{enumerate}[<+->]
		\item \textbf{Key generation}:
		      \begin{itemize}
			      \item Choose two large primes $p$ and $q$
			      \item Compute $n = p \times q$ (public modulus)
			      \item Choose public exponent $e$ (commonly 65537)
			      \item Compute private exponent $d$ such that $e \times d \equiv 1 \pmod{\phi(n)}$
			            \begin{itemize}
				            \item $\phi(n)$ is Euler's totient function: counts integers less than $n$ that are coprime to $n$
				            \item For $n = p \times q$: $\phi(n) = (p-1)(q-1)$
			            \end{itemize}
		      \end{itemize}
		\item \textbf{Encryption}: $c = m^e \bmod n$ (anyone can do this)
		\item \textbf{Decryption}: $m = c^d \bmod n$ (only the private key holder can do this)
	\end{enumerate}
	\vspace{0.5cm}
	\textbf{Security assumption}: Factoring $n$ back into $p$ and $q$ is computationally infeasible.
\end{frame}

\begin{frame}{RSA's dual functionality}
	\begin{columns}[c]
		\begin{column}{0.5\textwidth}
			\textbf{Encryption}
			\begin{itemize}[<+->]
				\item Public key encrypts
				\item Private key decrypts
				\item Provides confidentiality
				\item Anyone can send you secrets
			\end{itemize}
		\end{column}
		\begin{column}{0.5\textwidth}
			\textbf{Digital Signatures}
			\begin{itemize}[<+->]
				\item Private key signs
				\item Public key verifies
				\item Provides authenticity
				\item You can prove you wrote something
			\end{itemize}
		\end{column}
	\end{columns}
	\vspace{0.5cm}
	\begin{itemize}[<+->]
		\item \textbf{Historical significance}: First algorithm to do both!
		\item \textbf{Widespread adoption}: Used in SSL/TLS, email encryption, code signing
		\item \textbf{Patent controversy}: Was patented until 2000, limiting early adoption
	\end{itemize}
\end{frame}

\begin{frame}{RSA in practice}
	\textbf{Where RSA dominated for decades}:
	\begin{itemize}[<+->]
		\item \textbf{HTTPS certificates}: RSA signatures dominated for two decades; ECDSA certificates only became common in the late 2010s
		\item \textbf{Email encryption}: PGP/GPG relied heavily on RSA
		\item \textbf{Software signing}: Windows, macOS, Linux packages
		\item \textbf{SSH keys}: RSA was the default for years
	\end{itemize}
	\vspace{0.5cm}
	\textbf{Key sizes and security}:
	\begin{itemize}[<+->]
		\item 1024-bit RSA: NIST disallowed it after 2013 (about 80-bit security)
		\item 2048-bit RSA: Current minimum (about 112-bit security), but showing its age
		\item 4096-bit RSA: Secure but slow and unwieldy
		\item Compare to ECDSA: 256-bit keys offer similar security to 3072-bit RSA!
	\end{itemize}
\end{frame}

\begin{frame}{Why we're not covering RSA in depth}
	\begin{center}
		\Large\textbf{RSA is a cryptographic dinosaur}
	\end{center}
	\begin{itemize}[<+->]
		\item \textbf{On its way out}: Being actively replaced everywhere
		      \begin{itemize}
			      \item TLS 1.3, SSH, Signal and WireGuard all default to elliptic curves
			      \item Trail of Bits, 2019: ``Seriously, stop using RSA''
		      \end{itemize}
		\item \textbf{Its math doesn't travel}: DH moves to elliptic curves unchanged; RSA's trapdoor is tied to factoring, where the number field sieve keeps pushing key sizes up
		\item \textbf{Performance}: Terribly slow to sign or decrypt
		      \begin{itemize}
			      \item 4096-bit RSA signature: $\sim$10ms; Ed25519 signature: $\sim$0.1ms
			      \item (RSA \emph{verification} is fast, which is why it lingered in certificates)
		      \end{itemize}
		\item \textbf{Implementation pitfalls}: Textbook RSA is deterministic and malleable, so it needs padding --- and PKCS\#1 v1.5 padding gave us Bleichenbacher's oracle attack (1998), rediscovered as ROBOT in 2018
		\item \textbf{The future is elliptic curves and post-quantum}: Time better spent there!
	\end{itemize}
\end{frame}
