\section{Chosen-Plaintext Attacks}

\begin{frame}{CPA security}
	\begin{columns}[c]
		\begin{column}{0.6\textwidth}
			\definitionbox{Security Against Chosen-Plaintext Attacks (CPA Security)}{
				Let $\Sigma$ be an encryption scheme, and let $\Sigma.\mathcal{C}(\ell)$ denote the set of possible ciphertexts for plaintexts of length $\ell$. If $\Sigma$ supports only plaintexts of a single length, we can simply write $\Sigma.\mathcal{C}$ to denote the entire set of ciphertexts.\footnote{i.e., ``just forget about the length''.}\\[0.5em]

				$\Sigma$ has security against \textbf{chosen-plaintext attacks} if the following two libraries are indistinguishable:
			}
		\end{column}
		\begin{column}{0.4\textwidth}
			\sslinked{
				\sslibrary{\Sigma}{cpa-real}{
					$K \twoheadleftarrow \Sigma.\mathcal{K}$\\[1em]
					\sslibrarysubroutine{cpa.enc}{M}{
						$C \coloneq \Sigma.\texttt{Enc}(K, M)$\\
						return $C$
					}{1}
				}{0.8}
			}{\approxeq}{
				\sslibrary{\Sigma}{cpa-rand}{
					\sslibrarysubroutine{cpa.enc}{M}{
						$C \twoheadleftarrow \Sigma.\mathcal{C}(|M|)$\\
						return $C$
					}{1}
				}{0.8}
			}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Why CPA security matters}
	\begin{itemize}[<+->]
		\item CPA security means: \textit{``Even if I let you encrypt any message you want, you can't obtain any distinguisher regarding my scheme.''}
		\item \textbf{Why this matters in real life:}
		      \begin{itemize}
			      \item Attackers can often trick systems into encrypting data they choose.
			      \item Without CPA security, seeing these encryptions could reveal your secrets.
			      \item Example: If identical bank transactions always encrypt to identical ciphertexts, attackers could identify your purchases.
		      \end{itemize}
		\item The only thing that's allowed to leak is the length of messages/ciphertexts.
		\item Every mode we meet today (CBC, CTR, GCM) is designed to have it.
	\end{itemize}
\end{frame}

\begin{frame}{Message length: not important, not unimportant}
	\begin{itemize}[<+->]
		\item Even with CPA security, the \textbf{length} of messages is still leaked.
		\item This seemingly minor leak can reveal surprising information:
		      \begin{itemize}
			      \item \textbf{Encrypted VoIP calls}: Length patterns can reveal which language is spoken.
			      \item \textbf{Encrypted web traffic}: Sizes of requests/responses can identify websites.
			      \item \textbf{Encrypted messages}: Length patterns can reveal the type of content (document, image, etc.).
			      \item \textbf{Encrypted commands}: Length often reveals which command was issued.
		      \end{itemize}
		\item Mitigation typically involves padding messages to fixed lengths or standard increments.
		\item This is why Tor moves data in fixed-size cells and TLS 1.3 lets records carry padding.
	\end{itemize}
\end{frame}

\begin{frame}{CPA security is why we avoid deterministic encryption}
	\begin{columns}[c]
		\begin{column}{0.5\textwidth}
			\begin{itemize}
				\item Deterministic encryption will always fail CPA security!
				\item If the same message always encrypts to the same ciphertext:
				      \begin{itemize}
					      \item Attackers can recognize repeated messages and build a ``dictionary'' of known plaintexts.
				      \end{itemize}
				\item The distinguisher on the right: the real library always answers \texttt{true}, the random one almost never does.
				\item ECB mode is a classic example of insecure deterministic encryption.
			\end{itemize}
		\end{column}
		\begin{column}{0.5\textwidth}
			\begin{center}
				\ssprogram{}{
					$M \twoheadleftarrow \mathcal{M}$\\
					$C_1 \coloneq \func{cpa.enc}{M}$\\
					$C_2 \coloneq \func{cpa.enc}{M}$\\
					return $C_1 \stackrel{?}{=} C_2$
				}{1}
			\end{center}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{AES by itself is deterministic}
	\begin{columns}[c]
		\begin{column}{0.7\textwidth}
			\begin{itemize}[<+->]
				\item AES as a block cipher is inherently \textbf{deterministic}:
				      \begin{itemize}
					      \item $\texttt{AES}(K, M)$ will always produce the same $C$.
					      \item This makes raw AES fail CPA security.
				      \end{itemize}
				\item What makes AES secure in practice?
				      \begin{itemize}
					      \item \textbf{Block cipher modes} (CBC, CTR, GCM) add randomness/state using ``initialization vectors''.
					            \begin{itemize}
						            \item IV is a fancy name for ``random bytes used once''.\footnote{Also referred to as ``nonces'' for ``number used once''. Note: in CBC mode the IV must be \emph{unpredictable}, while in CTR mode the nonce must be \emph{unique} but need not be random.}
					            \end{itemize}
					      \item IVs ensure the same plaintext encrypts differently each time.
				      \end{itemize}
				\item Without a proper mode, AES is like ECB mode: predictable patterns.
			\end{itemize}
		\end{column}
		\begin{column}{0.3\textwidth}
			\begin{center}
				\ssprogram{AES}{
					$K \twoheadleftarrow \bits^{\lambda}$\\
					$M \twoheadleftarrow \mathcal{M}$\\[0.5em]
					$C_1 \coloneq \texttt{AES}(K, M)$\\
					$C_2 \coloneq \texttt{AES}(K, M)$\\[0.5em]
					return $C_1 \stackrel{?}{=} C_2$
				}{0.85}
			\end{center}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{CBC mode: a closer look}
	\begin{columns}[c]
		\begin{column}{0.6\textwidth}
			\begin{itemize}[<+->]
				\item CBC (Cipher Block Chaining) uses the previous ciphertext block to randomize the current block.
				\item Requires a random initialization vector (IV).
				\item Each block's encryption depends on all previous blocks.
				\item Changes to one block affect all subsequent ciphertext blocks.
				\item Sequential encryption (\textbf{can't parallelize}; decryption can be).
			\end{itemize}
		\end{column}
		\begin{column}{0.4\textwidth}
			\sssubroutine{Enc}{K, M_1\|\cdots\|M_\ell}{
				$C_0 \twoheadleftarrow \bits^\lambda$\\
				$\text{for } i=1 \text{ to } \ell:$\\
				$\quad C_i \coloneq F(K, C_{i-1} \oplus M_i)$\\
				$\textbf{return } C_0\|C_1\|\cdots\|C_\ell$
			}{1}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{CTR mode: a closer look}
	\begin{columns}[c]
		\begin{column}{0.6\textwidth}
			\begin{itemize}[<+->]
				\item CTR (Counter) mode turns a block cipher into a stream cipher.
				\item Uses a nonce ($C_0$) plus a counter to create unique inputs to $F$.
				\item Each block's encryption is independent of other blocks.
				\item \textbf{Highly parallelizable} (unlike CBC).
				\item The nonce must never be reused with the same key.
				\item Widely used in modern protocols (TLS, SSH, etc.).
			\end{itemize}
		\end{column}
		\begin{column}{0.4\textwidth}
			\sssubroutine{Enc}{K, M_1\|\cdots\|M_\ell}{
				$C_0 \twoheadleftarrow \bits^\lambda$\\
				$\text{for } i=1 \text{ to } \ell:$\\
				$\quad C_i \coloneq F(K, C_0 + i - 1) \oplus M_i$\\
				$\textbf{return } C_0\|C_1\|\cdots\|C_\ell$
			}{1}
		\end{column}
	\end{columns}
\end{frame}

\section{Chosen-Ciphertext Attacks}

\begin{frame}{Introducing chosen-ciphertext attacks}
	\begin{itemize}
		\item Imagine this scenario at your company:
		      \begin{itemize}
			      \item You discover a bug: the software crashes when decrypting ciphertexts that yield plaintexts with null characters.
			      \item You think: ``We never encrypt messages with null bytes, so no problem!''
		      \end{itemize}
		\item An attacker can:
		      \begin{itemize}
			      \item Send specially crafted ciphertexts to your system.
			      \item Observe which ones crash your software (contain null bytes) and which don't.
			      \item Use this information to completely break your encryption scheme!
		      \end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{Introducing chosen-ciphertext attacks}
	\begin{itemize}
		\item This is a \textbf{chosen-ciphertext attack} (CCA):
		      \begin{itemize}
			      \item The attacker can submit arbitrary ciphertexts for decryption.
			      \item The system leaks information about the decryption results (the crash).
			      \item Even small information leaks can completely compromise security.
			      \item All encryption schemes we've seen so far are vulnerable to this!
		      \end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{CPA vs. CCA security definitions}
	\begin{columns}[c]
		\begin{column}{0.5\textwidth}
			\definitionbox{CPA Security (informal)}{
				\textit{``Even if I let you encrypt any message you want, you can't obtain any distinguisher regarding my scheme.''}
			}
		\end{column}
		\begin{column}{0.5\textwidth}
			\definitionbox{CCA Security (informal)}{
				\textit{``Even if I let you encrypt any message you want AND decrypt any ciphertext you want, you can't obtain any distinguisher regarding my scheme.''}
			}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{CPA vs. CCA security definitions}{To be more precise\ldots}
	\begin{columns}[c]
		\begin{column}{0.5\textwidth}
			\definitionbox{CPA Security (more precise)}{
				\textit{``Even if I let you choose any messages to encrypt, you can't tell which of two chosen messages my challenge ciphertext encrypts, except with probability close to $\frac{1}{2}$.''}
			}
		\end{column}
		\begin{column}{0.5\textwidth}
			\definitionbox{CCA Security (more precise)}{
				\textit{``Even if I let you choose any messages to encrypt and any ciphertexts to decrypt (except the challenge ciphertext itself), you can't tell which of two chosen messages my challenge ciphertext encrypts, except with probability close to $\frac{1}{2}$.''}
			}
		\end{column}
	\end{columns}
	\vspace{0.6em}
	\begin{center}
		{\small These are the ``left-or-right'' flavours. Our library definitions ask for more: ciphertexts indistinguishable from \emph{uniformly random} strings, which implies the left-or-right version.}
	\end{center}
\end{frame}

\begin{frame}{Format-oracle attacks \& malleability}
	\begin{columns}[c]
		\begin{column}{0.6\textwidth}
			\begin{itemize}[<+->]
				\item Scenario: a victim decrypting CTR-mode ciphertexts reveals whether the result contains a null character.
				\item Information can leak through crashes, error messages, or behavior differences.
				\item A single yes/no bit of information can be enough to break encryption!
				\item Malleability: many encryption schemes allow attackers to make predictable changes to plaintexts by modifying ciphertexts.
				\item CTR mode is especially vulnerable because XOR allows targeted bit-flipping.
			\end{itemize}
		\end{column}
		\begin{column}{0.4\textwidth}
			\sssubroutine{nulloracle}{C}{
				$M \coloneq \texttt{CTR.Dec}(K, C)$\\
				if $M$ contains a $\bit{0}\bit{0}$ character:\\
				$\quad$ return $\texttt{true}$\\
				else return $\texttt{false}$
			}{1}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Why is it called \func{nulloracle}{C}?}
	\begin{columns}[c]
		\begin{column}{0.55\textwidth}
			\begin{itemize}
				\item In mythology, an oracle answered on behalf of the gods --- often enigmatically.
				\item In cryptography, an oracle is a black box you may query but never open; we use them in proofs and in attacks.
				\item Our \func{nulloracle}{C} answers one yes/no question: does $\texttt{Dec}(K, C)$ contain a null (\bit{0}\bit{0}) byte?
				\item One bit per query. Watch what we can do with it.
			\end{itemize}
		\end{column}
		\begin{column}{0.45\textwidth}
			\imagewithcaption[0.7\textwidth]{oracle-painting-zoom.jpg}{\textit{The Oracle} by Camillo Miola, 1880\footnote{Well worth viewing in high resolution: \url{https://upload.wikimedia.org/wikipedia/commons/9/94/Camillo_Miola_\%28Biacca\%29_-_The_Oracle_-_72.PA.32_-_J._Paul_Getty_Museum.jpg}}}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Null-oracle attack}
	\begin{columns}[c]
		\begin{column}{0.6\textwidth}
			\begin{itemize}
				\item An adversary who has access to \func{nulloracle}{C} can efficiently compute $\texttt{Dec}(K, C)$ for any $C$!
				\item Yes, it really is enough.
			\end{itemize}
		\end{column}
		\begin{column}{0.4\textwidth}
			\sssubroutine{nulloracle}{C}{
				$M \coloneq \texttt{CTR.Dec}(K, C)$\\
				if $M$ contains a $\bit{0}\bit{0}$ character:\\
				$\quad$ return $\texttt{true}$\\
				else return $\texttt{false}$
			}{1}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{CTR mode is malleable}{Defining malleability}
	\begin{columns}[c]
		\begin{column}{0.5\textwidth}
			\definitionbox{Malleability}{
				An encryption scheme is \textbf{malleable} if, given an encryption $C$ of an unknown plaintext $M$, it is possible to create a new ciphertext $C' \neq C$ where $M' = \texttt{Dec}(K, C')$ is somehow related to $M$, so that $M'$ reveals some information about $M$.
			}
		\end{column}
		\begin{column}{0.5\textwidth}
			\begin{itemize}
				\item In CTR mode: $C_i = F(K, C_0 + i - 1) \oplus M_i$
				\item Flipping a bit in $C_i$ flips exactly the same bit in $M_i$.
				\item Attackers can make targeted modifications without knowing the key.
				\item Example: change ``transfer \$100'' to ``transfer \$900'' by modifying just one byte.
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{CTR mode is malleable}{The attack}
	\begin{columns}[c]
		\begin{column}{0.5\textwidth}
			\begin{itemize}
				\item Suppose adversary captures ciphertext $C$ and wants to find plaintext $M = \texttt{Dec}(K, C)$.
				\item Let's focus on discovering just the last byte of $M$.
				\item If the last byte of $M$ is $b$, then $M = M' \| b$.
				\item What if we modify the ciphertext by XORing the last byte with $b$?
				\item This would turn the last byte of the plaintext into all zeros (null byte)!
			\end{itemize}
		\end{column}
		\begin{column}{0.5\textwidth}
			\begin{align*}
				C \oplus (\ldots \bit{0}\bit{0} \| b) & \text{ decrypts to:}                            \\
				                                      & M \oplus (\ldots \bit{0}\bit{0} \| b)           \\
				                                      & = (M' \| b) \oplus (\ldots \bit{0}\bit{0} \| b) \\
				                                      & = M' \| (b \oplus b)                            \\
				                                      & = M' \| \bit{0}\bit{0}
			\end{align*}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{CTR mode is malleable}{Finding one byte}
	\begin{columns}[c]
		\begin{column}{0.5\textwidth}
			\begin{itemize}
				\item Problem: We don't know $b$ in advance!
				\item Solution: Try all 255 possible values for $b$.
				\item For each guess $g$, we:
				      \begin{itemize}
					      \item Create $C' = C \oplus (\ldots \bit{0}\bit{0} \| g)$.
					      \item Send $C'$ to the \func{nulloracle}{}.
					      \item If the oracle returns \texttt{true}, then $g = b$!
				      \end{itemize}
				\item When we find the right value, we've discovered the last byte of $M$.
			\end{itemize}
		\end{column}
		\begin{column}{0.5\textwidth}
			\begin{center}
				Try these ciphertexts:\\
				$C \oplus (\ldots \bit{0}\bit{0} \| \bit{0}\bit{1})$\\
				$C \oplus (\ldots \bit{0}\bit{0} \| \bit{0}\bit{2})$\\
				$C \oplus (\ldots \bit{0}\bit{0} \| \bit{0}\bit{3})$\\
				$\vdots$\\
				$C \oplus (\ldots \bit{0}\bit{0} \| \bit{f}\bit{e})$\\
				$C \oplus (\ldots \bit{0}\bit{0} \| \bit{f}\bit{f})$
			\end{center}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{CTR mode is malleable}{Finding all bytes}
	\begin{columns}[c]
		\begin{column}{0.6\textwidth}
			\begin{itemize}
				\item We can use the same approach for any byte in the plaintext.
				\item To find the $i$-th byte:
				      \begin{itemize}
					      \item Create a string of all zeros.
					      \item Set only the $i$-th byte to our guess $g$.
					      \item XOR this with the ciphertext.
					      \item Query the oracle.
				      \end{itemize}
				\item By repeating for all bytes, we can recover the entire plaintext!
			\end{itemize}
		\end{column}
		\begin{column}{0.4\textwidth}
			\ssprogram{null-oracle}{
				$n \coloneq |C_1\|\cdots\|C_{\ell}|$\\
				$M \coloneq ""$\\
				\\
				for $i = 1$ to $n$:\\
				$\quad$ for each $b \in \{\bit{0}\bit{1}, \ldots, \bit{f}\bit{f}\}$:\\
				$\quad\quad$ $\Delta \coloneq (\bit{0}\bit{0})^n$\\
				$\quad\quad$ $\Delta[i] \coloneq b$\\
				$\quad\quad$ if $\func{nulloracle}{C \oplus (0^\lambda \| \Delta)}$:\\
				$\quad\quad\quad$ $M \coloneq M \| b$\\
				$\quad\quad\quad$ next $i$\\
				return $M$
			}{0.75}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{CTR mode is malleable}{The power of chosen-ciphertext attacks}
	\begin{itemize}
		\item With at most 255 queries per byte, we can completely decrypt any ciphertext!
		\item For context: a 1KB file falls to at most $1024 \times 255 \approx 261{,}000$ queries --- about half that on average.
		\item This is extremely practical for an attacker.
		\item All from a single bit of information about the plaintext (contains null or not).
		\item This attack works because:
		      \begin{itemize}
			      \item CTR mode is malleable (we can make predictable changes to the plaintext).
			      \item The system leaks a tiny bit of information about decrypted plaintexts.
			      \item Together, these flaws completely break the encryption.
		      \end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{Is the null-oracle attack just brute-force?}
	\begin{itemize}
		\item Yes and no!
		\item It's brute-force on each byte independently:
		      \begin{itemize}
			      \item To recover an $n$-byte plaintext: at most $255n$ oracle queries.
			      \item True brute-force on the entire plaintext: $256^n$ (exponentially worse!)
		      \end{itemize}
		\item For a 16-byte message:
		      \begin{itemize}
			      \item Null-oracle attack: $\approx 4{,}080$ queries ($16 \times 255$)
			      \item True brute-force: $\approx 3.4 \times 10^{38}$ queries ($256^{16} = 2^{128}$)
		      \end{itemize}
		\item The oracle confirms each byte \emph{independently}, turning a product ($256^n$) into a sum ($255 n$).
	\end{itemize}
\end{frame}

\begin{frame}{Discussion time}
	\begin{enumerate}
		\item \textbf{Why 255 guesses per byte and not 256?}
		      \begin{itemize}
			      \item What would happen if the original plaintext already contained a null byte somewhere?
		      \end{itemize}
		\item \textbf{Does this work on CBC mode?}
		      \begin{itemize}
			      \item Flipping a bit in $C_i$ scrambles all of $M_i$ and flips one bit of $M_{i+1}$. Is that still enough for an attacker? (Hold that thought until the padding oracle.)
		      \end{itemize}
		\item \textbf{Where else could a one-bit oracle hide?}
		      \begin{itemize}
			      \item Error messages, response times, packet sizes, whether a connection gets closed\ldots{} name one from a system you use.
		      \end{itemize}
	\end{enumerate}
\end{frame}

\begin{frame}{Can we just rate-limit the number of queries?}
	\begin{columns}[c]
		\begin{column}{0.6\textwidth}
			\begin{itemize}
				\item Rate-limiting might help, but:
				      \begin{itemize}
					      \item It only increases attack time; it doesn't prevent the attack.
					      \item Attackers can be patient or use multiple accounts.
					      \item Legitimate users suffer from the rate-limiting.
				      \end{itemize}
				\item Better approach: cryptographic solutions!
				      \begin{itemize}
					      \item Fix the fundamental vulnerability, not just limit its exploitation.
					      \item Create systems that are mathematically proven to resist chosen-ciphertext attacks.
				      \end{itemize}
			\end{itemize}
		\end{column}
		\begin{column}{0.4\textwidth}
			\imagewithcaption{yoda-meme.jpg}{}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Is fixing the null-byte bug enough?}
	\begin{itemize}[<+->]
		\item The null-character bug is just one example of a broader class of attacks.
		\item Format-oracle attacks can exist without implementation bugs!
		\item They only need:
		      \begin{itemize}
			      \item A system that accepts untrusted ciphertexts
			      \item Decrypts them
			      \item Behaves differently based on the decryption result
			      \item In a way the attacker can observe
		      \end{itemize}
		\item This behavior can come from:
		      \begin{itemize}
			      \item Accidental bugs (null-byte crashes)
			      \item Intentional features (error messages, timing differences)
			      \item Normal application logic (a web app behaving differently based on decrypted data)
		      \end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{Real-world format-oracle attacks}{Padding oracle}
	\begin{columns}[c]
		\begin{column}{1\textwidth}
			\begin{itemize}[<+->]
				\item CBC mode requires padding to handle plaintexts that aren't block-aligned.
				\item Many implementations report invalid padding differently from other decryption failures (first exploited by Serge Vaudenay, 2002).
				\item This exposes an oracle that tells attackers: ``Does $\texttt{Dec}(K, C)$ have valid padding?''
				\item Attackers can systematically exploit this to decrypt arbitrary ciphertexts.
				\item This has led to major vulnerabilities in SSH and SSL/TLS protocols.
				\item Example: The POODLE attack against SSL 3.0 affected millions of websites.\footnote{\url{https://appliedcryptography.page/paper/\#google-poodle}}
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Real-world format-oracle attacks}{Timing side-channel}
	\begin{columns}[c]
		\begin{column}{1\textwidth}
			\begin{itemize}[<+->]
				\item The victim interprets the plaintext as a number $n$ and performs $n$ operations.
				\item Attackers can measure how long the system takes to respond.
				\item The response time reveals approximate numerical values inside $\texttt{Dec}(K, C)$.
				\item Extremely subtle: even microsecond differences can leak information.
				\item Successfully used to break older SSH and SSL/TLS implementations.
				\item Example: The Lucky Thirteen attack against TLS revealed message contents through timing differences.\footnote{\url{https://appliedcryptography.page/paper/\#lucky-thirteen}}
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Real-world format-oracle attacks}{Apple iMessage}
	\begin{columns}[c]
		\begin{column}{1\textwidth}
			\begin{itemize}[<+->]
				\item Older Apple iMessage versions encrypted the payload with AES-CTR (malleable, exactly as in our null-oracle attack) and gzip-compressed it first.
				\item The system responded differently when a ciphertext decrypted to:
				      \begin{itemize}
					      \item A valid gzip file (processed normally)
					      \item An invalid gzip file (error reported)
				      \end{itemize}
				\item This created an oracle revealing: ``Is $\texttt{Dec}(K, C)$ a valid gzip file?''\footnote{\url{https://appliedcryptography.page/paper/\#jhu-imessage}}
				\item Attackers who understood the gzip format could exploit this to:
				      \begin{itemize}
					      \item Silently recover private messages
					      \item Bypass encryption entirely
				      \end{itemize}
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Real-world format-oracle attacks}{XML format oracles}
	\begin{itemize}[<+->]
		\item Many systems decrypt data expecting valid XML format.
		\item If decrypted data isn't valid XML, the system returns an error.
		\item This exposes an oracle: ``Is $\texttt{Dec}(K, C)$ valid XML?''
		\item XML has complex syntax rules that attackers can exploit.
		\item Jager and Somorovsky (2011) did exactly this to the W3C XML Encryption standard: full plaintext recovery from CBC ciphertexts.
		\item Similar attacks exist for other formats (JSON, HTML, etc.).
	\end{itemize}
\end{frame}

\begin{frame}{I thought CTR mode was secure?}{CPA vs. CCA security}
	\begin{itemize}[<+->]
		\item CTR mode \textit{is} secure against chosen-plaintext attacks (CPA-secure).
		      \begin{itemize}
			      \item It uses randomness to ensure identical messages encrypt differently each time.
			      \item The adversary cannot distinguish its ciphertexts from random, even for chosen plaintexts.
		      \end{itemize}
		\item But CPA security isn't enough in many real-world scenarios!
		      \begin{itemize}
			      \item CPA security only considers attackers who can request encryptions.
			      \item It doesn't protect against attackers who can submit chosen ciphertexts.
		      \end{itemize}
		\item In the null-oracle attack:
		      \begin{itemize}
			      \item The victim decrypts ciphertexts chosen by the adversary.
			      \item Even leaking one bit about the plaintext (contains nulls or not) is fatal.
			      \item CPA security doesn't model or prevent this type of attack.
		      \end{itemize}
		\item This is why we need stronger security notions (CCA security).
	\end{itemize}
\end{frame}

\begin{frame}{CCA security: now you also get a decryption oracle}
	\begin{columns}[c]
		\begin{column}{0.5\textwidth}
			\definitionbox{CCA Security}{
				$\Sigma$ is \textbf{CCA-secure} if these two libraries are indistinguishable:
			}
			\begin{itemize}
				\item The adversary now gets a \emph{decryption} oracle too --- not just encryption.
				\item The ideal \lib{\Sigma}{cca-rand} answers its own ciphertexts from memory ($\mathcal{D}$); anything else is decrypted for real, so whatever the scheme leaks about tampered ciphertexts is on display.
			\end{itemize}
		\end{column}
		\begin{column}{0.5\textwidth}
			\sslinked{
				\sslibrary{\Sigma}{cca-real}{
					$K \twoheadleftarrow \Sigma.\mathcal{K}$\\[1em]
					\sslibrarysubroutine{cca.enc}{M}{
						$C \coloneq \Sigma.\texttt{Enc}(K, M)$\\
						return $C$
					}{1}\\[1em]
					\sslibrarysubroutine{cca.dec}{C}{
						return $\Sigma.\texttt{Dec}(K, C)$
					}{1}
				}{0.64}
			}{\approxeq}{
				\sslibrary{\Sigma}{cca-rand}{
					$K \twoheadleftarrow \Sigma.\mathcal{K}$\\[1em]
					\sslibrarysubroutine{cca.enc}{M}{
						$C \twoheadleftarrow \Sigma.\mathcal{C}(|M|)$\\
						$\mathcal{D}[C] \coloneq M$\\
						return $C$
					}{1}\\[1em]
					\sslibrarysubroutine{cca.dec}{C}{
						if $\mathcal{D}[C]$ defined: return $\mathcal{D}[C]$\\
						return $\Sigma.\texttt{Dec}(K, C)$
					}{1}
				}{0.64}
			}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Malleability breaks CCA --- the one you've already met}
	\begin{columns}[c]
		\begin{column}{0.55\textwidth}
			\begin{itemize}
				\item Take our CPA-secure PRF scheme. It is \textbf{not} CCA-secure.
				\item Recall CTR's malleability: flipping a bit in the ciphertext flips the \emph{same} bit in the plaintext. This scheme has it too --- $S$ is just $M$ XOR'd with a keystream.
				\item \textbf{The recipe} (any malleable scheme): find a ciphertext tweak that makes a \emph{predictable} change to the plaintext, then let the decryption oracle confirm it.
				\item Here: XOR any $\Delta$ into $S$, and the decryption comes back as $M \oplus \Delta$.
			\end{itemize}
		\end{column}
		\begin{column}{0.45\textwidth}
			\sssubroutine{enc}{K, M}{
				$R \twoheadleftarrow \bits^\lambda$\\
				$S \coloneq F(K, R) \oplus M$\\
				return $R\|S$
			}{1}
			\sssubroutine{dec}{K, R\|S}{
				$M \coloneq F(K, R) \oplus S$\\
				return $M$
			}{1}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{One bit of malleability already breaks CCA}
	\begin{columns}[c]
		\begin{column}{0.5\textwidth}
			\begin{itemize}
				\item Encrypt $M$, maul the ciphertext by XORing $\Delta$ into $S$, then ask for a decryption:
			\end{itemize}
			\ssprogram{}{
				$M \twoheadleftarrow \bits^\lambda$\\
				$\Delta \coloneq \text{arbitrary nonzero string}$\\
				$R\|S \coloneq \func{cca.enc}{M}$\\
				$M' \coloneq \func{cca.dec}{R\|(S \oplus \Delta)}$\\
				return $M' \stackrel{?}{=} M \oplus \Delta$
			}{0.9}
		\end{column}
		\begin{column}{0.5\textwidth}
			\begin{itemize}
				\item \textbf{Real library:} $M' = M \oplus \Delta$ every time, so it returns \texttt{true} with probability $1$.
				\item \textbf{Ideal (\lib{}{cca-rand}):} the mauled ciphertext isn't in $\mathcal{D}$, so $M'$ is unrelated to $M$ --- \texttt{true} with probability $\approx \frac{1}{2^\lambda}$.
				\item One predictable relationship is all it takes. \textbf{Not CCA-secure.}
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Forging ciphertexts you never sent}{The Frankenstein trick, part 1}
	\begin{columns}[c]
		\begin{column}{0.5\textwidth}
			\begin{itemize}
				\item A tougher scheme: $S = F(K, R) \oplus F(K, M)$, and decryption \emph{inverts} the PRF --- so $F$ must be a PRP like AES.
				\item No single-bit maul works here --- so we get sneakier.
				\item \textbf{The recipe}: request two encryptions (use the \emph{same} $M$), then mix and match their halves into ``Frankenstein'' ciphertexts.
			\end{itemize}
		\end{column}
		\begin{column}{0.5\textwidth}
			\sslibrary{}{cca-real}{
				$K \twoheadleftarrow \bits^\lambda$\\[1em]
				\sslibrarysubroutine{cca.enc}{M}{
					$R \twoheadleftarrow \bits^\lambda$\\
					$S \coloneq F(K, R) \oplus F(K, M)$\\
					return $R\|S$
				}{1}\\[1em]
				\sslibrarysubroutine{cca.dec}{R\|S}{
					$M \coloneq F^{-1}(K, F(K, R) \oplus S)$\\
					return $M$
				}{1}
			}{0.85}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Forging ciphertexts you never sent}{The Frankenstein trick, part 2}
	\begin{columns}[c]
		\begin{column}{0.55\textwidth}
			\begin{itemize}
				\item Encrypt the same $M$ twice: $R_1\|S_1$ and $R_2\|S_2$.
				\item Swap the halves and decrypt $R_1\|S_2$ and $R_2\|S_1$. Both unwrap to the \emph{same} value:
				      \[ F^{-1}\big(K,\; F(K, R_1) \oplus F(K, R_2) \oplus F(K, M)\big). \]
				\item Equal outputs can't happen in the ideal world (only w.p.\ $\approx \frac{1}{2^\lambda}$). Distinguished --- and forged.
			\end{itemize}
		\end{column}
		\begin{column}{0.45\textwidth}
			\ssprogram{}{
				$M \twoheadleftarrow \bits^\lambda$\\
				$R_1\|S_1 \coloneq \func{cca.enc}{M}$\\
				$R_2\|S_2 \coloneq \func{cca.enc}{M}$\\
				$M_1^* \coloneq \func{cca.dec}{R_1\|S_2}$\\
				$M_2^* \coloneq \func{cca.dec}{R_2\|S_1}$\\
				return $M_1^* \stackrel{?}{=} M_2^*$
			}{0.95}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Why forgeries are worse than leaks}
	\begin{columns}[c]
		\begin{column}{0.55\textwidth}
			\begin{itemize}
				\item Reading plaintext is bad. \textbf{Minting valid ciphertexts is worse.}
				\item Suppose a bank encrypts transactions with the Frankenstein scheme:
				      \begin{itemize}
					      \item The attacker intercepts a few encrypted transactions.
					      \item Mixing their halves forges ciphertexts that decrypt \emph{cleanly} to transactions nobody ever authorized.
					      \item The system has no way to say \texttt{err}, so it accepts them.
				      \end{itemize}
				\item Both attacks ride on the same flaw: \textbf{malleability}.
			\end{itemize}
		\end{column}
		\begin{column}{0.45\textwidth}
			\definitionbox{The fix}{
				Reject any ciphertext we didn't create ourselves.\\[0.5em]
				That is exactly what \textbf{authenticated encryption} gives us, and it's where we're headed next.
			}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Symmetric primitive example: hash functions}{Reminder}
	\begin{columns}[c]
		\begin{column}{0.55\textwidth}
			\definitionbox{Hash Function Properties}{
				\begin{itemize}\item Takes input of \textbf{any size}
					\item Produces output of \textbf{fixed size}
					\item Is \textbf{deterministic} (same input $\rightarrow$ same output)
					\item Even a \textbf{tiny change} in input creates completely different output
					\item Is \textbf{efficient} to compute\end{itemize}
			}
		\end{column}
		\begin{column}{0.45\textwidth}
			\begin{tcolorbox}
				[colback=black!5!white,colframe=ciphergray] $\mathsf{SHA256}(\texttt{hello}) =$ \\ {\small\texttt{2cf24dba5fb0a30e26e83b2ac5}\\ \texttt{b9e29e1b161e5c1fa7425e7304}\\
					\texttt{3362938b9824}}\\[1em]
				$\mathsf{SHA256}(\texttt{hullo}) =$ \\ {\small\texttt{7835066a1457504217688c8f5d}\\
					\texttt{06909c6591e0ca78c254ccf174}\\ \texttt{50d0d999cab0}}
			\end{tcolorbox}
			\textcolor{cipherprimary}{\textbf{Note:} \small One character change $\rightarrow$
				completely different hash!}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Expected properties of a hash function}{Reminder}
	\begin{columns}[c]
		\begin{column}{0.6\textwidth}
			\begin{itemize}
				\item \textbf{Collision resistance}: computationally infeasible to find
				      two different inputs producing the same hash.
				\item \textbf{Preimage resistance}: given an output of a hash function,
				      it is computationally infeasible to find \emph{any} input that hashes to it.
				\item \textbf{Second preimage resistance}: given an input and an output,
				      it's computationally infeasible to find a different input
				      producing the same output.
			\end{itemize}
		\end{column}
		\begin{column}{0.4\textwidth}
			\imagewithcaption{sha2.png}{SHA-2 compression function. Source: Wikipedia}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Hash functions: what are they good for?}{Reminder}
	\begin{itemize}
		\item \textbf{Data integrity verification}: Hash a file. Later hash it
		      again and compare hashes to check if the file has changed, suffered storage
		      degradation, etc.
		\item \textbf{Proof of work}: Server asks client to hash something a lot of
		      times before they can access some resource. Useful for anti-spam, Bitcoin
		      mining, etc.
	\end{itemize}
\end{frame}

\begin{frame}{Data integrity: verifying a Linux ISO download}
	\begin{columns}[c]
		\begin{column}{0.60\textwidth}
			\begin{itemize}
				\item Linux distributions publish a hash (usually SHA-256) for each ISO image they release.
				\item After downloading, you hash your own copy and compare it to the published value:
				      \begin{center}
					      {\small $\mathsf{SHA256}(\texttt{your.iso}) \stackrel{?}{=} \texttt{1620295f\ldots}$}
				      \end{center}
				\item A match means the file arrived \textbf{intact}: no corruption from a flaky network or a failing disk.
				\item This only catches \emph{accidental} corruption: a malicious mirror could swap both the ISO \emph{and} the hash, which is why real verification also uses a signature (OpenPGP).
			\end{itemize}
		\end{column}
		\begin{column}{0.40\textwidth}
			\imagewithcaption{fedora-iso-hash.png}{Fedora's download page showing the expected SHA-256 checksum.}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Proof of work: Anubis}
	\begin{columns}[c]
		\begin{column}{0.60\textwidth}
			\begin{itemize}
				\item Anubis shields web servers from aggressive bots (e.g.\ AI scrapers) by making each visitor's browser solve a hash puzzle.
				\item \textbf{The challenge}: given a value $C$, find a nonce $N$ such that $\mathsf{SHA256}(C \| N)$ begins with a set number of leading zero bits (the \emph{difficulty}).
				\item Finding $N$ takes brute force over many guesses, but checking a submitted $N$ is one hash: \textbf{hard to solve, easy to verify}.
				\item A human barely notices the delay; a bot making millions of requests pays the cost every time.
			\end{itemize}
		\end{column}
		\begin{column}{0.40\textwidth}
			\imagewithcaption{anubis.png}{Anubis running a proof-of-work challenge in the browser before granting access.}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Message authentication codes}
	\begin{columns}[c]
		\begin{column}{0.5\textwidth}
			\definitionbox{Message Authentication Code (MAC)}{
				A MAC is a cryptographic function that takes a key $K$ and a message $M$ and produces a tag $T$ that authenticates the message. Only someone with the same key can verify the tag.
			}
		\end{column}
		\begin{column}{0.5\textwidth}
			\begin{itemize}
				\item A MAC provides \textbf{integrity} and \textbf{authenticity} for messages.
				\item Unlike hash functions, MACs need a secret key: anyone can recompute a SHA-256 checksum; nobody can recompute a tag without $K$.
				\item MACs address the malleability problem we just saw: without one, attackers modify ciphertexts at will.
				\item A secure MAC should be unforgeable, even after seeing tags for chosen messages.
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{A super basic practical example: AES-CTR + HMAC}
	\begin{columns}[c]
		\begin{column}{0.5\textwidth}
			\begin{itemize}
				\item Recall: AES-CTR alone is CPA-secure but \textbf{malleable}, so it's not CCA-secure.
				\item Fix with \textbf{Encrypt-then-MAC} using HMAC-SHA256:
				      \begin{itemize}
					      \item Pick two independent keys: $K_e$ for AES-CTR, $K_m$ for HMAC.
					      \item Encrypt with AES-CTR, then HMAC the \textit{ciphertext}.
					      \item On decryption, verify the tag \textbf{before} decrypting.
				      \end{itemize}
				\item Now any tampered ciphertext fails tag verification and is rejected.
			\end{itemize}
		\end{column}
		\begin{column}{0.5\textwidth}
			\sssubroutine{Enc}{(K_e, K_m), M}{
				$C \coloneq \texttt{AES-CTR.Enc}(K_e, M)$\\
				$T \coloneq \texttt{HMAC-SHA256}(K_m, C)$\\
				return $C\|T$
			}{1}
			\sssubroutine{Dec}{(K_e, K_m), C\|T}{
				if $\texttt{HMAC-SHA256}(K_m, C) \neq T$:\\
				$\quad$ return $\texttt{err}$\\
				return $\texttt{AES-CTR.Dec}(K_e, C)$
			}{1}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{PRFs as MACs}
	\begin{columns}[c]
		\begin{column}{0.45\textwidth}
			\begin{itemize}
				\item A pseudorandom function (PRF) can be used directly as a MAC!
				\item The MAC key is the PRF key $K$.
				\item To authenticate a message $X$:
				      \begin{itemize}
					      \item Compute the tag $T = F(K, X)$.
					      \item Send both $X$ and $T$ to the recipient.
				      \end{itemize}
				\item Why it works: for a never-tagged $X$, $F(K, X)$ looks uniformly random, so a forged tag is right w.p.\ $\frac{1}{2^\lambda}$.
			\end{itemize}
		\end{column}
		\begin{column}{0.55\textwidth}
			\begin{flushright}
				\sslinked{
					\sslibrary{}{mac-real}{
						$K \twoheadleftarrow \bits^\lambda$\\[1em]
						\sslibrarysubroutine{mac.guess}{X, Y}{
							return $Y \stackrel{?}{=} F(K, X)$
						}{1}\\[1em]
						\sslibrarysubroutine{mac.reveal}{X}{
							return $F(K, X)$
						}{1}
					}{0.7}
				}{\approxeq}{
					\sslibrary{}{mac-ideal}{
						\sslibrarysubroutine{mac.guess}{X, Y}{
							if $L[X]$ undefined: return $\texttt{false}$\\
							return $Y \stackrel{?}{=} L[X]$
						}{1}\\[1em]
						\sslibrarysubroutine{mac.reveal}{X}{
							if $L[X]$ undefined: $L[X] \twoheadleftarrow \bits^\lambda$\\
							return $L[X]$
						}{1}
					}{0.7}
				}
			\end{flushright}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Encrypt-then-MAC: formal construction}
	\begin{columns}[c]
		\begin{column}{0.45\textwidth}
			\definitionbox{Encrypt-then-MAC Construction}{
				Let $\Sigma$ be an SKE scheme and $F$ be a PRF with output length $\lambda$ whose domain includes $\Sigma.\mathcal{C}$.
				Define a new encryption scheme:
				\begin{align*}
					\mathcal{K}       & = \Sigma.\mathcal{K} \times \bits^\lambda       \\
					\mathcal{M}       & = \Sigma.\mathcal{M}                            \\
					\mathcal{C}(\ell) & = \Sigma.\mathcal{C}(\ell) \times \bits^\lambda
				\end{align*}
			}
		\end{column}
		\begin{column}{0.55\textwidth}
			\sssubroutine{Enc}{(K_e, K_m), M}{
				$C \coloneq \Sigma.\texttt{Enc}(K_e, M)$\\
				$T \coloneq F(K_m, C)$\\
				return $C\|T$
			}{1}
			\sssubroutine{Dec}{(K_e, K_m), C\|T}{
				if $F(K_m, C) \neq T$: return $\texttt{err}$\\
				return $\Sigma.\texttt{Dec}(K_e, C)$
			}{1}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{CCA security of Encrypt-then-MAC}
	\begin{columns}[c]
		\begin{column}{0.45\textwidth}
			\definitionbox{CCA Security Claim}{
				If $\Sigma$ is a CPA-secure encryption scheme and $F$ is a secure PRF, then the Encrypt-then-MAC construction is CCA-secure.
			}
			\begin{itemize}
				\item Without the MAC key, the adversary can't create valid tags.
				\item \textbf{Decryption oracle only returns plaintexts for ciphertexts with valid tags.}
				\end{itemize}
		\end{column}
		\begin{column}{0.55\textwidth}
			\sslinked{
				\sslibrary{}{cca-real}{
					$K_e \twoheadleftarrow \Sigma.\mathcal{K}$\\
					$K_m \twoheadleftarrow \bits^\lambda$\\[1em]
					\sslibrarysubroutine{cca.enc}{M}{
						$C \coloneq \Sigma.\texttt{Enc}(K_e, M)$\\
						$T \coloneq F(K_m, C)$\\
						return $C\|T$
					}{1}\\[1em]
					\sslibrarysubroutine{cca.dec}{C\|T}{
						if $F(K_m, C) \neq T$: return $\texttt{err}$\\
						return $\Sigma.\texttt{Dec}(K_e, C)$
					}{1}
				}{0.63}
			}{\approxeq}{
				\sslibrary{}{cca-rand}{
					$K_e \twoheadleftarrow \Sigma.\mathcal{K}$\\
					$K_m \twoheadleftarrow \bits^\lambda$\\[1em]
					\sslibrarysubroutine{cca.enc}{M}{
						$C \twoheadleftarrow \Sigma.\mathcal{C}(|M|)$\\
						$T \twoheadleftarrow \bits^\lambda$\\
						$\mathcal{D}[C\|T] \coloneq M$\\
						return $C\|T$
					}{1}\\[1em]
					\sslibrarysubroutine{cca.dec}{C\|T}{
						if $\mathcal{D}[C\|T]$ defined: return $\mathcal{D}[C\|T]$\\
						if $F(K_m, C) \neq T$: return $\texttt{err}$\\
						return $\Sigma.\texttt{Dec}(K_e, C)$
					}{1}
				}{0.63}
			}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Encrypt-then-MAC is CCA-secure}{The proof in three moves}
	\begin{enumerate}
		\item \textbf{Replace the MAC by a random function.} $F$ is a secure PRF, so swapping $F(K_m, \cdot)$ for a uniformly random function changes nothing the distinguisher can see.
		\item \textbf{Neuter the decryption oracle.} A random function is uniform on fresh inputs, so a $C\|T$ the library never produced passes the tag check with probability $\frac{1}{2^\lambda}$ per query. Bad-event argument: replace ``check the tag'' by ``return \texttt{err} unless we produced it''.
		\item \textbf{We are in the CPA game.} The decryption oracle now only echoes what the encryption oracle said, so every remaining call is an encryption of a chosen plaintext: CPA security of $\Sigma$ lets us swap real ciphertexts for random ones, and the tags are already random from move 1.
	\end{enumerate}
	\vspace{0.5em}
	Three hybrids we have already practised; chain them and \lib{}{cca-real} $\approxeq$ \lib{}{cca-rand}.
\end{frame}

\begin{frame}{Combining MACs and encryption}
	\begin{itemize}
		\item Not every way of combining a MAC and CPA-secure encryption achieves CCA security.
		\item There are three common approaches to combining them:
		      \begin{itemize}
			      \item \textbf{Encrypt-then-MAC}: Encrypt message, then MAC the ciphertext.
			      \item \textbf{Encrypt-and-MAC}: Encrypt message and MAC the plaintext separately.
			      \item \textbf{MAC-then-encrypt}: MAC the plaintext, then encrypt both message and tag.
		      \end{itemize}
		\item Only one of them guarantees CCA security for \emph{every} CPA-secure encryption scheme and secure MAC (Bellare and Namprempre, 2000).
	\end{itemize}
\end{frame}

\begin{frame}{Encrypt-then-MAC}{CCA-secure}
	\begin{columns}[c]
		\begin{column}{0.5\textwidth}
			\begin{itemize}
				\item \textbf{MAC verifies ciphertext integrity} before decryption.
				\item Prevents attackers from submitting modified ciphertexts.
				\item \textbf{Always CCA-secure} if encryption is CPA-secure and MAC is secure.
			\end{itemize}
		\end{column}
		\begin{column}{0.5\textwidth}
			\sssubroutine{Enc}{(K_e, K_m), M}{
				$C \coloneq \Sigma.\texttt{Enc}(K_e, M)$\\
				$T \coloneq F(K_m, C)$\\
				return $C\|T$
			}{1}
			\sssubroutine{Dec}{(K_e, K_m), C\|T}{
				if $F(K_m, C) \neq T$: return $\texttt{err}$\\
				return $\Sigma.\texttt{Dec}(K_e, C)$
			}{1}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Encrypt-and-MAC}{Not even CPA-secure!}
	\begin{columns}[c]
		\begin{column}{0.5\textwidth}
			\begin{itemize}
				\item \textbf{MAC is computed on the plaintext}, and our MAC (a PRF) is deterministic.
				\item Same plaintext $\Rightarrow$ same tag: our old ``encrypt $M$ twice, compare'' distinguisher wins again.
				\item \textbf{Not even CPA-secure}, let alone CCA-secure.
				\item SSH's packet format is Encrypt-and-MAC; it gets away with it by mixing a sequence number into the MAC input.
			\end{itemize}
		\end{column}
		\begin{column}{0.5\textwidth}
			\sssubroutine{Enc}{(K_e, K_m), M}{
				$C \coloneq \Sigma.\texttt{Enc}(K_e, M)$\\
				$T \coloneq F(K_m, M)$\\
				return $C\|T$
			}{1}
			\sssubroutine{Dec}{(K_e, K_m), C\|T}{
				$M \coloneq \Sigma.\texttt{Dec}(K_e, C)$\\
				if $F(K_m, M) \neq T$: return $\texttt{err}$\\
				return $M$
			}{1}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{MAC-then-encrypt}{It's complicated}
	\begin{columns}[c]
		\begin{column}{0.5\textwidth}
			\begin{itemize}
				\item \textbf{Tag is hidden inside the ciphertext}, so padding checks and parsers run on attacker-controlled input \emph{before} anything is verified: hello again, format oracles.
				\item \textbf{Sometimes CCA-secure} (e.g.\ CTR-style encryption), \textbf{sometimes not}: case-by-case analysis.
				\item TLS 1.0--1.2 with CBC did this (MAC, pad, encrypt); POODLE and Lucky Thirteen are the bill.
			\end{itemize}
		\end{column}
		\begin{column}{0.5\textwidth}
			\sssubroutine{Enc}{(K_e, K_m), M}{
				$T \coloneq F(K_m, M)$\\
				$C \coloneq \Sigma.\texttt{Enc}(K_e, M\|T)$\\
				return $C$
			}{1}
			\sssubroutine{Dec}{(K_e, K_m), C}{
				$M\|T \coloneq \Sigma.\texttt{Dec}(K_e, C)$\\
				if $F(K_m, M) \neq T$: return $\texttt{err}$\\
				return $M$
			}{1}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Encrypt + MAC security comparison}
	\begin{center}
		\begin{tabular}{|l|c|c|}
			\hline
			\textbf{Construction} & \textbf{CPA-secure?} & \textbf{CCA-secure?} \\
			\hline
			Encrypt-then-MAC      & Yes                  & Yes                  \\
			\hline
			Encrypt-and-MAC       & No                   & No                   \\
			\hline
			MAC-then-encrypt      & Yes                  & Maybe                \\
			\hline
		\end{tabular}
	\end{center}
	\vspace{1em}
	\begin{itemize}
		\item \textbf{Encrypt-then-MAC} is the safest option.
		\item \textbf{Encrypt-and-MAC} should never be used.
		\item \textbf{MAC-then-encrypt} requires case-by-case analysis.
	\end{itemize}
\end{frame}

\subsection{Authenticated Encryption}

\begin{frame}{Authenticated encryption: beyond CCA security}
	\begin{itemize}[<+->]
		\item CCA security is stronger than CPA security, but still not the gold standard.
		\item CCA security says: adversary-generated ciphertexts won't reveal useful information.
		\item But CCA security doesn't require that they decrypt to \texttt{err}.
		\item For many applications, we want a stronger guarantee:
		      \begin{itemize}
			      \item Only key-holders can create valid ciphertexts.
			      \item All other ciphertexts should be rejected as invalid.
		      \end{itemize}
		\item This property is called \textbf{Authenticated Encryption (AE)}.
		\item Authenticated encryption provides both confidentiality and authenticity.
	\end{itemize}
\end{frame}

\begin{frame}{Authenticated encryption: formal definition}
	\begin{columns}[c]
		\begin{column}{0.4\textwidth}
			\definitionbox{Authenticated Encryption}{
				An SKE scheme $\Sigma$ is a secure authenticated encryption (AE) scheme if the following two libraries are indistinguishable:
			}
		\end{column}
		\begin{column}{0.6\textwidth}
			\begin{flushright}
				\sslinked{
					\sslibrary{\Sigma}{ae-real}{
						$K \twoheadleftarrow \Sigma.\mathcal{K}$\\[1em]
						\sslibrarysubroutine{ae.enc}{M}{
							return $\Sigma.\texttt{Enc}(K, M)$
						}{1}\\[1em]
						\sslibrarysubroutine{ae.dec}{C}{
							return $\Sigma.\texttt{Dec}(K, C)$
						}{1}
					}{0.75}
				}{\approxeq}{
					\sslibrary{\Sigma}{ae-rand}{
						\sslibrarysubroutine{ae.enc}{M}{
							$C \twoheadleftarrow \Sigma.\mathcal{C}(|M|)$\\
							$\mathcal{D}[C] \coloneq M$\\
							return $C$
						}{1}\\[1em]
						\sslibrarysubroutine{ae.dec}{C}{
							if $\mathcal{D}[C]$ defined: return $\mathcal{D}[C]$\\
							else: return $\texttt{err}$
						}{1}
					}{0.75}
				}
			\end{flushright}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{AE vs. CCA security}
	\begin{columns}[c]
		\begin{column}{0.5\textwidth}
			\begin{itemize}[<+->]
				\item Key difference: \textbf{how we handle adversary-created ciphertexts}.
				\item In \lib{\Sigma}{ae-rand}, any ciphertext not created by the library always decrypts to \texttt{err}.
				\item In \lib{\Sigma}{cca-rand}, such ciphertexts could decrypt to anything (not necessarily \texttt{err}).
				\item So AE requires:
				      \begin{itemize}
					      \item Adversary cannot tell real from random ciphertexts (as in CPA).
					      \item Adversary cannot create new valid ciphertexts (authentication).
				      \end{itemize}
			\end{itemize}
		\end{column}
		\begin{column}{0.5\textwidth}
			\begin{itemize}[<+->]
				\item AE is \textbf{strictly stronger} than CCA security.
				\item Every AE scheme is CCA-secure, but not every CCA-secure scheme is an AE scheme.
				\item Separating example: an AE scheme whose \texttt{Dec} outputs $\bit{0}^\lambda$ instead of \texttt{err} is still CCA-secure (the zeros leak nothing), but no longer AE.
				\item AE is what you should aim for in practice.
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{How Encrypt-then-MAC achieves AE}
	\begin{columns}[c]
		\begin{column}{0.6\textwidth}
			\begin{itemize}[<+->]
				\item Remember our Encrypt-then-MAC construction:
				      \begin{itemize}
					      \item Encrypt the plaintext: $C \coloneq \Sigma.\texttt{Enc}(K_e, M)$
					      \item MAC the ciphertext: $T \coloneq F(K_m, C)$
					      \item Send both: $C\|T$.
				      \end{itemize}
				\item It achieves AE security because:
				      \begin{itemize}
					      \item Without $K_m$, adversary can't forge valid tags.
					      \item Any ciphertext not created by the system will fail MAC verification.
					      \item MAC verification failures lead to \texttt{err}.
				      \end{itemize}
				\item The proof is nearly identical to the CCA security proof.
			\end{itemize}
		\end{column}
		\begin{column}{0.4\textwidth}
			\definitionbox{AE Security of Encrypt-then-MAC}{
				Encrypt-then-MAC is a secure AE if the underlying $\Sigma$ is a CPA-secure SKE and $F$ is a secure PRF.
			}
			\vspace{0.5em}
			\definitionbox{AE implies CCA}{
				If an encryption scheme $\Sigma$ is a secure AE, then it is also CCA-secure.
			}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Authenticated encryption: in practice}
	\begin{itemize}[<+->]
		\item Modern cryptographic protocols almost always use authenticated encryption.
		\item Common AE implementations:
		      \begin{itemize}
			      \item \textbf{AES-GCM} (Galois/Counter Mode): most widely used, combines CTR mode with a MAC.
			      \item \textbf{ChaCha20-Poly1305}: popular alternative to AES-GCM, especially on devices without AES hardware.
			      \item \textbf{AES-CBC + HMAC-SHA256}: older approach, uses Encrypt-then-MAC with AES in CBC mode.
		      \end{itemize}
		\item Important implementation rule: \textbf{verify before decrypt}!
		      \begin{itemize}
			      \item Always check the MAC before decrypting.
			      \item Prevents timing side-channels based on decryption behavior.
			      \item Helps protect against padding oracle attacks and similar vulnerabilities.
		      \end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{AES-GCM: Galois/Counter Mode}
	\begin{columns}[c]
		\begin{column}{0.6\textwidth}
			\begin{itemize}[<+->]
				\item AES-GCM (Galois/Counter Mode) is the most widely used AEAD scheme.
				\item Combines AES in CTR mode (for encryption) with a polynomial MAC, GHASH (for authentication).
				\item Extremely efficient:
				      \begin{itemize}
					      \item Single pass over the data.
					      \item \textbf{Parallelizable}.
					      \item Hardware acceleration widely available.
				      \end{itemize}
				\item Used in: TLS 1.2/1.3, IPsec, SSH, and many other protocols.
			\end{itemize}
		\end{column}
		\begin{column}{0.4\textwidth}
			\begin{center}
				\sssubroutine{AES-GCM.Enc}{K, N, A, M}{
					$H \coloneq \texttt{AES}(K, \bit{0}^{128})$\\
					$\text{for } i=1 \text{ to } \lceil|M|/128\rceil:$\\
					$\quad C_i \coloneq M_i \oplus \texttt{AES}(K, N\|(i+1))$\\
					$T \coloneq \texttt{GHASH}_H(A, C) \oplus$\\
					$\quad\quad \texttt{AES}(K, N\|1)$\\
					$\textbf{return } C\|T$
				}{0.87}
			\end{center}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{AES-GCM: Galois/Counter Mode}
	\begin{columns}[c]
		\begin{column}{0.5\textwidth}
			\begin{itemize}
				\item \textbf{Inputs}:
				      \begin{itemize}
					      \item Key $K$ (128, 192, or 256 bits)
					      \item Nonce $N$ (usually 96 bits)
					      \item Associated data $A$ (optional)
					      \item Plaintext $M$
				      \end{itemize}
				\item \textbf{Encryption process}:
				      \begin{itemize}
					      \item AES-CTR for confidentiality; GHASH (Galois-field\footnote{Another name for finite fields; we meet them again in 1.7 with Diffie-Hellman.} multiplication) for authentication.
				      \end{itemize}
				\item The tag $T$ protects both ciphertext and associated data.
			\end{itemize}
		\end{column}
		\begin{column}{0.5\textwidth}
			\imagewithcaption[0.85\textwidth]{gcm.pdf}{Source: The Joy of Cryptography}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{What is GHASH?}
	\begin{columns}[c]
		\begin{column}{0.65\textwidth}
			\begin{itemize}
				\item \textbf{GHASH} is a universal hash function based on Galois field arithmetic.
				\item Key properties:
				      \begin{itemize}
					      \item Operates in the finite field $GF(2^{128})$ (see box).
				      \end{itemize}
				\item How it works:
				      \begin{itemize}
					      \item Treats input blocks as coefficients of a polynomial.
					      \item Evaluates the polynomial at a secret point $H = \texttt{AES}(K, \bit{0}^{128})$.
					      \item Result: $\text{GHASH}_H(X) = X_1 \cdot H^n + X_2 \cdot H^{n-1} + \ldots + X_n \cdot H$.
					      \item Two distinct $n$-block inputs collide at a random $H$ w.p.\ at most $n/2^{128}$: a degree-$n$ polynomial has at most $n$ roots.
				      \end{itemize}
			\end{itemize}
		\end{column}
		\begin{column}{0.35\textwidth}
			\definitionbox{Galois Field $GF(2^{128})$}{
				\begin{itemize}
					\item Elements: 128-bit strings
					\item Addition: XOR
					\item Multiplication: polynomial multiplication mod $x^{128} + x^7 + x^2 + x + 1$
					\item Every non-zero element has an inverse
				\end{itemize}
			}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Why GHASH instead of HMAC?}
	\begin{columns}[c]
		\begin{column}{0.65\textwidth}
			\begin{itemize}
				\item \textbf{Why not use HMAC-SHA256 or similar?}
				      \begin{itemize}
					      \item HMAC would work, but it's much slower.
					      \item HMAC requires two nested hash computations.
					      \item SHA-256 is inherently sequential.
				      \end{itemize}
				\item \textbf{GHASH advantages:}
				      \begin{itemize}
					      \item Single multiplication per block (after setup)
					      \item \textbf{Fully parallelizable} across blocks
					      \item Hardware acceleration: Intel \texttt{PCLMULQDQ} instruction
					      \item Can process at line speed (10+ Gbps)
				      \end{itemize}
				\item \textbf{Provable, with one assumption}: the polynomial part is unconditional (root counting); only the mask needs AES to be a good PRP.
			\end{itemize}
		\end{column}
		\begin{column}{0.35\textwidth}
			\definitionbox{Ballpark speeds}{
				On a modern x86 core:
				\begin{itemize}
					\item \textbf{HMAC-SHA256} (no SHA extensions): several cycles per byte
					\item \textbf{GHASH} with \texttt{PCLMULQDQ}: a fraction of a cycle per byte
				\end{itemize}
				\vspace{0.5em}
				An order of magnitude, and it parallelizes.
			}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{If GHASH is so great, why not use it everywhere?}
	\begin{columns}[c]
		\begin{column}{1\textwidth}
			\begin{itemize}
				\item \textbf{GHASH limitations:}
				      \begin{itemize}
					      \item \textbf{Needs a secret key}: without $H$, collisions are trivial to find. It is not a collision-resistant hash.
					      \item \textbf{Linear}: $\text{GHASH}_H(A \oplus B) = \text{GHASH}_H(A) \oplus \text{GHASH}_H(B)$ for equal-length inputs.
					      \item \textbf{Key exposure is total}: recover $H$ once (a forgery, a reused nonce) and every future tag is forgeable.\footnote{\url{https://appliedcryptography.page/paper/\#universal-hash}}
				      \end{itemize}
				\item \textbf{When we need SHA-2/SHA-3:}
				      \begin{itemize}
					      \item \textbf{Public verification}: digital signatures, certificates, Git commits
					      \item \textbf{Proof of work}: Bitcoin mining, anti-spam
					      \item \textbf{Key derivation}: generating keys from passwords (PBKDF2)
				      \end{itemize}
				\item \textbf{Different tools for different jobs}: GHASH is a fast MAC when you hold a key; SHA-2/3 when anyone must be able to verify.
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{AES-GCM: Galois/Counter Mode}{Security, and the rules you must not break}
	\begin{columns}[c]
		\begin{column}{1\textwidth}
			\begin{itemize}
				\item \textbf{Security strengths}:
				      \begin{itemize}
					      \item Provides confidentiality, integrity, and authenticity.
					      \item Formally proven secure (assuming AES is secure).
					      \item Fast and widely trusted.
				      \end{itemize}
				\item \textbf{Critical implementation requirements}:
				      \begin{itemize}
					      \item \textbf{Never reuse a nonce} with the same key!
					      \item One repeated nonce leaks the XOR of two plaintexts (CTR) \emph{and} the GHASH key $H$ (Joux's ``forbidden attack''), after which every tag is forgeable.
					      \item 96-bit nonces are recommended (other sizes are hashed first, costing time and security margin).
					      \item Use the full 128-bit authentication tag (don't truncate it).
				      \end{itemize}
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Key commitment in authenticated encryption}
	\begin{itemize}
		\item \textbf{Key commitment}: a ciphertext should decrypt validly under \emph{one} key only --- the one that produced it.
		\item AES-GCM and ChaCha20-Poly1305 do \textbf{not} have it: their tags are polynomial equations in the key, so given $K_1$ and $K_2$ one can solve for a ciphertext that verifies under both.\footnote{\url{https://appliedcryptography.page/paper/\#key-commitment}}
		\item Attack: craft one ciphertext $C$ that decrypts to a harmless message under $K_1$ and to malicious content under $K_2$. Whoever holds which key sees a different, fully ``authenticated'' plaintext.
		\item Real casualties: Facebook Messenger's abuse-reporting system (the ``invisible salamanders'' attack, 2018) and password-guessing ``partitioning oracles'' against Shadowsocks (2021).
		\item Fixes exist (commit to the key with a hash, or use a committing AEAD). The lesson: ``authenticated'' does not mean ``bound to the key you think''.
	\end{itemize}
\end{frame}

\begin{frame}{ChaCha20-Poly1305}
	\begin{columns}[c]
		\begin{column}{0.55\textwidth}
			\begin{itemize}
				\item ChaCha20-Poly1305 is a modern AEAD construction:
				      \begin{itemize}
					      \item ChaCha20 stream cipher for encryption.
					      \item Poly1305 MAC for authentication.
				      \end{itemize}
				\item Designed by Daniel J. Bernstein.\footnote{Does the class want to hear about Bernstein vs. United States?}
				\item Key characteristics:
				      \begin{itemize}
					      \item No table lookups (better resistance to timing attacks).
					      \item Excellent performance on devices without AES hardware.
				      \end{itemize}
				\item Widely used in TLS 1.3, SSH, WireGuard\ldots
			\end{itemize}
		\end{column}
		\begin{column}{0.45\textwidth}
			\begin{center}
				\sssubroutine{ChaCha20-Poly1305.Enc}{K, N, A, M}{
					$\text{key}_p \coloneq \texttt{ChaCha20}(K, N, 0)_{0..31}$\\
					$C \coloneq M \oplus \texttt{ChaCha20}(K, N, 1)$\\
					$\text{data} \coloneq \text{pad}(A) \| \text{pad}(C) \|$\\
					$\quad\quad \text{len}(A) \| \text{len}(C)$\\
					$T \coloneq \texttt{Poly1305}_{\text{key}_p}(\text{data})$\\
					$\textbf{return } C\|T$
				}{0.87}
			\end{center}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Replay attacks}{Authenticated encryption didn't save us}
	\begin{itemize}
		\item Even with authenticated encryption, \textbf{context matters}.
		\item The scenario in the pictures:
		      \begin{itemize}
			      \item Bob asks Alice which file to display; she answers with $C = \texttt{Enc}(K, \texttt{family-recipes.txt})$.
			      \item Later Bob asks which file to \emph{delete}. The adversary replays Alice's $C$.
			      \item $C$ is genuine, so it verifies --- and Bob deletes the family recipes.
		      \end{itemize}
		\item Nothing was forged. Bob trusted any ciphertext that decrypts successfully, and the ciphertext said nothing about \emph{which question} it was answering.
	\end{itemize}
\end{frame}

\begin{frame}{Associated data}
	\begin{itemize}
		\item Solution 1: Include context in the plaintext.
		      \begin{itemize}
			      \item ``\texttt{ACTION: DISPLAY}'' before the actual message.
			      \item Inefficient: increases message size.
			      \item Both parties already know the context.
		      \end{itemize}
		\item Better solution: Associated Data (AD).
		      \begin{itemize}
			      \item Context that sender and receiver already know.
			      \item Used during encryption and decryption.
			      \item Doesn't increase ciphertext size.
		      \end{itemize}
		\item How it works:
		      \begin{itemize}
			      \item $\texttt{Enc}(K, A, M) \rightarrow C$ where $A$ is associated data.
			      \item $\texttt{Dec}(K, A, C) \rightarrow M$ or $\texttt{err}$.
		      \end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{AEAD: formal definition}
	\begin{columns}[c]
		\begin{column}{0.5\textwidth}
			\definitionbox{AEAD}{

\begin{frame}{Usefulness of AEADs}
	\begin{itemize}
		\item An AEAD (Authenticated Encryption with Associated Data) scheme guarantees:
		      \begin{itemize}
			      \item Ciphertexts reveal nothing about plaintexts (confidentiality).
			      \item Only ciphertexts created by someone holding the key will decrypt without error (authenticity).
			      \item Decryption only succeeds when the correct associated data is used (context binding).
		      \end{itemize}
		\item What to use as associated data?
		      \begin{itemize}
			      \item Protocol information: ``\texttt{DISPLAY}'' vs. ``\texttt{DELETE}''.
			      \item Session identifiers or timestamps.
			      \item Previous messages in the conversation.
			      \item Any contextual information both parties already know.
		      \end{itemize}
		\item Use as much associated data as relevant: it's cryptographically ``free''!
	\end{itemize}
\end{frame}

\begin{frame}{What to take away}
	\begin{enumerate}
		\item \textbf{Deterministic encryption cannot be CPA-secure.} Randomness (an IV or a nonce) is what makes CBC and CTR work; ECB has none.
		\item \textbf{CPA security says nothing about attackers who send you ciphertexts.} Malleability plus a one-bit oracle recovers the whole plaintext at 255 queries per byte.
		\item \textbf{The fix is to refuse what you did not create}: a MAC over the ciphertext (Encrypt-then-MAC), checked before anything else.
		\item \textbf{Authenticated encryption is the actual target}, and associated data binds a ciphertext to its context (display vs.\ delete).
		\item \textbf{In practice}: AES-GCM or ChaCha20-Poly1305; never reuse a nonce, never truncate the tag, and remember that neither commits to the key.
	\end{enumerate}
\end{frame}

\begin{frame}{The Fujisaki-Okamoto transform}{From CPA to CCA in public-key encryption (optional slide)}
	\begin{columns}[c]
		\begin{column}{0.55\textwidth}
			\begin{itemize}
				\item ElGamal is CPA-secure but malleable: same disease, public-key edition.
				\item \textbf{Fujisaki-Okamoto (FO) transform}: derive the randomness from the message; decryption re-encrypts and compares --- ``verify, then decrypt''.
				\item Now deterministic, so it is used on a fresh random $M$ (key encapsulation), not on emails. More in 2.6.
			\end{itemize}
		\end{column}
		\begin{column}{0.45\textwidth}
			\definitionbox{FO Transform (simplified)}{
				Given a CPA-secure PKE $\Pi$ and a hash $H$:\\[0.2em]
				$\texttt{Enc}_{FO}(pk, M)$: $R \coloneq H(M)$; return $C \coloneq \Pi.\texttt{Enc}(pk, M; R)$.\\[0.2em]
				$\texttt{Dec}_{FO}(sk, C)$: $M' \coloneq \Pi.\texttt{Dec}(sk, C)$; return $M'$ if $C = \Pi.\texttt{Enc}(pk, M'; H(M'))$, else \texttt{err}.
			}
		\end{column}
	\end{columns}
\end{frame}
