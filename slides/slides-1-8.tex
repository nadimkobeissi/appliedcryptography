\section{Elliptic Curves: Theory}

\begin{frame}{Elliptic-curve cryptography}
	\begin{itemize}[<+->]
		\item \textbf{Origins (1985):} proposed independently by Neal Koblitz and Victor Miller: run Diffie-Hellman (and, later, signatures) in a new kind of group.
		\item \textbf{The idea:} the group elements are \emph{points on an elliptic curve}, and the hard problem is the \textbf{elliptic curve discrete logarithm problem (ECDLP)}.
		\item \textbf{Why bother?} The same security with far smaller keys and faster operations:
		      \begin{itemize}
			      \item 256-bit ECC key $\approx$ 3,072-bit RSA key $\approx$ 3,072-bit finite-field DH.
			      \item Smaller certificates, signatures and handshakes; cheaper on phones and embedded devices.
			      \item A good resource for estimating key lengths: \url{https://keylength.com}
		      \end{itemize}
		\item \textbf{Adoption timeline:}
		      \begin{itemize}
			      \item 1999--2000: standardized (ANSI X9.62, NIST FIPS 186-2, SECG).
			      \item 2005--2011: OpenSSL, then OpenSSH support.
			      \item 2018: TLS 1.3, with elliptic-curve key exchange as the de-facto default.
		      \end{itemize}
		\item \textbf{Today:} HTTPS, SSH, Signal and WhatsApp, WireGuard, Bitcoin and Ethereum.
	\end{itemize}
\end{frame}

\begin{frame}{ECDH vs. finite field Diffie-Hellman}
	\begin{columns}
		\begin{column}{0.5\textwidth}
			\textbf{Traditional Finite Field DH:}
			\begin{itemize}[<+->]
				\item Works in the multiplicative group $\mathbb{Z}_p^*$
				\item Security based on discrete log in $\mathbb{Z}_p^*$
				\item Requires large primes (2,048+ bits)
				\item Key exchange: $g^{ab} \bmod p$
			\end{itemize}
		\end{column}
		\begin{column}{0.5\textwidth}
			\textbf{Elliptic Curve DH (ECDH):}
			\begin{itemize}[<+->]
				\item Works in the group of points on an elliptic curve
				\item Security based on ECDLP\footnote{Elliptic-curve discrete logarithm problem}
				\item Requires smaller keys (256 bits)
				\item Key exchange: $a \cdot (b \cdot G)$
			\end{itemize}
		\end{column}
	\end{columns}
	\vspace{0.5cm}
	\textbf{ECDH advantages:}
	\begin{itemize}[<+->]
		\item \textbf{Efficiency:} 10--40x faster than finite field DH for equivalent security.
		\item \textbf{Scalability:} Performance gap widens with higher security levels.
		\item \textbf{Ubiquity:} the default key exchange in TLS 1.3, SSH, Signal and WireGuard.
	\end{itemize}
\end{frame}

\begin{frame}{Why finite field DH attacks don't work on ECDH}
	\begin{itemize}[<+->]
		\item \textbf{From 1.7:} baby-step giant-step and Pollard's rho are \emph{generic}: $\mathcal{O}(\sqrt{n})$ in any group of order $n$. What really hurts $\mathbb{Z}_p^*$ is the \emph{extra} attack.
		\item \textbf{Index calculus}, the extra attack on $\mathbb{Z}_p^*$:
		      \begin{itemize}
			      \item Elements of $\mathbb{Z}_p^*$ are integers. Many of them factor into small primes, and that structure can be exploited.
			      \item Subexponential: this is why finite-field DH needs 2,048--3,072-bit primes.
		      \end{itemize}
		\item \textbf{On an elliptic curve, there is nothing to factor:}
		      \begin{itemize}
			      \item Points are not numbers. There is no notion of a ``small prime'' point, so no index calculus.
			      \item For well-chosen curves, only the generic $\mathcal{O}(\sqrt{n})$ attacks are known.
		      \end{itemize}
		\item \textbf{Result:} a 256-bit curve gives $2^{128}$ security, the same as 3,072-bit $\mathbb{Z}_p^*$.
		\item \textbf{Caveat:} ``well-chosen'' is doing real work in that sentence. The Practice section is about what happens otherwise.
	\end{itemize}
\end{frame}

\begin{frame}{What is an elliptic curve?}
	\begin{columns}
		\begin{column}{0.6\textwidth}
			\begin{itemize}
				\item \textbf{Weierstrass form:} In cryptography, elliptic curves typically have equation:
				      $$y^2 = x^3 + ax + b \qquad (\text{with } 4a^3 + 27b^2 \neq 0)$$

				\item \textbf{Example:} $y^2 = x^3 - 4x$, i.e. $a = -4$ and $b = 0$.
				\item \textbf{Why we care:} points can be \emph{added}, and that addition makes the curve a group.\footnote{Historical name: they arose from computing arc lengths of ellipses. They are not ellipses.}
			\end{itemize}

\begin{frame}{Adding two points on an elliptic curve}
	\begin{columns}
		\begin{column}{0.5\textwidth}
			\begin{itemize}
				\item Point addition follows a simple geometric process:
				      \begin{itemize}
					      \item Draw the line through $P$ and $Q$.
					      \item Find the third point where this line meets the curve.
					      \item Reflect it across the $x$-axis: that's $P + Q$.
				      \end{itemize}
				\item \textbf{Why the reflection?} It makes a beautiful rule hold: \emph{any three curve points on one line sum to zero}---the property that will make the curve a group (coming up).
			\end{itemize}
		\end{column}
		\begin{column}{0.5\textwidth}
			\imagewithcaption{elliptic-curve-add.png}{Adding two points on an elliptic curve.\\Source: Serious Cryptography}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Doubling a point on an elliptic curve}
	\begin{columns}
		\begin{column}{0.5\textwidth}
			\begin{itemize}
				\item \textbf{Point doubling:} When $P = Q$, adding $P$ and $Q$ is equivalent to computing $P + P = 2P$.
				\item \textbf{Geometric process:}
				      \begin{itemize}
					      \item Can't draw a line between $P$ and itself.
					      \item Instead, draw the line tangent to the curve at point $P$.
					      \item Find where this tangent line intersects the curve.
					      \item $2P$ is the reflection of this intersection point with respect to the $x$-axis.
				      \end{itemize}
			\end{itemize}
		\end{column}
		\begin{column}{0.5\textwidth}
			\imagewithcaption{elliptic-curve-double.png}{Doubling a point on an elliptic curve.\\Source: Serious Cryptography}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{The same thing, as formulas}{Over $\mathbb{F}_p$ there are no pictures, only arithmetic mod $p$}
	\begin{columns}[c]
		\begin{column}{0.5\textwidth}
			\textbf{Addition} $R = P + Q$, for $x_P \neq x_Q$:
			\begin{align*}
				\lambda & = \frac{y_Q - y_P}{x_Q - x_P} \\
				x_R     & = \lambda^2 - x_P - x_Q      \\
				y_R     & = \lambda (x_P - x_R) - y_P
			\end{align*}
		\end{column}
		\begin{column}{0.5\textwidth}
			\textbf{Doubling} $R = 2P$, for $y_P \neq 0$:
			\begin{align*}
				\lambda & = \frac{3x_P^2 + a}{2y_P}   \\
				x_R     & = \lambda^2 - 2x_P          \\
				y_R     & = \lambda (x_P - x_R) - y_P
			\end{align*}
		\end{column}
	\end{columns}
	\vspace{0.1cm}
	\begin{itemize}
		\item $\lambda$ is the slope of the chord (or tangent); the division is a modular inverse, the expensive step.
		\item Two formulas plus special cases ($P = -Q$, $y_P = 0$, $\mathcal{O}$): remember this when we get to implementation bugs.
		\item \textbf{Notice what is missing:} $b$ appears nowhere. File that away for the invalid-curve attack.
	\end{itemize}
\end{frame}

\begin{frame}{Remember this?}{We need an equivalent for elliptic curves}
	For our cryptographic group $\mathbb{Z}_p^*$, these rules always hold:
	\begin{itemize}[<+->]
		\item \textbf{Closure}: Multiplying any two elements gives another element in the group
		      \begin{itemize}
			      \item In $\mathbb{Z}_5^*$: $2 \times 3 = 1$ (still in the group!)
		      \end{itemize}
		\item \textbf{Identity}: There's a special element (1) that doesn't change others
		      \begin{itemize}
			      \item $1 \times 4 = 4$, $1 \times 2 = 2$, etc.
		      \end{itemize}
		\item \textbf{Inverses}: Every element has a ``partner'' that multiplies to 1
		      \begin{itemize}
			      \item In $\mathbb{Z}_5^*$: $2 \times 3 = 1$, so 2 and 3 are inverses
		      \end{itemize}
		\item \textbf{Associativity}: $(a \times b) \times c = a \times (b \times c)$
	\end{itemize}
	\vspace{0.5cm}
	\textbf{Why care?} These rules guarantee that our cryptographic operations will behave predictably!
\end{frame}

\begin{frame}{Group structure of elliptic curves}
	We need to define all these operations so that our elliptic curves have a group structure, allowing us to use them as a new basis for Diffie-Hellman: doing DH with point addition instead of modular multiplication.
	\begin{itemize}[<+->]
		\item \textbf{Closure property:} If points $P$ and $Q$ belong to a curve, then $P + Q$ also belongs to the curve.
		\item \textbf{Associativity:} $(P + Q) + R = P + (Q + R)$ for any points $P$, $Q$, and $R$.
		\item \textbf{Identity element:} The \emph{point at infinity} $\mathcal{O}$ such that $P + \mathcal{O} = P$ for any $P$.
		      \begin{itemize}
			      \item Where does $\mathcal{O}$ come from? The line through $P$ and $-P$ is vertical: it never meets the curve a third time, so we say it meets it ``at infinity.''
		      \end{itemize}
		\item \textbf{Inverse elements:} Every point $P = (x_P, y_P)$ has an inverse $-P = (x_P, -y_P)$ such that $P + (-P) = \mathcal{O}$.
		\item Great! We have a group structure!
	\end{itemize}
\end{frame}

\begin{frame}{Remember this?}{We need an equivalent for elliptic curves}
	\begin{itemize}
		\item The discrete logarithm problem:
		      \begin{itemize}
			      \item Given a finite cyclic group $G$, a generator $g \in G$, and an element
			            $h \in G$, find the integer $x$ such that $g^{x}=h$
		      \end{itemize}
		\item In more concrete terms:
		      \begin{itemize}
			      \item Let $p$ be a large prime and let $g$ be a generator of the multiplicative
			            group $\mathbb{Z}_{p}^{*}$ (all nonzero integers modulo $p$).
			      \item Given:
			            \begin{itemize}
				            \item $g \in \mathbb{Z}_{p}^{*}$, $h \in \mathbb{Z}_{p}^{*}$

				            \item Find $x \in \{0, 1, \ldots, p-2\}$ such that $g^{x} \equiv h \pmod
					                  {p}$
			            \end{itemize}
			      \item This problem is believed to be computationally hard when $p$ is large
			            and $g$ is a primitive root modulo $p$.
			            \begin{itemize}
				            \item ``Believed to be'' = we don't know of any way to do it that doesn't
				                  take forever, unless we have a strong, stable quantum computer (Shor's
				                  algorithm)
			            \end{itemize}
		      \end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{The elliptic curve discrete logarithm problem (ECDLP)}
	\begin{itemize}[<+->]
		\item \textbf{Remember the original DLP:} Given $g$, $h$, and prime $p$, find $x$ such that $g^x \equiv h \pmod{p}$.
		\item \textbf{ECDLP is the elliptic curve version:}
		      \begin{itemize}
			      \item Given an elliptic curve and a base point $G$ on that curve,
			      \item Given another point $H$ on the same curve,
			      \item Find the integer $k$ such that $k \cdot G = H$.
		      \end{itemize}
		\item \textbf{Why is this hard?}
		      \begin{itemize}
			      \item Easy direction: Given $k$ and $G$, computing $k \cdot G$ is efficient (double-and-add).
			      \item Hard direction: Given $G$ and $H = k \cdot G$, finding $k$ is very difficult.
			      \item No known efficient algorithms (except with quantum computers).
		      \end{itemize}
		\item \textbf{How hard, concretely?} Best known attacks (e.g. Pollard's rho) cost $\approx \sqrt{n}$ steps: $2^{128}$ for a 256-bit curve.
		      \begin{itemize}
			      \item Going forward: a few hundred point operations. Going backward: $2^{128}$.
		      \end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{A toy curve you can compute by hand}
	\textbf{Toy curve:} $E: y^2 = x^3 + 2x + 2$ over $\mathbb{F}_{17}$ has 19 points ($\mathcal{O}$ included). 19 is prime, so every other point is a generator. Take $G = (5, 1)$.
	\begin{columns}[t]
		\begin{column}{0.5\textwidth}
			\begin{itemize}
				\item \textbf{$2G$, by the doubling formula:}
				      \begin{itemize}
					      \item $\lambda = (3 \cdot 5^2 + 2) / (2 \cdot 1) \equiv 9 / 2 \equiv 9 \cdot 9 \equiv 13$
					      \item $x = 13^2 - 10 \equiv 6$, \quad $y = 13(5 - 6) - 1 \equiv 3$
					      \item $2G = (6, 3)$; check: $9 \equiv 230 \pmod{17}$ \mycheckmark
				      \end{itemize}
				\item \textbf{Then} $3G = (10, 6)$, \ldots, $19G = \mathcal{O}$.
				\item \textbf{Hasse's theorem:} $\#E = p + 1 \pm 2\sqrt{p}$. Here $18 \pm 8.2$; indeed 19.
			\end{itemize}
		\end{column}
		\begin{column}{0.5\textwidth}
			\begin{itemize}
				\item \textbf{Toy ECDLP:} given $H = (10, 6)$, find $k$ with $k \cdot G = H$. Try $k = 1, 2, 3$: done. On a real curve: $\approx 2^{256}$ points, $\approx 2^{128}$ steps.
				\item \textbf{Domain parameters} $(p, a, b, G, n, h)$: prime, coefficients, base point, its order, cofactor $h = \#E/n$. Here $(17, 2, 2, (5,1), 19, 1)$.
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Diffie-Hellman key agreement over elliptic curves}
	\begin{itemize}
		\item \textbf{Classical Diffie-Hellman recap:}
		      \begin{itemize}
			      \item Alice picks secret $a$, computes $A = g^a$, sends $A$ to Bob.
			      \item Bob picks secret $b$, computes $B = g^b$, sends $B$ to Alice.
			      \item Both compute the shared secret: $A^b = B^a = g^{ab}$.
		      \end{itemize}
		\item \textbf{Elliptic Curve Diffie-Hellman (ECDH):}
		      \begin{itemize}
			      \item Alice picks secret $a$, computes $A = a \cdot G$, sends $A$ to Bob.
			      \item Bob picks secret $b$, computes $B = b \cdot G$, sends $B$ to Alice.
			      \item Both compute the shared secret: $a \cdot B = b \cdot A = ab \cdot G$.
		      \end{itemize}
		\item \textbf{Key differences:}
		      \begin{itemize}
			      \item Exponentiation $g^x$ $\rightarrow$ Scalar multiplication $x \cdot G$
			      \item Modular arithmetic $\rightarrow$ Elliptic curve point operations
			      \item Generator $g$ $\rightarrow$ Base point $G$
		      \end{itemize}
	\end{itemize}
\end{frame}

\section{Digital Signatures}

\begin{frame}{What are digital signatures?}
	\begin{itemize}
		\item \textbf{Digital signatures:} the public-key analogue of a MAC. The signer holds a \textbf{private key}; \emph{anyone} with the matching \textbf{public key} can verify.
		\item \textbf{What a valid signature tells the verifier:}
		      \begin{itemize}[<+->]
			      \item Authenticity: the message came from the holder of the private key.
			      \item Integrity: the message was not modified.
			      \item Non-repudiation: the signer cannot later deny it. A MAC can't offer this: both parties hold the same key.
		      \end{itemize}
		\item \textbf{Unlike a handwritten signature,} a digital one is bound to the message: you cannot lift it off one document and paste it onto another.
		\item \textbf{Where you meet them:} software updates, TLS certificates, SSH logins, cryptocurrency transactions, email (PGP, S/MIME).
	\end{itemize}
\end{frame}

\begin{frame}{Digital signature API: the three operations}
	\begin{itemize}[<+->]
		\item \textbf{1. Key generation:} \texttt{(private\_key, public\_key) = KeyGen()}
		      \begin{itemize}
			      \item A random secret number, and a public value derived from it. Publish the second, guard the first.
		      \end{itemize}
		\item \textbf{2. Signing:} \texttt{signature = Sign(private\_key, message)}
		      \begin{itemize}
			      \item Only the holder of the private key can produce a valid signature on a message.
		      \end{itemize}
		\item \textbf{3. Verification:} \texttt{is\_valid = Verify(public\_key, message, signature)}
		      \begin{itemize}
			      \item Anyone can run it. Correctness: honestly produced signatures always verify.
		      \end{itemize}
	\end{itemize}
	\vspace{0.2cm}
	\definitionbox{{Security goal: unforgeability (EUF-CMA)}}{An attacker who can request signatures on \emph{any} messages of their choice still cannot produce a valid signature on a \emph{new} message. Every scheme in this lecture aims for this; the attacks in the Practice section are ways implementations fall short of it.}
\end{frame}

\begin{frame}{Digital signatures with elliptic curves}
	\begin{itemize}[<+->]
		\item \textbf{Why elliptic curve signatures?} The same reasons as ECDH: 64-byte signatures and 32-byte keys at 128-bit security, and fast signing and verification.
		\item \textbf{Two main approaches:}
		      \begin{itemize}
			      \item \textbf{ECDSA:} Elliptic Curve Digital Signature Algorithm (1990s). The elliptic-curve version of DSA.
			      \item \textbf{EdDSA:} Edwards-curve Digital Signature Algorithm (2011). A Schnorr-style design; Ed25519 is the instance you will meet.
		      \end{itemize}
		\item \textbf{Real-world adoption:}
		      \begin{itemize}
			      \item ECDSA: Bitcoin, Ethereum, TLS certificates, SSH.
			      \item Ed25519: OpenSSH, Signal Protocol, software signing, many newer systems.
		      \end{itemize}
		\item \textbf{Plan:} ECDSA first (the standard), then EdDSA (the fix), then a comparison.
	\end{itemize}
\end{frame}

\begin{frame}{ECDSA: the established standard}
	\begin{itemize}
		\item \textbf{Elliptic Curve Digital Signature Algorithm (ECDSA):}
		      \begin{itemize}
			      \item Proposed in 1992, standardized by ANSI in 1999 and by NIST in 2000.
			      \item Elliptic curve version of the Digital Signature Algorithm (DSA).
			      \item Widely adopted in blockchain and web security.
		      \end{itemize}
		\item \textbf{Key components:}
		      \begin{itemize}
			      \item Private key: secret number $d$
			      \item Public key: elliptic curve point $P = d \cdot G$
			      \item Base point $G$ (of order $n$) on an agreed elliptic curve
		      \end{itemize}
		\item \textbf{Security foundation:} Based on ECDLP hardness.
		\item \textbf{Signature format:} Two numbers $(r, s)$
		      \begin{itemize}
			      \item For 256-bit curves: 512-bit total signature size.
			      \item Much smaller than equivalent RSA signatures.
		      \end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{ECDSA signature generation}
	\begin{columns}
		\begin{column}{0.6\textwidth}
			\textbf{Input:} Message $M$, private key $d$
			\begin{enumerate}
				\item \textbf{Hash the message:} $h = \func{hash}{M}$
				      \begin{itemize}
					      \item Use SHA-256, SHA-3, or similar
					      \item Interpret the hash as a number $h \in [0, n-1]$
				      \end{itemize}
				\item \textbf{Generate random nonce:} Pick a random $k \in [1, n-1]$
				\item \textbf{Compute signature point:} $k \cdot G = (x, y)$
				\item \textbf{Calculate $r$:} $r = x \bmod n$
				\item \textbf{Calculate $s$:} $s = \frac{h + rd}{k} \bmod n$
				\item \textbf{Output signature:} $(r, s)$ \quad (if $r = 0$ or $s = 0$, start over with a new $k$)
			\end{enumerate}
		\end{column}
		\begin{column}{0.4\textwidth}
			\textbf{Critical requirement:}
			\begin{itemize}
				\item Random $k$ must be:
				      \begin{itemize}
					      \item Cryptographically random
					      \item Different for every signature
					      \item Never reused
				      \end{itemize}
				\item \textbf{Reusing $k$ = private key exposure!}
				\item So is \emph{leaking} $k$: from $(r, s)$ and $k$, anyone computes $d = \frac{sk - h}{r} \bmod n$.
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{ECDSA signature verification}
	\begin{columns}
		\begin{column}{0.58\textwidth}
			\textbf{Input:} Message $M$, signature $(r, s)$, public key $P$
			\begin{enumerate}
				\item \textbf{Range check:} reject unless $r, s \in [1, n-1]$
				\item \textbf{Hash the message:} $h = \func{hash}{M}$
				\item \textbf{Compute modular inverse:} $w = \frac{1}{s} \bmod n$
				\item \textbf{Calculate verification values:} $u = h \cdot w \bmod n$, \quad $v = r \cdot w \bmod n$
				\item \textbf{Compute verification point:} $Q = u \cdot G + v \cdot P$
				\item \textbf{Check signature:} Accept if $Q_x \bmod n = r$
			\end{enumerate}
		\end{column}
		\begin{column}{0.42\textwidth}
			\textbf{Why this works:}
			\begin{itemize}
				\item $Q = u \cdot G + v \cdot d \cdot G = (u + vd) \cdot G$
				\item For an honest signature, $s = \frac{h + rd}{k}$, so $u + vd = w (h + rd) = \frac{h + rd}{s} = k$
				\item So $Q = k \cdot G$, and $Q_x \bmod n = r$.
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{EdDSA: the modern alternative}
	\begin{itemize}
		\item \textbf{Background:} Built on Schnorr signatures (1989).
		      \begin{itemize}
			      \item Schnorr's patent prevented adoption until it expired in 2008.
			      \item Edwards-curve DSA developed by Bernstein et al. (2011).
			      \item (Bitcoin came full circle: it adopted Schnorr signatures with Taproot, 2021.)
		      \end{itemize}
		\item \textbf{Key advantages over ECDSA:}
		      \begin{itemize}
			      \item \textbf{Deterministic:} No random number generation during signing.
			      \item \textbf{Faster:} Both signing and verification.
			      \item \textbf{Simpler:} Cleaner mathematical structure.
			      \item \textbf{Safer:} No per-signature randomness to get wrong (a failure mode we will meet shortly).
		      \end{itemize}
		\item \textbf{Design philosophy:} Avoid the pitfalls that plague ECDSA.
		\item \textbf{Most popular instance:} Ed25519 (based on Curve25519).
	\end{itemize}
\end{frame}

\begin{frame}{EdDSA signature generation}
	\begin{columns}
		\begin{column}{0.7\textwidth}
			\textbf{Input:} message $M$, private key $k$ (32 random bytes); base point $B$ of prime order $\ell$.
			\begin{enumerate}
				\item \textbf{Expand private key:} $a \parallel \mathit{prefix} = \func{hash}{k}$ \quad (SHA-512)
				      \begin{itemize}
					      \item $a$: the signing scalar (first half, ``clamped''); $\mathit{prefix}$: secret randomness (second half)
				      \end{itemize}
				\item \textbf{Compute public key:} $A = a \cdot B$
				\item \textbf{Nonce, deterministically:} $r = \func{hash}{\mathit{prefix} \parallel M} \bmod \ell$
				\item \textbf{Compute signature point:} $R = r \cdot B$
				\item \textbf{Compute signature scalar:} $S = r + \func{hash}{R \parallel A \parallel M} \cdot a \bmod \ell$
				\item \textbf{Output signature:} $(R, S)$, 32 + 32 = 64 bytes
			\end{enumerate}
		\end{column}
		\begin{column}{0.3\textwidth}
			\textbf{Benefits:}
			\begin{itemize}
				\item No randomness needed at signing time
				\item Same message = same signature
				\item A broken RNG cannot leak the key
				\item Faster (no modular inverse)
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{EdDSA signature verification}
	\begin{columns}
		\begin{column}{0.56\textwidth}
			\textbf{Input:} Message $M$, signature $(R, S)$, public key $A$
			\begin{enumerate}
				\item \textbf{Decode} $R$ and $A$ as curve points; check $S < \ell$.
				\item \textbf{Verify equation:} accept if
				      $$S \cdot B = R + \func{hash}{R \parallel A \parallel M} \cdot A$$
			\end{enumerate}
			\textbf{Why this works:}
			\begin{itemize}
				\item From signing: $S = r + \func{hash}{R \parallel A \parallel M} \cdot a$
				\item So: $S \cdot B = r \cdot B + \func{hash}{R \parallel A \parallel M} \cdot a \cdot B$
				\item $= R + \func{hash}{R \parallel A \parallel M} \cdot A$
			\end{itemize}
		\end{column}
		\begin{column}{0.44\textwidth}
			\textbf{vs. ECDSA:} no modular inverse; signatures can be verified in batches.
			\vspace{0.3cm}

			\textbf{Fine print:} RFC 8032 writes the check as $8S \cdot B = 8R + 8\func{hash}{\cdot} \cdot A$ (multiplied by the cofactor 8) and calls the form above ``sufficient, but not required.'' Two equations, one standard. Hold that thought.
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{ECDSA vs. Ed25519: the comparison}
	\begin{columns}[t]
		\begin{column}{0.5\textwidth}
			\textbf{ECDSA}
			\begin{itemize}
				\item[$+$] The standard since 1999: every TLS and X.509 stack; mandatory in Bitcoin and Ethereum
				\item[$-$] Needs a fresh, secret, uniform $k$ per signature: a bad RNG leaks the key
				\item[$-$] Modular inverse and special cases: hard to get right, and constant-time
				\item[$-$] Malleable: $(r, n - s)$ also verifies
			\end{itemize}
		\end{column}
		\begin{column}{0.5\textwidth}
			\textbf{Ed25519}
			\begin{itemize}
				\item[$+$] Deterministic nonce: nothing to get wrong at signing time
				\item[$+$] Complete formulas; fast; fixed 64-byte signatures and 32-byte keys
				\item[$-$] Younger (2011), and RFC 8032 left verifiers choices (coming up)
				\item[$-$] Unusable where secp256k1 is mandated; a fault during deterministic signing leaks the key
			\end{itemize}
		\end{column}
	\end{columns}
	\vspace{0.3cm}
	\textbf{Recommendation:} Use Ed25519 for new projects unless ECDSA is specifically required.
\end{frame}

\section{Elliptic Curves: Practice}

\begin{frame}{Choosing the right elliptic curve}
	\begin{columns}[c]
		\begin{column}{0.5\textwidth}
			\begin{itemize}
				\item \textbf{Not all elliptic curves are created equal:} the curve's structure directly determines how hard its ECDLP is.
				\item \textbf{A poor curve choice} can make the ECDLP dramatically easier than $\sqrt{n}$.
				\item \textbf{In practice} you will use an established curve, but knowing what makes one safe lets you choose sensibly and read a proposal (or a library default) critically.
			\end{itemize}
		\end{column}
		\begin{column}{0.5\textwidth}
			\imagewithcaption[0.75\textwidth]{safecurves.png}{Elliptic curves have many distinct and complex security criteria.\\Source: \url{https://safecurves.cr.yp.to}}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Criteria for safe elliptic curves}
	\begin{itemize}
		\item \textbf{Group order:} the number of points must have a huge prime factor. Pohlig-Hellman (1.7) eats small factors; ideally the order is prime, or has a tiny cofactor.
		\item \textbf{Complete addition formulas:} the Weierstrass formulas have special cases ($P = Q$, $P = -Q$, $\mathcal{O}$) that need separate code paths. That is where bugs and timing leaks hide.
		      \begin{itemize}
			      \item Edwards curves have one formula that works for every pair of points.
		      \end{itemize}
		\item \textbf{Twist security:} a point with an invalid $x$-coordinate lands on the curve's \emph{twist}. If the twist is weak, that is an attack (coming up). X25519 needs no point validation because its twist is strong too.
		\item \textbf{Rigidity:} every constant should have a stated, checkable origin.
		      \begin{itemize}
			      \item ``Nothing up my sleeve'' numbers increase trust. ``Trust us'' constants are how you get the next few slides.
		      \end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{NIST curves: the establishment standard}
	\begin{itemize}
		\item \textbf{Official standardization:} NIST standardized several curves in FIPS 186 (2000).
		      \begin{itemize}
			      \item ``Recommended Elliptic Curves for Federal Government Use''
			      \item Five prime curves working modulo prime numbers.
			      \item Ten binary polynomial curves (rarely used today).
			      \item All five prime curves have a prime number of points (we will see why that is convenient).
		      \end{itemize}
		\item \textbf{Most popular: P-256}
		      \begin{itemize}
			      \item Works modulo $p = 2^{256} - 2^{224} + 2^{192} + 2^{96} - 1$
			      \item Equation: $y^2 = x^3 - 3x + b$ (256-bit $b$ parameter)
			      \item Other sizes: P-192, P-224, P-384, P-521 (yes, 521, not 512!)
		      \end{itemize}
		\item \textbf{Wide adoption:} Used in TLS, government systems, many commercial applications.
	\end{itemize}
\end{frame}

\begin{frame}{The NIST controversy: suspicious constants}
	\begin{itemize}[<+->]
		\item \textbf{The problem:} Only the NSA knows the true origin of the $b$ coefficient in NIST curves.
		\item \textbf{NSA's explanation:} $b$ results from hashing a ``random-looking'' constant with SHA-1.
		      \begin{itemize}
			      \item P-256's $b$ comes from: \texttt{c49d3608 86e70493 6a6678e1 139d26b7 819f7e90}
			      \item But why this particular constant? Nobody knows.
		      \end{itemize}
		\item \textbf{Community response:}
		      \begin{itemize}
			      \item Most experts don't believe the curves hide backdoors.
			      \item But the lack of transparency creates suspicion.
			      \item Led to development of alternative curves with transparent parameters.
		      \end{itemize}
		\item \textbf{Post-Snowden era:} Increased scrutiny of NSA-designed cryptographic standards.
	\end{itemize}
\end{frame}

\begin{frame}{Dual\_EC\_DRBG: the NSA backdoor scandal}
	\begin{itemize}[<+->]
		\item \textbf{What is Dual\_EC\_DRBG?} Dual Elliptic Curve Deterministic Random Bit Generator.
		      \begin{itemize}
			      \item A NIST-standardized random number generator (SP 800-90, 2006).
			      \item Based on elliptic curve operations.
			      \item Suspiciously slow: 1,000x slower than alternatives.
		      \end{itemize}
		\item \textbf{The red flags:}
		      \begin{itemize}
			      \item NSA strongly pushed for its standardization despite poor performance.
			      \item Contains mysterious constants $P$ and $Q$ with no explanation.
			      \item Security researchers warned of potential backdoor (2007).
		      \end{itemize}
		\item \textbf{Snowden revelations (2013):} Reuters subsequently reported that the NSA paid RSA \$10 million to make Dual\_EC\_DRBG the default in its BSafe library.
		\item \textbf{The aftermath:}
		      \begin{itemize}
			      \item NIST withdrew the standard (2014).
			      \item Damaged trust in NSA-influenced standards.
			      \item Increased scrutiny of all elliptic curve parameters.
		      \end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{How the Dual\_EC\_DRBG backdoor works}
	\begin{itemize}[<+->]
		\item \textbf{The algorithm:} two curve points $P$ and $Q$. State $s_i = x(s_{i-1} \cdot P)$; output $r_i = x(s_i \cdot Q)$ minus its top 16 bits.
		\item \textbf{The trapdoor:} whoever knows $d$ with $Q = d \cdot P$ can run it backwards.
		      \begin{itemize}
			      \item From $r_i$, guess the 16 missing bits: $2^{16}$ candidates for the point $s_i \cdot Q$.
			      \item Then $s_i \cdot P = \frac{1}{d}(s_i \cdot Q)$, whose $x$-coordinate is the next state $s_{i+1}$.
			      \item With 32 bytes of output, NSA can predict all future outputs.\footnote{\url{https://appliedcryptography.page/paper/\#dual-ec}}
		      \end{itemize}
		\item \textbf{Plausible deniability:} without $d$ the backdoor is undetectable: recovering $d$ from $(P, Q)$ is an ECDLP.
	\end{itemize}
	\alertbox{{A backdoored generator, not a backdoored curve}}{Dual\_EC runs on the ordinary P-256 curve; the trapdoor is the secret relation between $P$ and $Q$. The NIST curves have no known weakness. What died was the presumption of good faith behind unexplained constants.}
\end{frame}

\begin{frame}{Curve25519: fast, and nothing up its sleeve}
	\begin{itemize}
		\item \textbf{Created by Daniel J. Bernstein (2006):} Motivated by performance and security concerns.
		\item \textbf{Performance advantages:}
		      \begin{itemize}
			      \item Faster than the NIST curves at the same security level.
			      \item Designed for fast, simple software implementation.
		      \end{itemize}
		\item \textbf{Security improvements:}
		      \begin{itemize}
			      \item No suspicious constants---all parameters have clear origins.
			      \item Edwards form (Ed25519) has complete addition formulas.
			      \item More resistant to timing attacks.
		      \end{itemize}
		\item \textbf{Mathematical form:} $y^2 = x^3 + 486662x^2 + x$
		      \begin{itemize}
			      \item Works modulo $2^{255} - 19$ (largest prime below $2^{255}$, enabling fast reduction).
			      \item Coefficient 486662 is the smallest integer satisfying security criteria.
		      \end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{Curve25519: from rebel to standard}
	\begin{itemize}
		\item \textbf{Widespread adoption, long before any blessing:}
		      \begin{itemize}
			      \item Signal Protocol, and therefore WhatsApp's end-to-end encryption
			      \item TLS 1.3 key exchange (X25519 is the most common choice)
			      \item OpenSSH, Tor, WireGuard
			      \item Several cryptocurrencies (Monero, Solana, Cardano)
		      \end{itemize}
		\item \textbf{Official recognition:} Added to NIST-approved curves in February 2023.
		      \begin{itemize}
			      \item SP 800-186: ``Recommendations for Discrete Logarithm-based Cryptography''
			      \item Took 17 years for official government approval!
		      \end{itemize}
		\item \textbf{Trust through transparency:} Clear parameter origins make Curve25519 more trustworthy than NIST curves.
		\item \textbf{Related: Ed25519} for digital signatures using the same curve. Let's untangle the names.
	\end{itemize}
\end{frame}

\begin{frame}{Curve25519, X25519, Ed25519: what's in a name?}
	\begin{center}
		\begin{tikzpicture}[every node/.style={align=center}]

			\node[draw, thick, cipherprimary, rounded corners, fill=cipherprimary!8, text width=4.9cm] (ed) at (8.6,1.9) {\textbf{edwards25519}\\ \small the same curve,\\ \small in Edwards form\\ \footnotesize $-x^2 + y^2 = 1 + dx^2y^2$};

			\node[draw, thick, cipheraccent, rounded corners, fill=cipheraccent!15, text width=4.9cm] (x) at (0,0) {\textbf{X25519}\\ \small key exchange (ECDH)};
			\node[draw, thick, cipheraccent, rounded corners, fill=cipheraccent!15, text width=4.9cm] (edd) at (8.6,0) {\textbf{Ed25519}\\ \small signatures (EdDSA)};

			\draw[->, thick, cipherdark] (ed) -- (edd);
		\end{tikzpicture}
	\end{center}
	\begin{itemize}
		\item \textbf{Same curve, two forms. Each form is ideal for a different job:}
		      \begin{itemize}
			      \item Montgomery form: $x$-coordinate-only ``ladder''---simple, fast, constant-time ECDH.
			      \item Edwards form: complete addition formulas, no special cases---safe signatures.
		      \end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{Other curves in the ecosystem}
	\begin{itemize}
		\item \textbf{secp256k1: the blockchain workhorse.}
		      \begin{itemize}
			      \item $y^2 = x^3 + 7$: standardized by SECG (2000), chosen for Bitcoin by Satoshi Nakamoto, inherited by Ethereum.
			      \item Almost no arbitrary constants---``rigid'' by construction, unlike the NIST curves.
			      \item (This is the curve behind the \texttt{k256} library naming.)
		      \end{itemize}
		\item \textbf{National standards:}
		      \begin{itemize}
			      \item \textbf{ANSSI curves (France):} Constants of unexplained origin, no unified addition law.
			      \item \textbf{Brainpool curves (Germany):} Transparent seeds (digits of $e$ and $\pi$), but slow arithmetic and no unified addition law.
		      \end{itemize}
		\item \textbf{Higher-security alternatives:}
		      \begin{itemize}
			      \item \textbf{Ed448-Goldilocks:} 448-bit curve (RFC 8032, 2017), $\approx$224-bit security margin.
			      \item \textbf{Curve41417:} 414-bit curve by Bernstein et al. ($\approx$200-bit security).
		      \end{itemize}
		\item \textbf{Ristretto initiative:} Technique for safe point representation.
		      \begin{itemize}
			      \item Constructs prime-order groups from non-prime-order curves (more on this soon).
		      \end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{Practical curve selection guidance}
	\begin{itemize}
		\item \textbf{For new projects:} Use Curve25519 (X25519 for key exchange, Ed25519 for signatures).
		      \begin{itemize}
			      \item Excellent performance and security; transparent parameter generation.
			      \item Wide library support; NIST-approved since 2023.
		      \end{itemize}
		\item \textbf{For government or compliance work:} NIST P-256 is still widely accepted.
		      \begin{itemize}
			      \item Required by some standards and regulations, and the only curve in most certificate ecosystems.
			      \item Well-audited implementations exist. Despite the parameter concerns, no known weakness.
		      \end{itemize}
		\item \textbf{Avoid:}
		      \begin{itemize}
			      \item ANSSI curves: unexplained constants.
			      \item Brainpool curves: transparent, but slow and without complete formulas.
			      \item Binary curves, and anything below 256 bits.
		      \end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{How things can go wrong}
	\begin{itemize}
		\item \textbf{The math is fine.} Every incident in the rest of this lecture is an implementation or protocol mistake wrapped around an unbroken curve.
		\item \textbf{Three failure modes, three case studies:}
		      \begin{enumerate}
			      \item \textbf{Bad randomness} in ECDSA signing: the PlayStation 3.
			      \item \textbf{Unvalidated points} in key exchange: invalid-curve and small-subgroup attacks on TLS and Monero.
			      \item \textbf{Ambiguous standards:} Ed25519 verifiers that disagree with each other.
		      \end{enumerate}
		\item \textbf{Plus the ever-present one:} side channels. Any operation whose timing depends on a secret is a leak. We return to it at the end.
	\end{itemize}
\end{frame}

\begin{frame}{ECDSA with bad randomness}
	\begin{itemize}
		\item \textbf{Recall:} each ECDSA signature uses a secret random $k$: $r = (k \cdot G)_x \bmod n$, \quad $s = \frac{h + rd}{k} \bmod n$.
		\item \textbf{Quick exercise:} you see two signatures $(r, s_1)$, $(r, s_2)$ on messages with hashes $h_1 \neq h_2$, with the \emph{same} $r$. What happened, and what can you compute?
	\end{itemize}
	\begin{itemize}[<+->]
		\item \textbf{Same $r$ means same $k$.} Then:
		      \begin{itemize}
			      \item $s_1 - s_2 = \frac{(h_1 + rd) - (h_2 + rd)}{k} = \frac{h_1 - h_2}{k}$, so $k = \frac{h_1 - h_2}{s_1 - s_2}$
			      \item and from either signature, $d = \frac{s_1 k - h_1}{r}$. Two signatures, one private key.
		      \end{itemize}
		\item \textbf{Even \emph{bias} is fatal:} a few leaky bits of $k$ across many signatures let lattice attacks recover $d$ (Minerva, 2019; LadderLeak, 2020).
		\item \textbf{In the wild:} Bos et al. (2014) found Bitcoin transactions signed with repeated nonces; those private keys were recoverable by anyone.\footnote{\url{https://appliedcryptography.page/paper/\#ecc-practice}}
		\item \textbf{Prevention:} a strong RNG---or no RNG at all: derive $k$ deterministically from the private key and message (RFC 6979, now in FIPS 186-5).
	\end{itemize}
\end{frame}

\begin{frame}{Case study: PlayStation 3 hack (2010)}
	\begin{itemize}
		\item \textbf{The vulnerability:} Sony's PS3 code signing used the \emph{same} $k$ for every ECDSA signature.
		\item \textbf{Discovery:} the fail0verflow team at the 27th Chaos Communication Congress (December 2010).
		      \begin{itemize}
			      \item Signatures on different files all shared the same $r$: the tell-tale sign from the previous slide.
			      \item Their slide on Sony's RNG: xkcd's \texttt{return 4; // chosen by fair dice roll}.
		      \end{itemize}
		\item \textbf{Consequences:}
		      \begin{itemize}
			      \item Anyone could sign code the console would run: homebrew, Linux, and piracy.
			      \item Keys burned into the boot chain of consoles already sold cannot simply be rotated.
			      \item Sony's response was legal: it sued George Hotz and fail0verflow members in January 2011.
		      \end{itemize}
		\item \textbf{Lesson:} the previous slide is not theoretical. One reused nonce cost Sony the root of trust of a console generation.
	\end{itemize}
\end{frame}

\begin{frame}{Invalid curve attacks}
	\begin{itemize}
		\item \textbf{The vulnerability:} an ECDH implementation that never checks whether the point it received is actually on the curve.
		\item \textbf{The insight (from our formulas slide):} the addition and doubling formulas use $a$ but never $b$.
		      \begin{itemize}
			      \item So the same code computes ``correctly'' on \emph{every} curve $y^2 = x^3 + ax + b'$, whatever $b'$ is.
			      \item Among those curves are some with terrible group orders.
		      \end{itemize}
		\item \textbf{Attack scenario:}
		      \begin{itemize}
			      \item Bob has a long-term ECDH key $b$ and computes $b \cdot P$ for whatever $P$ he is sent (as TLS-ECDH servers did).
			      \item The attacker picks a curve $E'$ with the same $a$, and a point $P \in E'$ of \emph{small} order $m$, and sends $P$.
			      \item Bob computes $b \cdot P$ on $E'$ and derives a key from it. But $b \cdot P$ is one of only $m$ points.
		      \end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{Invalid curve attacks: the math and the fix}
	\begin{itemize}
		\item \textbf{Recovering $b$:} try all $m$ candidates for $b \cdot P$ against Bob's response: that reveals $b \bmod m$.
		      \begin{itemize}
			      \item Repeat with other small orders $m_1, m_2, \ldots$ on other curves; combine with the CRT.
			      \item This is Pohlig-Hellman (1.7) with the attacker \emph{choosing} the smooth group. Enough handshakes recover all of $b$.
		      \end{itemize}
		\item \textbf{In the wild:} Jager, Schwenk and Somorovsky (2015) broke TLS-ECDH servers running Bouncy Castle and Oracle's Java.\footnote{\url{https://appliedcryptography.page/paper/\#invalid-curve}}
		\item \textbf{The fix} is cheap. Before using a received point $P = (x, y)$:
		      \begin{itemize}
			      \item check $x, y \in [0, p-1]$ and $y^2 \equiv x^3 + ax + b \pmod{p}$ with the \emph{correct} $b$;
			      \item check $P \neq \mathcal{O}$ and, if the curve has a cofactor, $n \cdot P = \mathcal{O}$.
		      \end{itemize}
		\item \textbf{Or design it away:} X25519 uses only $x$-coordinates: every $x$ is on the curve or its (strong) twist.
	\end{itemize}
\end{frame}

\begin{frame}{Cofactors: why Curve25519 doesn't have prime order}
	\definitionbox{Order and cofactor}{The \textbf{cofactor} of a curve with $\#E$ points and a base point of prime order $\ell$ is $h = \#E / \ell$. Cryptography lives in the order-$\ell$ subgroup; trouble lives in the rest.}
	\begin{itemize}
		\item \textbf{Prime order is available:} $\#E \approx p$ is prime about as often as a random number that size, so one can search for it. NIST P-256 and secp256k1 have $h = 1$.
		\item \textbf{But the fast shapes can't have it:} a Montgomery curve always has the point $(0, 0)$ of order 2; a complete Edwards addition law needs points of order 4. So $4 \mid h$.
		\item \textbf{Bernstein's trade:} the fast, complete formulas with the smallest cofactor they allow: Curve25519 has $\#E = 8\ell$, Ed448 has $h = 4$.
		\item \textbf{The price:} 7 points besides $\mathcal{O}$ of order dividing 8. Every protocol must decide what to do when one shows up.
	\end{itemize}
\end{frame}

\begin{frame}{Small subgroup attacks}
	\begin{columns}[c]
		\begin{column}{0.58\textwidth}
			\begin{itemize}
				\item \textbf{The attack:} Send a point $P$ of small order (say, 8) instead of an honest public key.
				      \begin{itemize}
					      \item Victim computes the ``shared secret'' $s \cdot P$.
					      \item Whatever the 256-bit secret $s$ is, $s \cdot P$ is one of only 8 points.
					      \item Attacker brute-forces all 8 candidates offline: every derived key falls.
					      \item Bonus: \emph{which} of the 8 it was leaks $s \bmod 8$.
				      \end{itemize}
				\item \textbf{Defense:} Check the order of received points, or use designs that make the issue vanish.
				      \begin{itemize}
					      \item X25519 ``clamps'' secrets to multiples of 8, so $s \cdot P = \mathcal{O}$ for small-order $P$; RFC 7748 then rejects the all-zero output.
				      \end{itemize}
			\end{itemize}
		\end{column}
		\begin{column}{0.4\textwidth}
			\begin{center}
				\begin{tikzpicture}
					\foreach \i in {0,...,7}{

\end{frame}

\begin{frame}{Invalid curve attacks vs. small subgroup attacks}
	\begin{itemize}
		\item \textbf{Both exploit missing point validation;} the defenses differ.
		\item \textbf{Invalid curve attacks:}
		      \begin{itemize}
			      \item Attacker sends a point from a \emph{different curve} $y^2 = x^3 + ax + b'$, exploiting that the formulas never use $b$.
			      \item Defense: check $y^2 \stackrel{?}{=} x^3 + ax + b$ with the correct $b$ (or use $x$-only arithmetic on a twist-secure curve).
		      \end{itemize}
		\item \textbf{Small subgroup attacks:}
		      \begin{itemize}
			      \item Attacker sends a point from the \emph{correct curve}, but of small order: $8P = \mathcal{O}$ instead of the expected order $\ell$.
			      \item Defense: check the order ($\ell \cdot P \stackrel{?}{=} \mathcal{O}$), clear the cofactor, or use a prime-order group.
		      \end{itemize}
		\item \textbf{One line:} invalid curve = wrong equation; small subgroup = wrong order.
		\item Both keep resurfacing in deployed protocols: see Cremers and Jackson (2019) on which protocols check what.\footnote{\url{https://appliedcryptography.page/paper/\#prime-order}}
	\end{itemize}
\end{frame}

\begin{frame}{Ed25519 validation inconsistencies}
	\begin{itemize}
		\item \textbf{Expectation:} one standard (RFC 8032, 2017), one answer to ``is this signature valid?''
		\item \textbf{Reality:} the RFC leaves the verifier choices, and libraries chose differently:
		      \begin{itemize}
			      \item \textbf{Which equation:} $S \cdot B = R + \func{hash}{\cdot} \cdot A$, or the cofactored $8S \cdot B = 8R + 8\func{hash}{\cdot} \cdot A$? They disagree exactly when $R$ or $A$ has a small-order component.
			      \item \textbf{Which encodings:} accept a $y$-coordinate $\geq p$, a non-canonical encoding of $R$ or $A$? The RFC's test vectors never ask.
			      \item \textbf{Which keys:} reject public keys of small order? Otherwise an attacker-chosen $A$ yields signatures valid for every message under the cofactored check.
		      \end{itemize}
		\item \textbf{Findings:} Henry de Valence's analysis,\footnote{\url{https://hdevalence.ca/blog/2020-10-04-its-25519am/}} and Chalkias, Garillot and Nikolaenko's ``Taming the many EdDSAs'' (2020), which fed edge cases to over a dozen libraries: the same signature is valid in one, invalid in the next.
	\end{itemize}
\end{frame}

\begin{frame}{Why verifier disagreements matter}
	\begin{itemize}
		\item \textbf{Consensus systems:} every blockchain node must reach the \emph{same} verdict on every transaction. If node A accepts a signature that node B rejects, the chain forks; a library upgrade becomes a consensus change.
		\item \textbf{Protocol assumptions:} anything that treats a signature or key as unique breaks when $(R, S)$ and $(R + T, S)$ both verify, for a small-order $T$.
		\item \textbf{The fix, Zcash-style (ZIP-215, 2020):} one explicit rule set (cofactored equation, $S < \ell$, explicit handling of non-canonical and small-order points), so that batch and single verification agree and every implementation gives the same answer.
		\item \textbf{Lesson:} ``sufficient, but not required'' is not a sentence that belongs in a signature verification spec.
	\end{itemize}
\end{frame}

\begin{frame}{Ristretto: a prime-order group on top of Curve25519}
	\begin{itemize}
		\item \textbf{The dilemma:} fast, complete curves (Curve25519) have cofactor 8, but most protocols and their security proofs assume a \emph{prime-order} group. Patching each protocol by hand (``multiply by 8 here'') is error-prone.
		\item \textbf{Real bugs:} Monero (2017): a missing subgroup check let one output be spent up to 8 times. Tor v3 onion services: 8 addresses for one service.
		\item \textbf{Ristretto's answer}\footnote{\url{https://ristretto.group/why_ristretto.html}} (after Hamburg's Decaf, 2015):\footnote{\url{https://appliedcryptography.page/paper/\#ecc-decaf}} a new \emph{encoding} in which points that differ by a small-order point encode to the same bytes.
		      \begin{itemize}
			      \item Protocols see a group of prime order $\ell$; underneath runs plain Curve25519 arithmetic at full speed.
			      \item Used by \texttt{curve25519-dalek}, Signal's private groups, and most new Curve25519 designs.
		      \end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{Constant time: why the textbook algorithm leaks}
	\begin{columns}[t]
		\begin{column}{0.5\textwidth}
			\textbf{The leak:} textbook double-and-add
			\begin{itemize}
				\item \texttt{R = 2R; if bit == 1: R = R + G}
				\item Adds happen only on 1-bits: timing, power or cache footprint spell out $k$.
				\item Minerva (2019) recovered ECDSA keys this way from smartcards.
			\end{itemize}
		\end{column}
		\begin{column}{0.5\textwidth}
			\textbf{The fix:} the Montgomery ladder
			\begin{itemize}
				\item Keep $(R_0, R_1)$ with $R_1 = R_0 + G$ always.
				\item Every bit does one add \emph{and} one double: $(2R_0, R_0 + R_1)$ for a 0, $(R_0 + R_1, 2R_1)$ for a 1.
				\item Select by arithmetic (a constant-time swap), never by a branch.
			\end{itemize}
		\end{column}
	\end{columns}
	\vspace{-0.15cm}
	\begin{itemize}
		\item \textbf{Same rule wherever secrets touch:} no branches, lookups, early exits or variable-time division on secrets.
		\item \textbf{Hard even for experts:} LadderLeak (2020) broke OpenSSL ECDSA through a leaky ladder swap; TPM-Fail (2019) extracted ECDSA keys from Intel and STMicro TPM chips via timing.
	\end{itemize}
\end{frame}

\begin{frame}{Library selection: don't implement ECC from scratch}
	\begin{itemize}
		\item \textbf{Everything in this section was an implementation bug} around unbroken math. Hardened libraries have already made, and fixed, these mistakes.
		\item \textbf{Rust:} \texttt{curve25519-dalek}, \texttt{ed25519-dalek} (X25519, Ed25519, Ristretto); \texttt{p256}, \texttt{k256} (RustCrypto); \texttt{aws-lc-rs}, \texttt{rustls}'s default provider (replacing \texttt{ring}).
		\item \textbf{Go:} \texttt{crypto/ecdh} and \texttt{crypto/ed25519} in the standard library, with misuse-resistant APIs; \texttt{filippo.io/edwards25519} for lower-level Edwards arithmetic. (\texttt{crypto/elliptic} point arithmetic is deprecated: use \texttt{crypto/ecdh}.)
		\item \textbf{C:} libsodium (the NaCl lineage: X25519, Ed25519, Ristretto, and almost no knobs to misconfigure); BoringSSL or OpenSSL when you need P-256 and certificates.
		\item \textbf{How to tell a good one:} independent audits, constant-time by contract, and it passes Project Wycheproof's edge-case test vectors: invalid points, non-canonical encodings, degenerate signatures.
	\end{itemize}
\end{frame}

\begin{frame}{Where this is heading: post-quantum hybrids}
	\begin{itemize}
		\item \textbf{The threat:} a large quantum computer running Shor's algorithm solves the ECDLP outright. Every curve in this lecture falls, and the choice of curve makes no difference.
		\item \textbf{The standards:} NIST finalized its first post-quantum standards in August 2024: ML-KEM (key exchange), ML-DSA and SLH-DSA (signatures), FIPS 203--205.
		\item \textbf{Already deployed:} the major browsers run \emph{hybrid} X25519 + ML-KEM key exchange in TLS by default. The session key is secure if \emph{either} half holds.
		\item \textbf{Why key exchange first:} ``harvest now, decrypt later.'' Traffic recorded today can be decrypted by a future quantum computer; a signature only needs to be unforgeable \emph{now}.
		\item \textbf{So elliptic curves stay:} as the classical half of every hybrid, and in the signatures not yet migrated. Post-quantum cryptography gets its own lecture (2.6); verified implementations get theirs (2.5).
	\end{itemize}
\end{frame}

\begin{frame}{What to remember from this lecture}
	\begin{itemize}
		\item \textbf{Elliptic curves are Diffie-Hellman in a stranger group:} $k \cdot G$ instead of $g^k$. Only the generic $\sqrt{n}$ attacks apply, so 256 bits suffice.
		\item \textbf{One-way by construction:} $k \cdot G$ costs a few hundred point operations; recovering $k$ costs $2^{128}$.
		\item \textbf{ECDSA lives or dies by its nonce.} Reused: two signatures leak the key (PS3). Biased: lattices leak the key. Use RFC 6979, or use Ed25519.
		\item \textbf{Validate what you receive, or use a design with nothing to validate.} Invalid-curve and small-subgroup attacks are skipped checks; X25519 and Ristretto leave no check to skip.
		\item \textbf{Ambiguity in a standard becomes disagreement in the field:} fifteen years on, Ed25519 verifiers still disagree at the edges.
		\item \textbf{Where things stand:} X25519 in TLS 1.3, SSH and Signal; Ed25519 is OpenSSH's default key type since 2023; Bitcoin, Ethereum and the web PKI stay on ECDSA.
	\end{itemize}
\end{frame}

\begin{frame}{Discussion time}
	\begin{enumerate}
		\item \textbf{P-256 or X25519?} The NIST curves have unexplained seeds and no known weakness; Curve25519 has explained constants and a cofactor. Which would you ship, and what evidence would change your mind?
		\item \textbf{Who owns a validation bug?} RFC 8032 said ``sufficient, but not required.'' The libraries picked differently. Zcash had to write its own rules. Where should the responsibility have sat?
		\item \textbf{Determinism cuts both ways.} Deterministic nonces ended the PS3 failure mode but opened the door to fault attacks. Would you rather trust the RNG or the hardware?
		\item \textbf{Bitcoin took twelve years to adopt Schnorr signatures} that had been patent-free since 2008. What does that tell you about upgrading cryptography in a system nobody controls?
	\end{enumerate}
\end{frame}
