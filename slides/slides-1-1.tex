\begin{frame}{Cryptographic building blocks}
	\begin{columns}[c]
		\begin{column}{0.5\textwidth}
			\textbf{Components}
			\begin{itemize}[<+->]
				\item Cryptography manifests as a set of primitives, from which we
				      build protocols intended to accomplish well-defined security goals.
				\item \textbf{Primitives}: AES, RSA, SHA-2, DH\ldots
				\item \textbf{Protocols and systems}: TLS, Signal, SSH, FileVault 2, BitLocker\ldots
			\end{itemize}

		\begin{column}{0.5\textwidth}
			\textbf{Examples}
			\begin{itemize}[<+->]
				\item \textbf{AES}: Symmetric encryption
				      \begin{itemize}
					      \item $\mathsf{Enc}(k, m) = c$, $\mathsf{Dec}(k, c) = m$.
				      \end{itemize}
				\item \textbf{SHA-2}: Hash function
				      \begin{itemize}
					      \item $\mathsf{H}(m) = h$.
				      \end{itemize}
				\item \textbf{Diffie-Hellman}: Public key agreement
				      \begin{itemize}
					      \item Allows two parties to agree on a secret key $k$.
				      \end{itemize}
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Security goals}
	\begin{columns}[c]
		\begin{column}{0.5\textwidth}
			\textbf{Definitions}
			\begin{itemize}[<+->]
				\item \textbf{Confidentiality}: Data exchanged between Client and Server
				      is only known to those parties.
				\item \textbf{Authentication}: If Server receives data from Client,
				      then Client sent it to Server.
				\item \textbf{Integrity}: If Server modifies data owned by Client,
				      Client can find out.
			\end{itemize}
		\end{column}

		\begin{column}{0.5\textwidth}
			\textbf{Examples}
			\begin{itemize}[<+->]
				\item \textbf{Confidentiality}: When you send a private message on Signal,
				      only you and the recipient can read the content.
				\item \textbf{Authentication}: When you connect to gmail.com, your browser
				      checks that it is really talking to Google, not to someone in the middle.
				\item \textbf{Integrity}: Your computer can verify that software update
				      downloads haven't been tampered with during transmission.
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Security goals: more examples}
	\begin{itemize}[<+->]
		\item \textbf{TLS (HTTPS)} ensures that data exchanged between the client

\begin{frame}{Why bother?}
	\begin{itemize}[<+->]
		\item Can't we just use access control?
		\item Strictly speaking, usernames and passwords can be implemented
		      without cryptography\ldots
		\item Server checks if the password matches, or if the IP address matches,
		      etc., before granting access.
		\item What's so bad about that?
	\end{itemize}
	\definitionbox{The problem with traditional access control}{
		\begin{itemize}[<+->]
			\item You must trust the server completely (it sees your password).
			\item Nothing protects the data in transit: your ISP sees it all.
			\item No way to detect tampering.
			\item No way for two strangers to establish trust.
		\end{itemize}
	}
\end{frame}

\section{Examples of the Magic in Cryptography}

\begin{frame}{Hard problems}
	\begin{itemize}[<+->]
		\item Cryptography is largely about equating the security of a system to the
		      difficulty of a math problem that is thought to be computationally
		      very expensive to solve.
		\item With cryptography, we get security systems that we can
		      mathematically prove secure (under assumptions).
		\item Also, this allows for actual magic.
		      \begin{itemize}[<+->]
			      \item Alice and Bob meet for the first time in the same room as you.
			      \item You are listening to everything they are saying.
			      \item Can they exchange a secret without you learning it?
		      \end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{Your turn: be the eavesdropper}
	\begin{columns}[c]
		\begin{column}{0.5\textwidth}
			\textbf{You saw everything on the wire:} $p = 13$, $g = 2$, $A = 6$, $B = 11$.
			\vspace{0.3cm}
			\begin{itemize}
				\item \textbf{Find Alice's secret $a$}: solve $2^{a} \bmod 13 = 6$.
				\item Hint: there are only twelve candidates.
			\end{itemize}
		\end{column}
		\begin{column}{0.5\textwidth}
			\pause
			\begin{itemize}
				\item Brute force: $2^1 = 2$, $2^2 = 4$, $2^3 = 8$, $2^4 \equiv 3$, $2^5 \equiv 6 \pmod{13}$. So $a = 5$, and $s = 11^{5} \bmod 13 = 7$.
			\end{itemize}
			\pause
			\alertbox{Why did this work?}{Because $p = 13$. With a 2048-bit $p$, no known algorithm, brute force or clever, finishes in any reasonable time: the \textbf{discrete logarithm problem}.}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{No known feasible computation}
	\begin{itemize}[<+->]
		\item The discrete logarithm problem:
		      \begin{itemize}
			      \item Given a finite cyclic group $G$, a generator $g \in G$, and an element
			            $h \in G$, find the integer $x$ such that $g^{x}=h$
		      \end{itemize}
		\item In more concrete terms:
		      \begin{itemize}
			      \item Let $p$ be a large prime and let $g$ be a generator of the multiplicative
			            group $\mathbb{Z}_{p}^{*}$ (all nonzero integers modulo $p$).

			      \item Given:
			            \begin{itemize}
				            \item $g \in \mathbb{Z}_{p}^{*}$, $h \in \mathbb{Z}_{p}^{*}$

				            \item Find $x \in \{0, 1, \ldots, p-2\}$ such that $g^{x} \equiv h \pmod
					                  {p}$
			            \end{itemize}

			      \item This problem is believed to be computationally hard when $p$ is large
			            and $g$ is a primitive root modulo $p$.
			            \begin{itemize}
				            \item ``Believed to be'' = we don't know of any way to do it that doesn't
				                  take forever, unless we have a large-scale, fault-tolerant quantum computer (Shor's
				                  algorithm)
			            \end{itemize}
		      \end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{Signal's double ratchet: DH everywhere}
	\begin{columns}[c]
		\begin{column}{0.7\textwidth}
			\begin{itemize}
				\item \textbf{Initial key exchange}: X3DH, \textbf{three} DH exchanges combined; works even when the recipient is offline (\textit{``asynchronous''}).\footnote{Everything on this slide will be covered in much more detail later in the course.}
				      \begin{itemize}
					      \item Since 2023: PQXDH, which mixes in a post-quantum KEM.
				      \end{itemize}
				\item \textbf{Ongoing communication}: the Double Ratchet.
				      \begin{itemize}
					      \item Fresh DH key exchange at every change of direction; a symmetric ``ratchet'' covers the messages in between.
					      \item Forward secrecy: phone compromised today? Yesterday's messages stay secure.
					      \item Post-compromise security: compromise over? Tomorrow's messages are secure again.
				      \end{itemize}
			\end{itemize}
		\end{column}
		\begin{column}{0.3\textwidth}
			\imagewithcaption{signal.jpg}{Signal uses DH key exchange dozens, even hundreds, of times per conversation.}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Two families of primitives}
	\begin{columns}[c]
		\begin{column}{0.5\textwidth}
			\textbf{Asymmetric primitives}
			\begin{itemize}[<+->]
				\item Diffie-Hellman, RSA, ML-KEM, etc.
				\item ``Asymmetric'' because there is a ``public key'' and a ``private
				      key'' for each party.
				\item Algebraic, assume the hardness of mathematical problems (as seen
				      just now).
			\end{itemize}
		\end{column}

		\begin{column}{0.5\textwidth}
			\textbf{Symmetric primitives}
			\begin{itemize}[<+->]
				\item AES, SHA-2, ChaCha20, HMAC\ldots
				\item ``Symmetric'' because there is a single shared secret key (or none at all: hash functions have no key).
				\item Not algebraic but unstructured, relying on their understood
				      resistance to $n$ years of cryptanalysis.
				\item Can act as substitutes for assumptions in security proofs!
				      \begin{itemize}
					      \item Example: hash function assumed to be a ``random oracle''
				      \end{itemize}
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Kerckhoffs' principle}
	\begin{itemize}[<+->]
		\item \textit{``A cryptosystem should be secure even if everything about
			      the system, except the key, is public knowledge.''} --- Auguste Kerckhoffs,
		      1883
		\item \textbf{Why it matters}:
		      \begin{itemize}[<+->]
			      \item No ``security through obscurity''
			      \item The key is the only secret: the rest can be audited, tested,
			            trusted
			      \item Encourages open standards and peer review
			      \item If your system's security depends on nobody knowing how it works,
			            it's not secure.
		      \end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{Symmetric primitive example: hash functions}
	\begin{columns}[c]
		\begin{column}{0.55\textwidth}
			\definitionbox{Hash Function Properties}{
				\begin{itemize}\item Takes input of \textbf{any size}
					\item Produces output of \textbf{fixed size}
					\item Is \textbf{deterministic} (same input $\rightarrow$ same output)
					\item Even a \textbf{tiny change} in input creates completely different output
					\item Is \textbf{efficient} to compute\end{itemize}
			}
		\end{column}

\begin{frame}{Expected properties of a hash function}
	\begin{columns}[c]
		\begin{column}{0.6\textwidth}
			\begin{itemize}[<+->]
				\item \textbf{Collision resistance}: computationally infeasible to find
				      two different inputs producing the same hash.
				\item \textbf{Preimage resistance}: given an output of a hash function,
				      it is computationally infeasible to find \emph{any} input that hashes to it.
				\item \textbf{Second preimage resistance}: given an input and an output,
				      it's computationally infeasible to find a different input
				      producing the same output.
			\end{itemize}
		\end{column}
		\begin{column}{0.4\textwidth}
			\imagewithcaption{sha2.png}{SHA-2 compression function. Source: Wikipedia}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Hash functions: what are they good for?}
	\begin{itemize}[<+->]
		\item \textbf{Data integrity verification}: Hash a file. Later hash it
		      again and compare hashes to check if the file has changed, suffered storage
		      degradation, etc.
		\item \textbf{Proof of work}: Server asks client to hash something a lot of
		      times before they can access some resource. Useful for anti-spam, Bitcoin
		      mining, etc.
		\item \textbf{Zero-knowledge proofs}: Time for more actual magic.
	\end{itemize}
\end{frame}

\begin{frame}{Data integrity: verifying a Linux ISO download}
	\begin{columns}[c]
		\begin{column}{0.60\textwidth}
			\begin{itemize}
				\item Linux distributions publish a hash (usually SHA-256) for each ISO image they release.
				\item After downloading, you hash your own copy and compare it to the published value:
				      \begin{center}
					      {\small $\mathsf{SHA256}(\texttt{your.iso}) \stackrel{?}{=} \texttt{1620295f\ldots}$}
				      \end{center}
				\item A match means the file arrived \textbf{intact}: no corruption from a flaky network or a failing disk.
				\item This only catches \emph{accidental} corruption: a malicious mirror could swap both the ISO \emph{and} the hash, which is why real verification also uses a signature (OpenPGP).
			\end{itemize}
		\end{column}
		\begin{column}{0.40\textwidth}
			\imagewithcaption{fedora-iso-hash.png}{Fedora's download page showing the expected SHA-256 checksum.}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Proof of work: Anubis}
	\begin{columns}[c]
		\begin{column}{0.60\textwidth}
			\begin{itemize}
				\item Anubis shields web servers from aggressive bots (e.g.\ AI scrapers) by making each visitor's browser solve a hash puzzle.
				\item \textbf{The challenge}: given a value $C$, find a nonce $N$ such that $\mathsf{SHA256}(C \| N)$ begins with a set number of leading zero bits (the \emph{difficulty}).
				\item Finding $N$ takes brute force over many guesses, but checking a submitted $N$ is one hash: \textbf{hard to solve, easy to verify}.
				\item A human barely notices the delay; a bot making millions of requests pays the cost every time.
			\end{itemize}
		\end{column}
		\begin{column}{0.40\textwidth}
			\imagewithcaption{anubis.png}{Anubis running a proof-of-work challenge in the browser before granting access.}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{AES is a block cipher}
	\begin{itemize}[<+->]
		\item AES takes a 16-byte input, produces a 16-byte output.

\begin{frame}{From primitives to protocols}{Reminder}
	\begin{columns}[c]
		\begin{column}{0.5\textwidth}
			\textbf{The security goals we set out}
			\begin{itemize}
				\item \textbf{Confidentiality}: Data exchanged between Client and Server
				      is only known to those parties.
				\item \textbf{Authentication}: If Server receives data from Client,
				      then Client sent it to Server.
				\item \textbf{Integrity}: If Server modifies data owned by Client,
				      Client can find out.
			\end{itemize}
		\end{column}
		\begin{column}{0.5\textwidth}
			\textbf{The primitives we now have}
			\begin{itemize}
				\item \textbf{Diffie-Hellman}: two strangers agree on a secret key.
				\item \textbf{AES}: encrypt with that key (in a sane mode).
				\item \textbf{SHA-2}: fingerprint anything.

\begin{frame}{From hard problems to real-world security}
	\textbf{What we saw today}
	\begin{enumerate}[<+->]
		\item A \textbf{hard problem} (discrete logarithms) that nobody knows how to solve at scale.
		\item \textbf{Primitives} resting on it, or on decades of cryptanalysis: Diffie-Hellman, AES, SHA-2.
		\item \textbf{Protocols} that compose primitives to reach \textbf{security goals} under a \textbf{threat model}: TLS 1.3, Signal.
	\end{enumerate}
	\pause
	\textbf{Where the course goes}
	\begin{itemize}[<+->]
		\item \textbf{Part 1: Provable Security} (1.2--1.8). Define ``secure'' precisely, build primitives from one another, and \emph{prove} the constructions secure (\textit{The Joy of Cryptography}).
		\item \textbf{Part 2: Real-World Cryptography}. Take apart deployed systems (TLS, secure messaging, encrypted cloud storage, post-quantum, zero-knowledge proofs) and see where they break.
	\end{itemize}
\end{frame}

\section{Course Goals}

\begin{frame}{Course goals}
	\begin{itemize}[<+->]
		\item Understand the reasoning behind the math of modern cryptography.
		\item Analyze and prove the security of cryptographic constructions.
		\item Understand how cryptographic constructions can be composed to build real-world
		      secure protocols and systems.
		\item Distinguish between the mathematical and engineering approaches to
		      cryptography.
		\item Critically assess security implementations and evaluate real-world cryptographic
		      protocols.
		\item Gain an understanding of the future of cryptography and its role in emerging
