\begin{frame}{Today: three kinds of fake randomness}
	\begin{itemize}
		\item In 1.3 we defined security as \textbf{indistinguishability} between a real library and an ideal one, and learned to prove it with hybrids and bad events.
		\item Today we meet the three objects that all of symmetric cryptography is built from. Each is a ``fake'' version of an ideal that is too big to exist:
		      \begin{itemize}
			      \item \textbf{Pseudorandom generator (PRG)}: a short seed that looks like a long uniform string. Stream ciphers.
			      \item \textbf{Pseudorandom function (PRF)}: a short key that looks like a giant dictionary of uniform values. HMAC.
			      \item \textbf{Pseudorandom permutation (PRP)}: a short key that looks like a uniformly shuffled, invertible table. Block ciphers such as AES.
		      \end{itemize}
		\item Along the way: how to break bad constructions (and the golden rule behind every break), how to build a PRP out of a PRF (Feistel), why DES died and why AES hasn't.
	\end{itemize}
\end{frame}

\section{Pseudorandom Generators}

\begin{frame}{Limitations of the one-time pad}
	\definitionbox{The Key Length Problem}{
		One-time pad is not a particularly useful encryption scheme in practice.
	}
	\begin{itemize}
		\item The key must be as long as the plaintext!
		\item This creates a chicken-and-egg situation:
		      \begin{itemize}
			      \item To privately send $n$ bits of information,
			      \item We must already privately share $n$ bits of information.
		      \end{itemize}
		\item Impractical for most real-world applications.
		      \begin{itemize}
			      \item Clearly this is not what we're doing when we use HTTPS,
			      \item or WhatsApp, or pay for something via a debit card\ldots
		      \end{itemize}
		\item We need encryption schemes where the key can be smaller than the message.
	\end{itemize}
\end{frame}

\begin{frame}{Idea: find a way to expand the key}
	\begin{itemize}
		\item Given a key $k_s$ of size $\left|k_s\right| < \left|m\right|$, find a way to obtain a key $k_e$ where $\left|k_e\right| \geq \left|m\right|$.
		\item In the real world, we have two kinds of symmetric encryption schemes:
		      \begin{itemize}
			      \item \textbf{Block ciphers}: AES, 3DES, etc.
			      \item \textbf{Stream ciphers}: ChaCha20, RC4, etc.
		      \end{itemize}
		\item This is exactly what stream ciphers do!
		      \begin{itemize}
			      \item Start with a small key $k_s$ of a fixed size $\left|k_s\right| = \lambda$,
			      \item Magically expand it to $k_e$ where $\left|k_e\right| \geq \left|m\right|$,
			      \item $\func{enc}{k_e, M} = k_e \oplus M$
		      \end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{Requirements for key expansion}
	\begin{itemize}
		\item Our method would need to be \textbf{deterministic}, so that both the sender and receiver can expand their key in the same way (to encrypt/decrypt).
		\item Its output distribution would need to be \textbf{uniform}, since that is a crucial property for the security of OTP.
		\item Unfortunately, it's \textbf{not possible} to achieve both of these properties simultaneously.
		      \begin{itemize}
			      \item Suppose the expansion method is a deterministic function $G: \bits^\lambda \rightarrow \bits^{\lambda+\ell}$.
			      \item Its outputs are $\ell$ bits longer than its inputs!
			      \item There are $2^{\lambda+\ell}$ strings of length $\lambda+\ell$ but only (at most) $2^\lambda$ possible outputs of $G$.
			      \item So the outputs of $G$ can never induce a uniform distribution.
		      \end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{Example: RC4 (a stream cipher)}
	\begin{columns}[c]
		\begin{column}{0.6\textwidth}
			\begin{itemize}
				\item Designed by Ron Rivest at RSA Security in 1987; leaked in 1994.
				\item The most widely deployed stream cipher ever: WEP, SSL/TLS, Kerberos, PDF, Office\ldots
				\item A PRG in exactly our sense: the key (40--2048 bits, usually 128) is the seed; the output is an endless \textbf{keystream} of bytes, XORed with the message.
				\item The whole state: a permutation $S$ of the 256 byte values and two indices.
				\item Its keystream is biased. Topic 2.2 tells the whole story.
			\end{itemize}
		\end{column}
		\begin{column}{0.4\textwidth}
			\alertbox{Spoiler}{
				RC4 is dead: banned from TLS in 2015 (RFC 7465). We study it because it is tiny, because it carried much of the web for two decades, and because its slow death is a lesson in itself.
			}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{RC4's keystream generator}
	\begin{columns}[c]
		\begin{column}{0.6\textwidth}
			\begin{itemize}
				\item \textbf{Key setup} (not shown) scrambles $S = (0, 1, \ldots, 255)$ with the key.
				\item Each keystream byte then costs two additions, a swap and a lookup:
				      \begin{itemize}
					      \item $i \coloneq i + 1$, \ $j \coloneq j + S[i]$ \ (mod 256),
					      \item swap $S[i]$ and $S[j]$,
					      \item output $S[S[i] + S[j]]$.
				      \end{itemize}
				\item Ten lines of Python, much faster than DES in software: that is why it was everywhere.
				\item Swapping keeps $S$ a permutation. Remember this when we get to PRPs.
			\end{itemize}
		\end{column}
		\begin{column}{0.4\textwidth}
			\imagewithcaption{rc4-prga-py.png}{RC4's keystream generation (PRGA) in Python.}
		\end{column}
	\end{columns}
\end{frame}

\section{Pseudorandom Functions}

\begin{frame}{From one long string to a giant dictionary}
	\begin{itemize}
		\item A PRG hands us \textbf{one long string} $G(S)$, in order. To use its millionth block, we must first generate the 999,999 blocks before it.
		\item What Alice and Bob would really like to share is a \textbf{dictionary} $L$ with $2^n$ entries, each a fresh uniform $m$-bit value, that either of them can look up in any order: $L[X]$.
		\item Storing it is hopeless: for $n = m = 128$ that is $2^{128}$ entries of 128 bits each.
		\item A \textbf{pseudorandom function} is a compressed dictionary. The $\lambda$-bit key $K$ stands in for the whole table, and $F(K, X)$ computes entry $X$ on demand.
		\item Security goal: without $K$, nobody can tell the compressed dictionary $F(K, \cdot)$ from a real one filled with uniform entries.
	\end{itemize}
\end{frame}

\begin{frame}{Pseudorandom function: definition}
	\begin{columns}[c]
		\begin{column}{0.48\textwidth}
			\definitionbox{Pseudorandom Function (PRF)}{
				A function $F: \bits^\lambda \times \bits^n \rightarrow \bits^m$ is a secure pseudorandom function (PRF) if the following two libraries are indistinguishable:
			}
			\begin{itemize}
				\item $n$ is the input length of the PRF.
				\item $m$ is the output length of the PRF.
				\item $\lambda$ is the key size and hence the security parameter.
			\end{itemize}
		\end{column}
		\begin{column}{0.52\textwidth}
			\sslinked{
				\sslibrary{F}{prf-real}{
					$K \twoheadleftarrow \bits^\lambda$\\[1em]
					\sslibrarysubroutine{prf.query}{X}{
						return $F(K, X)$
					}{1}
				}{1}
			}{\approxeq}{
				\sslibrary{F}{prf-rand}{
					$L \coloneq$ [\ ]\\[1em]
					\sslibrarysubroutine{prf.query}{X}{
						if $L[X]$ undefined:\\
						\quad $L[X] \twoheadleftarrow \bits^m$\\
						return $L[X]$
					}{1}
				}{1}
			}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Cousins of PRFs: hash functions (reminder from 1.1)}
	\begin{columns}[c]
		\begin{column}{0.55\textwidth}
			\definitionbox{Hash Function Properties}{
				\begin{itemize}\item Takes input of \textbf{any size}
					\item Produces output of \textbf{fixed size}
					\item Is \textbf{deterministic} (same input $\rightarrow$ same output)
					\item Even a \textbf{tiny change} in input creates completely different output
					\item Is \textbf{efficient} to compute\end{itemize}

		\end{column}

\begin{frame}{Expected properties of a hash function}
	\begin{columns}[c]
		\begin{column}{0.6\textwidth}
			\begin{itemize}
				\item \textbf{Collision resistance}: computationally infeasible to find
				      two different inputs producing the same hash.
				\item \textbf{Preimage resistance}: given an output of a hash function,
				      it is computationally infeasible to find \emph{any} input that hashes to it.
				\item \textbf{Second preimage resistance}: given an input and an output,
				      it's computationally infeasible to find a different input
				      producing the same output.
			\end{itemize}
		\end{column}
		\begin{column}{0.4\textwidth}
			\imagewithcaption{sha2.png}{SHA-2 compression function. Source: Wikipedia}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Hash functions: what are they good for?}
	\begin{itemize}
		\item \textbf{Data integrity verification}: Hash a file. Later hash it
		      again and compare hashes to check if the file has changed, suffered storage
		      degradation, etc.
		\item \textbf{Proof of work}: Server asks client to hash something a lot of
		      times before they can access some resource. Useful for anti-spam, Bitcoin
		      mining, etc.
	\end{itemize}
\end{frame}

\begin{frame}{Data integrity: verifying a Linux ISO download}
	\begin{columns}[c]
		\begin{column}{0.60\textwidth}
			\begin{itemize}
				\item Linux distributions publish a hash (usually SHA-256) for each ISO image they release.
				\item After downloading, you hash your own copy and compare it to the published value:
				      \begin{center}
					      {\small $\mathsf{SHA256}(\texttt{your.iso}) \stackrel{?}{=} \texttt{1620295f\ldots}$}
				      \end{center}
				\item A match means the file arrived \textbf{intact}: no corruption from a flaky network or a failing disk.
				\item This only catches \emph{accidental} corruption: a malicious mirror could swap both the ISO \emph{and} the hash, which is why real verification also uses a signature (OpenPGP).
			\end{itemize}
		\end{column}
		\begin{column}{0.40\textwidth}
			\imagewithcaption{fedora-iso-hash.png}{Fedora's download page showing the expected SHA-256 checksum.}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Proof of work: Anubis}
	\begin{columns}[c]
		\begin{column}{0.60\textwidth}
			\begin{itemize}
				\item Anubis shields web servers from aggressive bots (e.g.\ AI scrapers) by making each visitor's browser solve a hash puzzle.
				\item \textbf{The challenge}: given a value $C$, find a nonce $N$ such that $\mathsf{SHA256}(C \| N)$ begins with a set number of leading zero bits (the \emph{difficulty}).
				\item Finding $N$ takes brute force over many guesses, but checking a submitted $N$ is one hash: \textbf{hard to solve, easy to verify}.
				\item A human barely notices the delay; a bot making millions of requests pays the cost every time.
			\end{itemize}
		\end{column}
		\begin{column}{0.40\textwidth}
			\imagewithcaption{anubis.png}{Anubis running a proof-of-work challenge in the browser before granting access.}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{An insecure PRF construction}
	\begin{columns}[c]
		\begin{column}{0.45\textwidth}
			\definitionbox{Claim: An Insecure PRF}{
				The function $F(K, X) = G(K) \oplus X$ is not a secure PRF, even if $G$ is a secure PRG.
			}
			\begin{itemize}[<+->]
				\item This construction fails because:
				      \begin{itemize}
					      \item The key $K$ is only fed through the PRG once.
					      \item The same value $G(K)$ is used for all queries.
					      \item This creates exploitable patterns in outputs.
				      \end{itemize}
			\end{itemize}
		\end{column}
		\begin{column}{0.55\textwidth}
			\begin{center}
				\sslinked{
					\sslibrary{F}{prf-real}{
						$K \twoheadleftarrow \bits^\lambda$\\[1em]
						\sslibrarysubroutine{prf.query}{X}{
							return $G(K) \oplus X$
						}{1}
					}{1}
				}{\cancel{\approxeq}}{
					\sslibrary{F}{prf-rand}{
						$L \coloneq$ [\ ]\\[1em]
						\sslibrarysubroutine{prf.query}{X}{
							if $L[X]$ undefined:\\
							\quad $L[X] \twoheadleftarrow \bits^m$\\
							return $L[X]$
						}{1}
					}{1}
				}
			\end{center}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{An insecure PRF construction}
	\begin{columns}[c]
		\begin{column}{0.6\textwidth}
			\begin{itemize}
				\item When an adversary sees two outputs:
				      \begin{align*}
					      Y_1 & = G(K) \oplus X_1 \\
					      Y_2 & = G(K) \oplus X_2
				      \end{align*}
				\item Taking $Y_1 \oplus Y_2$ causes $G(K)$ to cancel:
				      \begin{align*}
					      Y_1 \oplus Y_2 & = G(K) \oplus X_1 \oplus G(K) \oplus X_2 \\
					                     & = X_1 \oplus X_2
				      \end{align*}
				\item With a truly random function, $Y_1 \oplus Y_2 = X_1 \oplus X_2$ would hold with probability only $2^{-m}$.
			\end{itemize}
		\end{column}
		\begin{column}{0.4\textwidth}
			\begin{center}
				\sslinked{
					\sslibrary{F}{prf-real}{
						$K \twoheadleftarrow \bits^\lambda$\\[1em]
						\sslibrarysubroutine{prf.query}{X}{
							return $G(K) \oplus X$
						}{1}
					}{0.7}
				}{\cancel{\approxeq}}{
					\sslibrary{F}{prf-rand}{
						$L \coloneq$ [\ ]\\[1em]
						\sslibrarysubroutine{prf.query}{X}{
							if $L[X]$ undefined:\\
							\quad $L[X] \twoheadleftarrow \bits^m$\\

\begin{frame}{Another insecure PRF construction}
	\begin{columns}[c]
		\begin{column}{0.5\textwidth}
			\definitionbox{Claim: Another Insecure PRF}{
				The function $H(K_1 \| K_2, X_1 \| X_2) = F(K_1, X_1) \oplus F(K_2, X_2)$ is not a secure PRF, even if $F$ is a secure PRF.
			}
			\begin{itemize}
				\item An unsuccessful attempt to use a PRF with a shorter input length to build one with a larger input length.
			\end{itemize}
		\end{column}
		\begin{column}{0.55\textwidth}
			\begin{center}
				\sslinked{
					\sslibrary{H}{prf-real}{
						$K_1 \| K_2 \twoheadleftarrow \bits^{2\lambda}$\\[1em]
						\sslibrarysubroutine{prf.query}{X_1 \| X_2}{
							return $F(K_1, X_1) \oplus F(K_2, X_2)$
						}{1}
					}{0.7}
				}{\cancel{\approxeq}}{
					\sslibrary{H}{prf-rand}{
						$L \coloneq$ [\ ]\\[1em]
						\sslibrarysubroutine{prf.query}{X_1 \| X_2}{
							if $L[X_1 \| X_2]$ undefined:\\
							\quad $L[X_1 \| X_2] \twoheadleftarrow \bits^m$\\
							return $L[X_1 \| X_2]$
						}{1}
					}{0.7}
				}
			\end{center}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Why the previous PRF is broken}
	\begin{columns}[c]
		\begin{column}{0.5\textwidth}
			\begin{itemize}[<+->]
				\item Query four inputs $A\|B$, $A\|B'$, $A'\|B$, $A'\|B'$ (with $A \neq A'$, $B \neq B'$). Outputs that share a first half share the term $F(K_1, \cdot)$:
				      \begin{align*}
					      Y_1 & = F(K_1, A) \oplus F(K_2, B)   \\
					      Y_2 & = F(K_1, A) \oplus F(K_2, B')  \\
					      Y_3 & = F(K_1, A') \oplus F(K_2, B)  \\
					      Y_4 & = F(K_1, A') \oplus F(K_2, B')
				      \end{align*}
			\end{itemize}
		\end{column}
		\begin{column}{0.5\textwidth}
			\begin{itemize}[<+->]
				\item XOR the pairs that share a first half, and the $F(K_1, \cdot)$ terms cancel:
				      \begin{align*}
					      Y_1 \oplus Y_2 & = F(K_2, B) \oplus F(K_2, B') \\
					      Y_3 \oplus Y_4 & = F(K_2, B) \oplus F(K_2, B')
				      \end{align*}
				\item So $\prob{Y_1 \oplus Y_2 = Y_3 \oplus Y_4} = 1$ for $H$, but only $\frac{1}{2^m}$ for a truly random function.
				\item Also, given $Y_1, Y_2, Y_3$, we can predict $Y_4 = Y_1 \oplus Y_2 \oplus Y_3$!
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{The golden rule of PRFs}
	\definitionbox{The Golden Rule of PRFs}{
		If a PRF $F$ is being used as a component in a larger construction $H$, then security usually rests on how well $H$ can ensure distinct inputs to $F$.
	}
	\begin{itemize}[<+->]
		\item Both attacks forced $F$ to see the \textbf{same input twice}, then XORed it away.
		\item On new inputs a secure PRF is fresh coin flips; on a repeated input it is a deterministic function, and that is what gets exploited.
		\item Attacking a construction: don't fight $F$ (you can't), hunt for the input collision that $H$ lets you cause. Designing one: give every call to $F$ a fresh input. That is what nonces and counters are for.
	\end{itemize}
\end{frame}

\begin{frame}{Your turn: secure PRF or not?}
	\begin{columns}[c]
		\begin{column}{0.52\textwidth}
			Let $F: \bits^\lambda \times \bits^n \rightarrow \bits^n$ be a secure PRF, and write $\overline{X}$ for the bitwise complement of $X$. Which of these are secure PRFs?
			\begin{enumerate}
				\item $H_1(K, X) = F(K, X) \oplus X$
				\item $H_2(K, X) = F(K, X) \oplus F(K, \overline{X})$
				\item $H_3(K, X_1 \| X_2) = F(K, X_1) \oplus F(K, X_2)$
				\item $H_4(K_1 \| K_2, X) = F(K_1, X) \oplus F(K_2, X)$
			\end{enumerate}
			Golden rule: can you make $F$ see the same input twice?
		\end{column}
		\pause
		\begin{column}{0.48\textwidth}
			\begin{itemize}
				\item $H_1$: \textbf{secure}. Each new $X$ gets a fresh uniform $F(K, X)$, and XORing a known value into a uniform value keeps it uniform.
				\item $H_2$: \textbf{broken}. $H_2(K, X) = H_2(K, \overline{X})$, always.
				\item $H_3$: \textbf{broken}. $H_3(K, X \| X) = 0^n$, always.
				\item $H_4$: \textbf{secure}. $F(K_1, X)$ alone already makes every output fresh, whatever $F(K_2, X)$ does.
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}

\section{Pseudorandom Permutations}

\begin{frame}{What is a permutation?}
	\begin{columns}[c]
		\begin{column}{0.5\textwidth}
			\definitionbox{Permutation}{
				A permutation is a rearrangement where each input value maps to exactly one output value, and each possible output appears exactly once.
			}
			\begin{itemize}[<+->]
				\item Permutations rearrange elements rather than transforming them.
				\item Every element in the domain appears exactly once in the range.
				\item The function is invertible.
			\end{itemize}
		\end{column}
		\begin{column}{0.5\textwidth}
			\begin{itemize}[<+->]
				\item \textbf{Example}: a simple substitution cipher maps each letter to another letter:
				      \begin{align*}
					      a & \mapsto g,\quad b \mapsto a,\quad c \mapsto r, \\
					      d & \mapsto b,\quad e \mapsto l,\quad g \mapsto d,\quad \ldots
				      \end{align*}
				\item ``cabbage'' $\mapsto$ ``rgaagdl''. Under the \textbf{inverse} (table read right to left), ``rgaagdl'' $\mapsto$ ``cabbage''.
				\item This works because it's a permutation over $\{a,\ldots,z\}$: each letter appears exactly once in the output alphabet.
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{PRF versus PRP}
	\begin{columns}[c]
		\begin{column}{0.5\textwidth}
			\textbf{Pseudorandom Function (HMAC-SHA-256)}
			\begin{itemize}
				\item \textbf{Input} is arbitrary-length,
				\item \textbf{Output} is fixed-length, looks random (as discussed
				      earlier).
				\item Indistinguishable from a truly random function by an adversary with
				      limited computational power.
			\end{itemize}
		\end{column}
		\pause
		\begin{column}{0.5\textwidth}
			\textbf{Pseudorandom Permutation (AES)}
			\begin{itemize}
				\item \textbf{Input and output} are the same length, forming a permutation.
				\item Each input maps uniquely to one output, allowing invertibility.
				\item Indistinguishable from a truly random permutation by an adversary
				      with limited computational power.
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{PRPs compared to PRFs}
	\begin{itemize}[<+->]
		\item \textbf{Invertibility}: given $K$, we can compute $F^{-1}(K, \cdot)$ efficiently. That is what lets a block cipher decrypt.
		\item \textbf{No collisions}: distinct inputs give distinct outputs. A random \emph{function} on $n$ bits starts repeating outputs after about $2^{n/2}$ queries (the birthday bound from 1.3); a permutation never does.
		\item That is the \emph{only} difference an adversary can see. \textbf{PRF/PRP switching lemma}: a secure PRP is also a secure PRF as long as far fewer than $2^{n/2}$ queries are made. After $q$ queries, the two ideal worlds differ by at most $q(q-1)/2^{n+1}$.
		      \begin{itemize}
			      \item For AES ($n = 128$): treating it as a PRF is safe until about $2^{64}$ blocks.
		      \end{itemize}
		\item The other direction is not free: turning a PRF into a PRP takes real work. That is the Feistel construction, coming up.
	\end{itemize}
\end{frame}

\begin{frame}{Building a permutation through Feistel ciphers}
	\begin{columns}[c]
		\begin{column}{0.5\textwidth}
			\begin{itemize}
				\item An $r$-round \textbf{Feistel cipher} with \textbf{round functions} $F_1,\ldots,F_r$ is defined as follows:\\
				      \begin{center}
					      \sssubroutine{$\mathbb{F}$}{X_{0}\|X_{1}}{
						      for $i = 1$ to $r$:\\
						      \quad $X_{i+1} \coloneq X_{i-1} \oplus F_{i}(X_{i})$\\
						      return $X_{r}\|X_{r+1}$
					      }{1}
				      \end{center}
				\item A Feistel cipher is always a permutation on $\bits^{2n}$, regardless of its round functions.
			\end{itemize}

\begin{frame}{Building a permutation through Feistel ciphers}
	\definitionbox{Feistel Ciphers Are Permutations}{
		A Feistel cipher is always a permutation on $\bits^{2n}$, regardless of its round functions.
	}
	\begin{itemize}
		\item \textbf{Proof:} Each round of a Feistel cipher computes the next block as:
		      \begin{itemize}
			      \item $X_{i+1} = X_{i-1} \oplus F_{i}(X_{i})$
		      \end{itemize}
		\item To invert this round, we can rearrange the equation to solve for $X_{i-1}$ in terms of $X_{i}$ and $X_{i+1}$:
		      \begin{itemize}
			      \item $X_{i-1} = X_{i+1} \oplus F_{i}(X_{i})$
		      \end{itemize}
		\item $F_i$ itself does not need to have an inverse: both the forward and inverse directions of the Feistel cipher evaluate $F_i$ in the forward direction! $\qed$
	\end{itemize}
\end{frame}

\begin{frame}{Running a Feistel cipher backwards}
	\begin{columns}[c]

				\item $\mathbb{F}^{-1}$ runs the rounds \textbf{backwards}: anyone who knows the round functions can undo $\mathbb{F}$:
				      \begin{center}
					      \sssubroutine{$\mathbb{F}^{-1}$}{X_{r}\|X_{r+1}}{
						      for $i = r$ \hl{down to} $1$:\\
						      \quad $X_{i-1} \coloneq X_{i+1} \oplus F_{i}(X_{i})$\\
						      return $X_{0}\|X_{1}$
					      }{1}
				      \end{center}
			\end{itemize}
		\end{column}

\begin{frame}{Keyed Feistel ciphers}
	\begin{columns}[c]
		\begin{column}{0.6\textwidth}
			\begin{itemize}
				\item Let's add a key in there. Wow, encryption!
				\item $K_{1\cdots r}$ is called the \textbf{key schedule}.
				\item Why must the round keys differ? Next slide.

			\begin{center}

\begin{frame}{Why a key schedule? The same key in every round is fatal}
	\begin{columns}[c]
		\begin{column}{0.6\textwidth}
			\begin{itemize}
				\item Decryption is encryption with the round functions in \textbf{reverse order} (previous slides).
				\item Same $F(K, \cdot)$ in every round? Then reversing changes nothing: the cipher is its own inverse, up to swapping the halves.
				\item Distinguisher: encrypt $A\|B$ to get $Y_1\|Y_2$, then encrypt $Y_2\|Y_1$: out comes $B\|A$, every time. An encryption oracle is now a decryption oracle.
				\item This really happened: DES's four \textbf{weak keys} make all 16 round keys identical, so $\mathsf{DES}_K(\mathsf{DES}_K(M)) = M$.
			\end{itemize}
		\end{column}
		\begin{column}{0.4\textwidth}
			\sssubroutine{$\mathbb{F}$}{K,\ X_0\|X_1}{
				for $i = 1$ to $r$:\\
				\quad $X_{i+1} \coloneq X_{i-1} \oplus F(K, X_{i})$\\
				return $X_{r}\|X_{r+1}$
			}{1}

				\ssprogram{}{

\begin{frame}{Luby and Rackoff: three rounds suffice}
	\begin{itemize}[<+->]
		\item \textbf{Luby and Rackoff (1988)} proved it:\footnote{\url{https://appliedcryptography.page/paper/\#luby-rackoff}} if $F$ is a secure PRF, then the 3-round Feistel cipher is a secure PRP.
		\item With 4 rounds it stays secure even against an adversary who may also query $\mathbb{F}^{-1}$ (a \textbf{strong} PRP, which is what a block cipher used for decryption needs).
		\item The proof is a bad-event argument in the style of 1.3. The bad event: two queries collide on their $X_2$ value or on their $X_3$ value. Over $q$ queries that has probability about $q^2/2^n$, and so does the adversary's advantage.
		\item So with a good round function, three rounds turn a PRF into a PRP. Real ciphers use many more (DES: 16) because their round functions are nowhere near ideal PRFs.
	\end{itemize}
\end{frame}

	\begin{frame}{The ``bad event'' proof technique}{Reminder}
		\definitionbox{Bad Event Technique}{
			Let \lib{}{1} and \lib{}{2} be libraries that each include a boolean variable named \hl{\texttt{bad}}, and assume that after \texttt{bad} is set to \texttt{true} it remains \texttt{true} forever. We say that the bad event is triggered if the library ever sets \texttt{bad := true}.
			\\[1em]
			If \lib{}{1} and \lib{}{2} have identical source code, except for statements reachable only when \texttt{bad := true}, then:
			\begin{align*}
				\left| \prob{\prog{} \link \lib{}{1} \Rightarrow \texttt{true}} - \prob{\prog{} \link \lib{}{2} \Rightarrow \texttt{true}} \right| \leq \prob{\prog{} \link \lib{}{1} \func{trigger}{\texttt{bad}}}
			\end{align*}
		}
	\end{frame}

	\begin{frame}{The ``bad event'' proof technique}{Reminder}
		\begin{itemize}
			\item \prog{}'s advantage is bounded by \prob{\prog{} \link \lib{}{1} \func{trigger}{\texttt{bad}}}.
			\item Practical application:
			      \begin{itemize}
				      \item Define a sequence of hybrid libraries.
				      \item Identify ``bad events'' between consecutive hybrids.
				      \item Show these events occur with negligible probability.
			      \end{itemize}
			\item Enables us to focus on analyzing specific failure cases rather than full behavior.
		\end{itemize}
	\end{frame}

	\begin{frame}{The ``end-of-time'' strategy for bad events}{Reminder}
		\begin{itemize}
			\item Sometimes analyzing bad events can be complex, especially when values are chosen by the adversary.
			\item The end-of-time strategy:
			      \begin{enumerate}
				      \item Postpone all bad-event logic to the end of the library execution.
				      \item Collect information during normal execution.
				      \item Check for bad events only at the very end.
			      \end{enumerate}
			\item Advantages:
			      \begin{itemize}
				      \item Simplifies analysis by separating normal behavior from bad-event checking.
				      \item Makes it easier to bound the probability of bad events.
				      \item Particularly useful for complex cryptographic proofs.
			      \end{itemize}
		\end{itemize}
	\end{frame}

					\item Note how nothing after $L[X_0\|X_1] \coloneq X_3\|X_4$ affects what the adversary sees!
					\item So, we can move all bad-event logic to the end of time, without changing the bad event's overall probability.
				\end{itemize}
			\end{column}
			\begin{column}{0.5\textwidth}
				\vspace{-1.5cm}
				\begin{center}
					\sslibrary{\mathbb{F}}{prp-real}{
						\sslibrarysubroutine{$\text{prp.query}_{\mathbb{F}}$}{X_0\|X_1}{
							if $L[X_0\|X_1]$ undefined:\\
							\quad \hl{$L[X_0\|X_1] \coloneq \bits^{2\lambda}$}\\
							\quad \hl{$\mathcal{X} \coloneq \mathcal{X} \cup \{X_0\|X_1\}$}\\
							\quad return $L[X_0\|X_1]$
						}{1}\\[1em]
						\sslibrarysubroutine{\hl{end of time}}{}{
							\hl{for each $X_0\|X_1 \in \mathcal{X}:$}\\
							\quad \hl{$X_3\|X_4 \coloneq L[X_0\|X_1]$}\\
							\quad if $L_1[X_1]$ undefined:\\
							\quad \quad $L_1[X_1] \twoheadleftarrow \bits^\lambda$\\
							\quad $X_{2} \coloneq X_{0} \oplus L_1[X_1]$\\
							\quad if $L_2[X_2]$ defined: $\texttt{bad} \coloneq \texttt{true}$\\
							\quad $L_2[X_2] \coloneq X_3 \oplus X_1$\\
							\quad if $L_3[X_3]$ defined: $\texttt{bad} \coloneq \texttt{true}$\\
							\quad $L_3[X_3] \coloneq X_4 \oplus X_2$

	\begin{frame}{Proof of security for 3-round Feistel cipher}{Step 9 (this content is optional)}
		\begin{columns}[c]
			\begin{column}{0.6\textwidth}
				\begin{itemize}
					\item So, we need to analyze when $X_2$ collisions occur.

	\begin{frame}{Proof of security for 3-round Feistel cipher}{Step 10 - Total probability analysis}
		\begin{columns}[c]
			\begin{column}{0.6\textwidth}
				\begin{itemize}
					\item For $q$ queries, we have at most $q(q-1)/2$ pairs of queries
					\item Prob. of any $X_2$ collision across $q$ queries:
					      $q(q-1)/(2 \cdot 2^{\lambda}) \approx q^2/2^{\lambda+1}$
					\item Similarly for $X_3$ values: the probability of any $X_3$ collision is at most $q^2/2^{\lambda+1}$.
					\item Total probability of bad event: at most $q^2/2^{\lambda}$
					\item With $\lambda \gg \log q$, this probability is negligible.
				\end{itemize}
			\end{column}
			\begin{column}{0.4\textwidth}

\begin{frame}{DES: Feistel in practice}
	\begin{columns}[c]
		\begin{column}{0.5\textwidth}
			\begin{itemize}
				\item \textbf{Data Encryption Standard (DES)} is the classic real-world Feistel cipher:
				      \begin{itemize}
					      \item 16 Feistel rounds on a 64-bit block.
					      \item 56-bit key (64 with parity), expanded into 16 round keys of 48 bits.
					      \item IBM design, NSA tweaks, standardized in 1977.
				      \end{itemize}
				\item Problem: by the 1990s, $2^{56}$ was a number you could count to.
			\end{itemize}
		\end{column}
		\begin{column}{0.5\textwidth}
			\begin{itemize}
				\item \textbf{Triple DES}: $C = E_{K_3}(D_{K_2}(E_{K_1}(P)))$. With $K_1 = K_2 = K_3$ it is plain DES.
				\item Why not \emph{double} DES? \textbf{Meet-in-the-middle}: encrypt $P$ under every $K_1$, decrypt $C$ under every $K_2$, match: about $2^{57}$ work, not $2^{112}$. The same attack caps 3DES at 112-bit security.
				\item Slow, and its 64-bit block hits the birthday bound at $2^{32}$ blocks. NIST disallowed it for encryption after 2023.
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{AES: a good example of a PRP}
	\begin{columns}[c]
		\begin{column}{0.5\textwidth}
			\begin{itemize}
				\item AES is Rijndael, by Joan Daemen and Vincent Rijmen. It won NIST's open competition in 2000 and became FIPS 197 in 2001.
				\item It is the most widely used PRP in the world.
				\item Block size: 128 bits. Key sizes: 128, 192 or 256 bits.
				\item Each key selects one permutation of the $2^{128}$ possible blocks; different keys, different permutations.
			\end{itemize}
		\end{column}
		\begin{column}{0.5\textwidth}
			\begin{itemize}
				\item Efficiently invertible: $\mathsf{Dec}(K, \mathsf{Enc}(K, M)) = M$.
				\item Unlike DES, AES is \textbf{not} a Feistel network: it is a substitution--permutation network in which every step is invertible by design.
				\item Believed to be computationally indistinguishable from a random permutation: it has withstood open cryptanalysis since 2001.
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{AES: security and attacks over time}
	\begin{columns}[c]
		\begin{column}{0.5\textwidth}
			\begin{itemize}
				\item 25 years of open analysis. The best attacks on the full cipher:
				      \begin{itemize}
					      \item 2011: biclique attack (Bogdanov, Khovratovich, Rechberger), $2^{126.1}$ for AES-128 versus $2^{128}$ for brute force.
					      \item 2009: related-key attacks on AES-256 in $2^{99.5}$ (Biryukov, Khovratovich), which need related keys that no sane protocol hands out.
					      \item Side channels (power analysis, cache timing): how AES actually gets broken in practice.\footnote{Side-channel attacks will be discussed in more depth later in the course.}
				      \end{itemize}
			\end{itemize}
		\end{column}
		\begin{column}{0.5\textwidth}
			\begin{itemize}
				\item So: no practical attack on full AES in 25 years. Shaving two bits off 128 is a scientific result, not a threat.
				\item Quantum computers change little here: Grover's algorithm brings AES-128 down to about $2^{64}$ quantum operations, and AES-256 to $2^{128}$.\footnote{More on quantum computers and how they affect cryptography later in the course.}
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Recap: three kinds of fake randomness}
	\begin{columns}[c]
		\begin{column}{0.5\textwidth}
			\begin{itemize}
				\item \textbf{PRG} $G: \bits^\lambda \rightarrow \bits^{\lambda+\ell}$: one short seed, one long random-looking string. Stream ciphers: ChaCha20 (and, once, RC4).
				\item \textbf{PRF} $F: \bits^\lambda \times \bits^n \rightarrow \bits^m$: one key, a giant random-looking dictionary you can read in any order. HMAC-SHA-256.
				\item \textbf{PRP} $F: \bits^\lambda \times \bits^n \rightarrow \bits^n$: the same, but a shuffle you can undo. Block ciphers: DES, AES.
			\end{itemize}
		\end{column}
		\begin{column}{0.5\textwidth}
			\begin{itemize}
				\item All three are defined the same way: no efficient program can tell the real library from the ideal one.
				\item \textbf{Golden rule}: a PRF-based construction is only as secure as its guarantee that $F$ never sees an input twice.
				\item PRP $\approx$ PRF below the birthday bound; PRF $\rightarrow$ PRP with three Feistel rounds.
				\item Next, 1.5: encrypting \emph{many} messages under one key: CPA security and block-cipher modes.
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}
