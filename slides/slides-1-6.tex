\section{Defining Hash Functions}

\begin{frame}{Symmetric primitive example: hash functions}{Reminder}
	\begin{columns}[c]
		\begin{column}{0.55\textwidth}
			\definitionbox{Hash Function Properties}{
				\begin{itemize}\item Takes input of \textbf{any size}
					\item Produces output of \textbf{fixed size}
					\item Is \textbf{deterministic} (same input $\rightarrow$ same output)
					\item Even a \textbf{tiny change} in input creates completely different output
					\item Is \textbf{efficient} to compute\end{itemize}
			}
		\end{column}

					\texttt{3362938b9824}}\\[0.6em]
				$\mathsf{SHA256}(\texttt{hullo}) =$ \\ {\small\texttt{7835066a1457504217688c8f5d}\\
					\texttt{06909c6591e0ca78c254ccf174}\\ \texttt{50d0d999cab0}}
			\end{tcolorbox}
			\textcolor{cipherprimary}{\textbf{Note:} \small One character change $\rightarrow$
				completely different hash!}

\begin{frame}{Expected properties of a hash function}{Reminder}
	\begin{columns}[c]
		\begin{column}{0.6\textwidth}
			\begin{itemize}
				\item \textbf{Collision resistance}: computationally infeasible to find
				      two different inputs producing the same hash.
				\item \textbf{Preimage resistance}: given an output of a hash function,
				      it is computationally infeasible to find \emph{any} input that hashes to it.
				\item \textbf{Second preimage resistance}: given an input and an output,
				      it's computationally infeasible to find a different input
				      producing the same output.
			\end{itemize}
		\end{column}
		\begin{column}{0.4\textwidth}
			\imagewithcaption{sha2.png}{SHA-2 compression function. Source: Wikipedia}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Hash functions: what are they good for?}{Reminder}
	\begin{itemize}
		\item \textbf{Data integrity verification}: Hash a file. Later hash it
		      again and compare hashes to check if the file has changed, suffered storage
		      degradation, etc.
		\item \textbf{Proof of work}: Server asks client to hash something a lot of
		      times before they can access some resource. Useful for anti-spam, Bitcoin
		      mining, etc.
	\end{itemize}
\end{frame}

\begin{frame}{Data integrity: verifying a Linux ISO download}
	\begin{columns}[c]
		\begin{column}{0.60\textwidth}
			\begin{itemize}
				\item Linux distributions publish a hash (usually SHA-256) for each ISO image they release.
				\item After downloading, you hash your own copy and compare it to the published value:
				      \begin{center}
					      {\small $\mathsf{SHA256}(\texttt{your.iso}) \stackrel{?}{=} \texttt{1620295f\ldots}$}
				      \end{center}
				\item A match means the file arrived \textbf{intact}: no corruption from a flaky network or a failing disk.
				\item This only catches \emph{accidental} corruption: a malicious mirror could swap both the ISO \emph{and} the hash, which is why real verification also uses a signature (OpenPGP).
			\end{itemize}
		\end{column}
		\begin{column}{0.40\textwidth}
			\imagewithcaption{fedora-iso-hash.png}{Fedora's download page showing the expected SHA-256 checksum.}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Proof of work: Anubis}
	\begin{columns}[c]
		\begin{column}{0.60\textwidth}
			\begin{itemize}
				\item Anubis shields web servers from aggressive bots (e.g.\ AI scrapers) by making each visitor's browser solve a hash puzzle.
				\item \textbf{The challenge}: given a value $C$, find a nonce $N$ such that $\mathsf{SHA256}(C \| N)$ begins with a set number of leading zero bits (the \emph{difficulty}).
				\item Finding $N$ takes brute force over many guesses, but checking a submitted $N$ is one hash: \textbf{hard to solve, easy to verify}.
				\item A human barely notices the delay; a bot making millions of requests pays the cost every time.
			\end{itemize}
		\end{column}
		\begin{column}{0.40\textwidth}
			\imagewithcaption{anubis.png}{Anubis running a proof-of-work challenge in the browser before granting access.}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Hash function collisions}
	\begin{columns}[c]
		\begin{column}{0.55\textwidth}
			\definitionbox{Hash Function Collision}{
				A \textbf{collision} occurs when two different inputs produce the same hash output:
				$x \neq y$ but $H(x) = H(y)$
			}
			\begin{itemize}
				\item Collisions \textbf{must exist} (pigeonhole principle): infinitely many inputs, finitely many outputs.
				\item Security means \emph{finding} one is infeasible. The generic cost is $\approx 2^{n/2}$ for an $n$-bit hash (birthday paradox, next).
			\end{itemize}
		\end{column}
		\begin{column}{0.45\textwidth}
			\begin{tcolorbox}[colback=black!5!white,colframe=ciphergray]
				\textbf{Collision Example:}

				\vspace{0.2cm}

				\texttt{Input 1: document.pdf}\\
				$\mathsf{SHA256} = \texttt{a1b2c3...}$

				\vspace{0.2cm}

				\texttt{Input 2: malware.exe}\\
				$\mathsf{SHA256} = \texttt{a1b2c3...}$

				\vspace{0.2cm}

				\textcolor{cipherprimary}{\textbf{Problem:} \small Attacker could substitute malware.exe for document.pdf without detection!}
			\end{tcolorbox}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Collision resistance, formally}{Library-style definition (The Joy of Cryptography)}
	\begin{columns}[c]
		\begin{column}{0.52\textwidth}
			\begin{center}
				\sslinked{
					\sslibrary{H}{cr-real}{
						$s \twoheadleftarrow \bits^\lambda$\\[0.3em]
						\sslibrarysubroutine{getsalt}{}{
							return $s$
						}{1}\\[0.3em]
						\sslibrarysubroutine{test}{x, x' \in \bits^*}{
							if $x \neq x'$ and $H(s, x) = H(s, x')$:\\
							\quad return true\\
							return false
						}{1}
					}{0.68}
				}{\interchangeable{}}{
					\sslibrary{H}{cr-fake}{
						$s \twoheadleftarrow \bits^\lambda$\\[0.3em]
						\sslibrarysubroutine{getsalt}{}{
							return $s$
						}{1}\\[0.3em]
						\sslibrarysubroutine{test}{x, x' \in \bits^*}{
							return false
						}{1}
					}{0.68}
				}
			\end{center}
		\end{column}
		\begin{column}{0.48\textwidth}
			\begin{itemize}
				\item $H\colon \bits^\lambda \times \bits^* \rightarrow \bits^n$ is \textbf{collision-resistant} if nobody can make \textsc{test} return true.
				\item The public salt $s$ is the ``surprise'': without it, a collision \emph{exists} and could be hard-coded into the adversary.
				\item SHA-256 is unsalted: think $H(s, \cdot)$ for a fixed $s$. ``Collision-resistant'' = nobody knows a collision (Rogaway: ``human ignorance'').
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Birthday paradox}
	\begin{columns}[c]
		\begin{column}{1\textwidth}
			\begin{itemize}
				\item In a classroom of 50 students, what's the probability that at least two share a birthday?
				      \begin{itemize}
					      \item 2\%?
					      \item 20\%?
					      \item 50\%?
					      \item 97\%?
				      \end{itemize}
				\item Many people guess around 15--20\%, but the actual probability is about 97\%!
				\item This is counterintuitive because \textbf{we're not looking for a specific birthday -- we're looking for \textit{any} match among all possible pairs}.
				\item With 50 students, we have $\binom{50}{2} = 1{,}225$ possible pairs to check!\footnote{$\binom{50}{2}$ is a binomial coefficient. It means: \textit{``how many ways can I choose two different items from a set of 50?''}}
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Birthday paradox}
	\begin{itemize}
		\item With just 23 people, the probability exceeds 50\%
		\item Formula for $n$ people:
		      \begin{align*}
			      P = 1 - \frac{365!}{(365-n)! \cdot 365^n}
		      \end{align*}
		\item Implication: Finding a collision in a space of size $N$ takes roughly $\sqrt{N}$ samples.
		\item \textbf{This is why many cryptographic systems need large output spaces!}
	\end{itemize}
\end{frame}

\begin{frame}{Birthday probabilities}
	\begin{columns}[c]
		\begin{column}{0.55\textwidth}
			\begin{itemize}
				\item Many cryptographic algorithms fail if two executions sample the same random value.
				\item General question: take $q$ independent uniform samples from a set of $N$ items. What's the probability that some value is chosen more than once?
				\item This probability is called $\func{birthday}{q,N}$.
			\end{itemize}
		\end{column}
		\begin{column}{0.45\textwidth}
			\begin{align*}
				\func{birthday}{q,N} = 1 - \prod_{i=1}^{q-1} \left(1 - \frac{i}{N}\right)
			\end{align*}
			\begin{itemize}
				\item Surprisingly high! At $q \approx 1.2\sqrt{N}$ it is already $\approx 0.5$.
				\item Handy upper bound: $\func{birthday}{q,N} \leq q(q-1)/2N \leq q^2/2N$.
				\item For $N = 2^{128}$: a repeat is likely after only $\approx 2^{64}$ samples.
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Quick check: how big must $n$ be?}
	\begin{columns}[c]
		\begin{column}{0.6\textwidth}
			\begin{enumerate}
				\item \textbf{MD5 has a 128-bit output. Birthday attack cost?} \pause
				      $2^{64} \approx 1.8 \times 10^{19}$ hashes. One consumer GPU does $1.6 \times 10^{11}$ MD5/s: about 3.5 years. A thousand GPUs: about a day.
				\item \textbf{And SHA-256?} \pause
				      $2^{128} \approx 3.4 \times 10^{38}$. At $2 \times 10^{10}$ hashes/s: around $10^{20}$ years.
				\item \textbf{So MD5 is ``merely'' expensive to break?} \pause
				      No: the attacks in the next section find MD5 collisions in \emph{seconds}. The birthday bound is what a \emph{perfect} 128-bit function would give you.
			\end{enumerate}
		\end{column}
		\begin{column}{0.4\textwidth}
			\alertbox{Rule of thumb}{An $n$-bit hash offers at most $n/2$ bits of collision resistance (and $n$ bits of preimage resistance). This is why 256-bit outputs are the modern floor.}
		\end{column}
	\end{columns}
\end{frame}

\section{Attacks on Hash Functions}

\begin{frame}{Example hash functions}
	\vspace{-0.4cm}
	\begin{columns}[c]
		\begin{column}{1\textwidth}
			\begin{itemize}[<+->]
				\item \textbf{Historical/broken hash functions:}
				      \begin{itemize}
					      \item \textbf{MD4} (1990): 128-bit, completely broken, collisions found instantly
					      \item \textbf{MD5} (1992): 128-bit, collisions found in 2004, still widely misused
					      \item \textbf{SHA-0} (1993): 160-bit, withdrawn shortly after release
					      \item \textbf{SHA-1} (1995): 160-bit, practical collisions demonstrated in 2017
				      \end{itemize}
				\item \textbf{Current standard hash functions:}
				      \begin{itemize}
					      \item \textbf{SHA-2 family} (2001): SHA-224, SHA-256, SHA-384, SHA-512
					      \item \textbf{SHA-3 family} (2015): Based on the Keccak sponge construction
				      \end{itemize}
				\item \textbf{Modern alternatives:}\footnote{\url{https://www.blake2.net}, \url{https://github.com/BLAKE3-team/BLAKE3}, \url{https://datatracker.ietf.org/doc/rfc9861/}}
				      \begin{itemize}
					      \item \textbf{BLAKE2} (2012): Faster than MD5; used inside Argon2 and WireGuard
					      \item \textbf{BLAKE3} (2020): Highly parallelizable, extremely fast
					      \item \textbf{KangarooTwelve} (2016), \textbf{TurboSHAKE} (2023): Reduced-round Keccak, RFC 9861 (2025)
				      \end{itemize}
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Evaluating a hash function's quality}
	\begin{columns}[c]
		\begin{column}{0.6\textwidth}
			\begin{itemize}
				\item \textbf{At a high level}:
				      \begin{itemize}[<+->]
					      \item \textbf{Asymmetric primitives} are based on mathematical
					            problems, can be mathematically proven secure (given assumptions!)
					      \item \textbf{Symmetric primitives} (encryption, hashing\ldots)
					            are statistically, empirically, heuristically shown to be secure,
					            not proven secure.
					      \item The more cryptanalysis they survive, the higher confidence
					            we have in their security.
				      \end{itemize}
			\end{itemize}
		\end{column}

		\begin{column}{0.4\textwidth}
			\imagewithcaption{qiao.png}{Cryptanalysis of AES.}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{A more nuanced view on provable security}
	\begin{columns}[c]
		\begin{column}{1\textwidth}
			\begin{itemize}[<+->]
				\item Reality is more nuanced:
				      \begin{itemize}
					      \item \textbf{Asymmetric primitives}: While based on mathematical problems, these are \textbf{conjectures}, not proofs.
					            \begin{itemize}
						            \item No proven lower bounds on factoring complexity.
						            \item Many newer problems (lattices, code-based) receive less scrutiny.
						            \item Real implementations often have exotic side assumptions.
					            \end{itemize}
					      \item \textbf{Symmetric primitives}: Some properties \textit{can} be formally proven!
					            \begin{itemize}
						            \item Example: No 4-round AES trail activates fewer than 25 bytes.
						            \item This excludes entire classes of attacks.
						            \item Modes of operation (encryption, hashing) often have security proofs.
					            \end{itemize}
				      \end{itemize}
				\item In both cases, security rests on assumptions -- the difference is in the \textit{type} of assumptions, not their existence.
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{What about attacks on hash functions?}
	\begin{itemize}
		\item The birthday bound is the \emph{generic} attack. Everything else in this section exploits how hash functions are actually \textbf{used} or \textbf{built}:
		      \begin{itemize}
			      \item \textbf{Precomputation}: hashing predictable inputs (passwords) lets attackers do the work once, in advance. Countermeasure: \textbf{salting}.
			      \item \textbf{Length extension}: Merkle--Damg\aa rd hashes leak their internal state. Countermeasures: the sponge construction, \textbf{HMAC}.
			      \item \textbf{Actual collisions}: MD5 (2004), rogue certificate authorities (2008), SHA-1 (2017).
		      \end{itemize}
		\item Then, in the next section, we turn these lessons into a recipe for storing passwords.
	\end{itemize}
\end{frame}

\begin{frame}{Precomputation attacks}{Why salts matter}
	\begin{itemize}
		\item Consider a server storing hashes of users' passwords.
		\item With \textbf{unsalted} hashes, an attacker can:
		      \begin{itemize}
			      \item Pre-compute hashes of common passwords.
			      \item Create a dictionary mapping $H(\texttt{password}) \rightarrow \texttt{password}$.
			      \item Instantly crack common passwords after a breach.
		      \end{itemize}
		\item All hard work is done \textbf{before} the attack.
		\item Results can be \textbf{reused} against any system using the same hash function.
	\end{itemize}
\end{frame}

\begin{frame}{Rainbow tables}
	\definitionbox{Rainbow Tables}{
		\textbf{Rainbow tables} are precomputed data structures that let an attacker invert an unsalted hash of any password from a chosen space (say, all 8-character alphanumeric strings) with a lookup and a few hash evaluations, instead of a brute-force search.
	}
	\begin{itemize}
		\item A \textbf{time--memory tradeoff} (Hellman, 1980; refined by Oechslin, 2003): store \emph{chains} of hashes rather than every hash--password pair, so a table covering billions of passwords fits on a disk.
		\item Cover essentially every password within the table's character set and length; miss everything outside it.
		\item Completely defeated by properly salted hashes, as we'll see next.
	\end{itemize}
\end{frame}

\begin{frame}{Salt in hash functions}{Preventing precomputation attacks}
	\definitionbox{Hash Salt}{
		A \textbf{salt} is a random value that is concatenated with the input before hashing:
		$H(S \| X)$ where $S$ is the salt and $X$ is the original input.
	}
	\begin{itemize}
		\item The salt doesn't need to be secret, but should be:
		      \begin{itemize}
			      \item Random and unique for each input.
			      \item Long enough to prevent precomputation ($\geq 16$ bytes).
			      \item Stored alongside the hash value.
		      \end{itemize}
		\item Without a salt, equal inputs give equal hashes, so one precomputed table works against \emph{every} system using that hash function. With a salt, the table is useless.
	\end{itemize}
\end{frame}

\begin{frame}{How salt neutralizes precomputation}
	\begin{itemize}
		\item With salted hashes, precomputation becomes useless:
		      \begin{itemize}
			      \item Attacker doesn't know the salts until they've already breached the system.
			      \item Can't precompute $H(S \| X)$ without knowing $S$.
		      \end{itemize}
		\item Using \textbf{different} salt for each user:
		      \begin{itemize}
			      \item Forces attacker to choose which user to attack.
			      \item Attacking two victims requires twice the effort.
			      \item No more ``attack all users for the price of one''.
		      \end{itemize}
		\item Even if attacker obtains both hash and salt, they still need to brute-force each password individually.
	\end{itemize}
\end{frame}

\begin{frame}{Length extension attacks}{Why $H(\texttt{secret} \| \texttt{message})$ is not a MAC}
	\definitionbox{Length Extension Attack}{
		A \textbf{length extension attack} allows an attacker who knows $H(M)$ and the \emph{length} of $M$ to calculate $H(M \| P \| X)$ for the padding $P$ and any attacker-chosen $X$, without knowing $M$ itself.
	}
	\begin{itemize}
		\item Affects hash functions built using the Merkle--Damg\aa rd construction:
		      \begin{itemize}
			      \item MD5, SHA-1, SHA-256, SHA-512, and others.
			      \item Not SHA-3 (sponge construction), and not truncated variants such as SHA-384 or SHA-512/256: their output hides part of the state.
		      \end{itemize}
		\item This breaks naive authentication schemes that use $H(\texttt{secret} \| \texttt{message})$ as a tag: the attacker extends \texttt{message} and gets a valid tag for the result, for free.
	\end{itemize}
\end{frame}

\begin{frame}{Length extension attacks}{Real-world implications}
	\begin{itemize}
		\item Consider a web API using:
		      \begin{itemize}
			      \item URL: \texttt{api.com/data?val=foo\&hash=123abc}
			      \item Hash calculated as: $H(\texttt{secret} \| \texttt{val=foo})$
			      \item Server verifies request by checking if hash matches
		      \end{itemize}
		\item An attacker can:
		      \begin{itemize}
			      \item Take a valid \texttt{hash} value (and guess the length of the secret: a few tries).
			      \item Compute a valid hash for \texttt{val=foo} $\|$ \texttt{padding} $\|$ \texttt{\&malicious=true}. The padding bytes end up in the URL, but sloppy parsers don't mind.
			      \item Without knowing the secret!
		      \end{itemize}
		\item This is exactly how Thai Duong and Juliano Rizzo forged API calls against Flickr in 2009. The fix, then and now: use \textbf{HMAC}.
	\end{itemize}
\end{frame}

\begin{frame}{The Merkle--Damg\aa rd construction}{SHA-1, SHA-2}
	\begin{columns}[c]
		\begin{column}{0.5\textwidth}
			\begin{itemize}
				\item Key components:
				      \begin{itemize}
					      \item The message is padded to a multiple of the block size.
					      \item It is then split into fixed-size blocks.
					      \item Blocks are processed sequentially through a compression function.
					      \item Each step combines the previous state with the current block.\footnote{Interactive visualization: \url{https://sha256algorithm.com}}
				      \end{itemize}
				\item If the compression function is collision-resistant, so is the hash function.
			\end{itemize}
		\end{column}
		\begin{column}{0.5\textwidth}
			\imagewithcaption{merkle-damgard.pdf}{Source: The Joy of Cryptography}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{The sponge construction}{SHA-3}
	\begin{columns}[c]
		\begin{column}{0.55\textwidth}
			\begin{itemize}
				\item Two phases:
				      \begin{itemize}
					      \item \textbf{Absorbing}: Input blocks are XORed into the state, which is then scrambled by the Keccak permutation.
					      \item \textbf{Squeezing}: Output blocks are extracted from the state.
				      \end{itemize}
				\item Naturally resistant to length extension attacks.
				\item Versatile: can squeeze output of any desired length (the SHAKE128 and SHAKE256 ``extendable-output functions'').
			\end{itemize}
		\end{column}
		\begin{column}{0.45\textwidth}
			\imagewithcaption{sponge.png}{Source: \emph{The Hash Function BLAKE} (Aumasson et al.)}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{How the sponge construction prevents length extension}
	\begin{columns}[c]
		\begin{column}{0.55\textwidth}
			\begin{itemize}
				\item In Merkle--Damg\aa rd, the final state \emph{is} the hash output.
				\item So whoever holds the output holds the state, and can keep hashing without knowing the input.
				\item The sponge construction protects against this by:
				      \begin{itemize}
					      \item Maintaining a larger internal state than the output size.
					      \item Only exposing part of the state (the \emph{rate}) to input and output.
					      \item Never revealing the rest (the \emph{capacity}).
				      \end{itemize}
			\end{itemize}
		\end{column}
		\begin{column}{0.45\textwidth}
			\begin{tcolorbox}[colback=black!5!white,colframe=ciphergray]
				\textbf{Sponge state (SHA3-256, 1600 bits):}

				\vspace{0.2cm}

				\texttt{Rate (r): 1088 bits}\\
				\textcolor{cipheraccent}{\texttt{[Exposed to input/output]}}

				\vspace{0.2cm}

				\texttt{Capacity (c): 512 bits}\\
				\textcolor{cipherprimary}{\texttt{[Hidden from attacker]}}

				\vspace{0.3cm}

				\textcolor{cipherprimary}{\textbf{Key:} \small Without knowing the full state, an attacker cannot continue hashing!}
			\end{tcolorbox}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{HMAC: protection against length extension}
	\definitionbox{HMAC}{
		\textbf{HMAC} (Hash-based Message Authentication Code) is a specific construction for calculating a message authentication code (MAC) using a cryptographic hash function and a secret key.
	}
	\begin{align*}
		\textsf{HMAC}(K, m) = H((K \oplus \mathsf{opad}) \| H((K \oplus \mathsf{ipad}) \| m))
	\end{align*}
	\begin{itemize}
		\item $\mathsf{opad}$, $\mathsf{ipad}$: fixed constants; $K$ is padded to the block size.
		\item The outer hash hides the inner state: length extension gains nothing.
		\item Bellare, Canetti and Krawczyk (1996); RFC 2104. In TLS, IPsec, everywhere.
		\item A MAC (remember 1.5): integrity and authentication, not confidentiality.
	\end{itemize}
\end{frame}

\begin{frame}{Xiaoyun Wang's MD5 collision}{CRYPTO 2004 rump session}
	\begin{itemize}
		\item In 2004, Xiaoyun Wang and her team at Shandong University demonstrated the first practical MD5 collision, twelve years after MD5 was published.
		\item Announced at the CRYPTO 2004 rump session, causing a sensation: MD5 was everywhere, from certificates to file integrity checks.
		\item Interesting backstory: the team's early attempts failed because of an \textbf{endianness} mix-up. The Chinese translation of Bruce Schneier's \emph{Applied Cryptography} had the byte ordering of MD5 wrong.
		\item Once corrected, the attack worked: about an hour on an IBM p690 in 2005. Today, a laptop finds an MD5 collision in seconds.
		\item This discovery accelerated the migration away from MD5 in security applications. ``Accelerated'' is generous, as we'll see.
	\end{itemize}
\end{frame}

\begin{frame}{Two kinds of collision}{Identical-prefix vs.\ chosen-prefix}
	\definitionbox{Identical-Prefix vs.\ Chosen-Prefix Collisions}{
		\textbf{Identical-prefix}: find $M \neq M'$ with $H(M) = H(M')$, where the attacker controls both messages entirely. \textbf{Chosen-prefix}: given \emph{any} two prefixes $P \neq P'$, find suffixes $S, S'$ with $H(P \| S) = H(P' \| S')$.
	}
	\begin{itemize}
		\item Wang's 2004 attack is identical-prefix: the two messages differ only in a few blocks of random-looking bytes, so the ``documents'' must hide the difference in a comment field or padding.
		\item A chosen-prefix collision lets the two documents say completely different things: for instance, two certificates with different names and different keys.
		\item Costlier to find, but far more dangerous.
	\end{itemize}
\end{frame}

\begin{frame}{MD5 chosen-prefix collisions in the wild}{Attacking certificate authorities}
	\begin{itemize}
		\item In 2008, Sotirov, Stevens, Appelbaum, Lenstra, Molnar, Osvik and de Weger forged a \textbf{rogue CA certificate}: $P$ = an ordinary website certificate that RapidSSL would sign, $P'$ = a CA certificate. One signature, valid for both.
		\item The catch: the CA fills in the serial number and validity period, so $P$ had to be predicted. RapidSSL used sequential serials and a fixed delay. The collision itself took a day or two on 200 PlayStation 3s.
		\item Demonstrated at the 25th Chaos Communication Congress (December 2008). MD5 was finally dropped for certificate signatures.
		\item Encore, 2012: the \textbf{Flame} malware forged a Microsoft code-signing certificate with a new, independently developed chosen-prefix MD5 collision.
	\end{itemize}
\end{frame}

\begin{frame}{SHAttered: the first practical SHA-1 collision}{2017}
	\begin{itemize}
		\item In 2017, Google and CWI Amsterdam researchers (Stevens, Bursztein, Karpman, Albertini, Markov) demonstrated the first practical collision for SHA-1.\footnote{\url{https://appliedcryptography.page/paper/\#shattered-sha1}}
		\item Two different PDF files with identical SHA-1 hashes, displaying different content.
		\item Cost: $2^{63.1}$ SHA-1 computations (6,500 CPU-years plus 110 GPU-years, about \$110,000 of cloud time), versus $2^{80}$ for the birthday bound.
		\item 2020, ``SHA-1 is a Shambles'' (Leurent and Peyrin): the first \textbf{chosen-prefix} SHA-1 collision, for about \$45,000 of rented GPU time, used to forge PGP identity certificates.
		\item SHA-1 should have died in 2005, when Wang's team published the first theoretical break. Browsers only stopped accepting SHA-1 certificates in early 2017; Git still uses it by default.
	\end{itemize}
\end{frame}

	\begin{frame}{Multi-collision attacks}
		\definitionbox{Multi-Collision}{
			A \textbf{multi-collision} is a set of $k$ distinct inputs that all hash to the same output value.
		}
		\begin{itemize}
			\item Joux's 2004 multi-collision attack:
			      \begin{itemize}
				      \item Finding $2^n$ colliding messages requires only about $n$ times the work of finding a single collision.
				      \item For a 128-bit hash, finding $2^{64}$ collisions takes only about 64 times more work than finding one collision.
				      \item Much more efficient than expected from the birthday paradox!
			      \end{itemize}

			      \begin{itemize}
				      \item Concatenating two hash functions ($H_1 \| H_2$) provides much less security than expected.
				      \item Influences modern hash function design.

	\end{frame}

\section{Password Hashing}

\begin{frame}{Password-based authentication}{Common use of hash functions}
	\begin{itemize}
		\item Passwords are the most common authentication mechanism.
		\item Server must verify if a user's password is correct.
		\item Different methods for password storage:
		      \begin{itemize}
			      \item<1-5> Storing in the clear (extremely bad)
			      \item<2-5> Encryption (also bad)
			      \item<3-5> Unsalted hashing (better, but problematic)
			      \item<4-5> Salted hashing (good practice)
			      \item<5-5> Specialized password hashing (best practice)
		      \end{itemize}
		\item We compare methods by asking: what happens if the server is compromised?
	\end{itemize}
\end{frame}

\begin{frame}{Storing passwords in the clear}{The worst approach}
	\begin{columns}[c]
		\begin{column}{0.55\textwidth}
			\begin{itemize}
				\item The server simply stores exact passwords.
				\item Password verification is trivial:
				      \begin{itemize}
					      \item User enters password $P'$
					      \item Server checks if $P' \stackrel{?}{=} P$
				      \end{itemize}
				\item If the server is compromised, the attacker immediately learns all passwords.
				\item Users often reuse passwords across sites, compounding the damage.
			\end{itemize}
		\end{column}
		\begin{column}{0.45\textwidth}
			\begin{tcolorbox}[colback=black!5!white,colframe=ciphergray]
				\textbf{Database Storage:}

				\vspace{0.2cm}

				\texttt{username: alice}\\
				\texttt{password: secret123}

				\vspace{0.2cm}

				\texttt{username: bob}\\
				\texttt{password: password1}

				\vspace{0.3cm}

				\textcolor{cipherprimary}{\textbf{Problem:} \small Attacker who accesses the database instantly has everyone's actual passwords!}
			\end{tcolorbox}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Encrypting passwords}{A common mistake}
	\begin{columns}[c]
		\begin{column}{0.55\textwidth}
			\begin{itemize}
				\item Some systems try to encrypt passwords:
				      \begin{itemize}
					      \item Store $\texttt{Enc}(K, P)$ instead of $P$.
					      \item Decrypt when verifying login.
				      \end{itemize}
				\item Problem: The decryption key $K$ must be accessible to the server.
				\item If an attacker compromises the server, they get both:
				      \begin{itemize}
					      \item The encrypted passwords.
					      \item The decryption key.
				      \end{itemize}
				\item Result: encryption adds (almost) nothing.
			\end{itemize}
		\end{column}
		\begin{column}{0.45\textwidth}
			\begin{tcolorbox}[colback=black!5!white,colframe=ciphergray]
				\textbf{Encrypted Password Storage:}

				\vspace{0.2cm}

				\texttt{username: alice}\\
				\texttt{password: A7F3D92C...}

				\vspace{0.2cm}

				\texttt{decryption\_key: 5B8E1...}

				\vspace{0.3cm}

				\textcolor{cipherprimary}{\textbf{Problem:} \small Server needs the key to verify passwords, so attacker gets the key too!}
			\end{tcolorbox}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Unsalted password hashing}{A basic approach}
	\begin{columns}[c]
		\begin{column}{0.55\textwidth}
			\begin{itemize}
				\item For a user with password $P$, the server stores $h = H(P)$.
				\item On login, server checks if $H(P') \stackrel{?}{=} h$.
				\item Advantages over cleartext:
				      \begin{itemize}
					      \item The attacker doesn't immediately get passwords.
					      \item Must spend effort guessing and hashing.
				      \end{itemize}
				\item Problems:
				      \begin{itemize}
					      \item Vulnerable to precomputation (rainbow tables).
					      \item Same password = same hash for all users.
				      \end{itemize}
			\end{itemize}
		\end{column}
		\begin{column}{0.45\textwidth}
			\begin{tcolorbox}[colback=black!5!white,colframe=ciphergray]
				\textbf{Unsalted Hash Storage:}

				\vspace{0.2cm}

				\texttt{username: alice}\\
				\texttt{hash: 5f4dcc3b5aa...}

				\vspace{0.2cm}

				\texttt{username: bob}\\
				\texttt{hash: 5f4dcc3b5aa...}

				\vspace{0.3cm}

				\textcolor{cipherprimary}{\textbf{Problem:} \small Alice and Bob visibly share a password, and anyone can Google that hash: it's $\mathsf{MD5}(\texttt{password})$.}
			\end{tcolorbox}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Salted password hashing}{The proper approach}
	\begin{columns}[c]
		\begin{column}{0.5\textwidth}
			\definitionbox{Salted Hash Authentication}{
				For a user with password $P$, the server stores:
				\begin{itemize}
					\item A random salt $S$ (unique per user)
					\item The hash value $h = H(S \| P)$
				\end{itemize}
				To verify a login with password $P'$, the server checks if $H(S \| P') \stackrel{?}{=} h$.
			}
		\end{column}
		\begin{column}{0.5\textwidth}
			\begin{tcolorbox}[colback=black!5!white,colframe=ciphergray]
				\small
				\textbf{Salted Hash Storage:}

				\vspace{0.2cm}

				\texttt{username: alice}\\
				\texttt{salt: 8a7b3c...}\\
				\texttt{hash: e8f74d...}

				\vspace{0.2cm}

				\texttt{username: bob}\\
				\texttt{salt: 2c9d5e...}\\
				\texttt{hash: 9f82e1...}

				\vspace{0.3cm}

				\textcolor{cipherprimary}{\textbf{Note:} \scriptsize Even if Alice and Bob have the same password, they'll have different hashes!}
			\end{tcolorbox}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Specialized password hashing functions}{Beyond regular hash functions}
	\begin{itemize}[<+->]
		\item Regular cryptographic hash functions (SHA-2, SHA-3) are designed to be \textbf{fast}.
		\item But for password hashing, slower is better!
		      \begin{itemize}
			      \item Makes attacks more expensive.
			      \item Users don't notice milliseconds of delay.
			      \item Attackers trying billions of guesses definitely notice.
		      \end{itemize}
		\item Simple approach: Iterate the hash function thousands of times.
		      \begin{itemize}
			      \item $H(S \| H(S \| \ldots H(S \| P)\ldots))$ with many iterations.
			      \item \textbf{Example}: PBKDF2 (1999). Still standardized and FIPS-approved, but no longer the first choice.
		      \end{itemize}
		\item \textbf{Modern approach}: Memory-hard functions like Argon2 and Scrypt.
	\end{itemize}
\end{frame}

\begin{frame}{PBKDF2: Password-Based Key Derivation Function 2}
	\begin{columns}[c]
		\begin{column}{0.6\textwidth}
			\definitionbox{PBKDF2}{
				\textbf{PBKDF2} (Password-Based Key Derivation Function 2) is a key derivation function that applies a pseudorandom function (like HMAC) to the input password along with a salt to produce a derived key.
			}
			\begin{itemize}
				\item Key components:
				      \begin{itemize}
					      \item \textbf{Password} $P$: Secret input.
					      \item \textbf{Salt} $S$: Provides randomization.
					      \item \textbf{Iterations} $c$: Cost knob (OWASP: 600,000).
					      \item \textbf{PRF}: Usually HMAC-SHA1 or HMAC-SHA256.
				      \end{itemize}
			\end{itemize}
		\end{column}
		\begin{column}{0.4\textwidth}
			\begin{tcolorbox}[colback=black!5!white,colframe=ciphergray]
				\small
				\begin{align*}
					U_1 & = \mathsf{PRF}(P, S \| \mathsf{Int}(1))    \\
					U_2 & = \mathsf{PRF}(P, U_1)                     \\
					U_3 & = \mathsf{PRF}(P, U_2)                     \\
					\vdots                                           \\
					U_c & = \mathsf{PRF}(P, U_{c-1})                 \\
					T_1 & = U_1 \oplus U_2 \oplus \cdots \oplus U_c
				\end{align*}
			\end{tcolorbox}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Why not PBKDF2?}{Vulnerable to specialized hardware}
	\begin{columns}[c]
		\begin{column}{0.6\textwidth}
			\begin{itemize}
				\item \textbf{Problem}: PBKDF2 is just a lot of SHA-256, and attackers build hardware for that.
				      \begin{itemize}
					      \item GPUs and ASICs hash orders of magnitude faster than the defender's server CPU.
					      \item More iterations slow down both sides equally.
				      \end{itemize}
				\item \textbf{Solution}: memory-hard functions
				      \begin{itemize}
					      \item Each guess needs a large amount of fast memory.
					      \item Memory is the one resource an ASIC can't pack in cheaply.
					      \item Attackers and defenders end up buying the same RAM.
				      \end{itemize}
			\end{itemize}
		\end{column}
		\begin{column}{0.4\textwidth}
			\imagewithcaption{gpu-farm.jpg}{A GPU farm, likely built to mine Bitcoin, but which can also be used to do a lot of hashing really fast (Bitcoin mining is just calculating a lot of SHA-256 hashes!)}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{What Bitcoin mining teaches us about fast hashes}
	\begin{itemize}
		\item Bitcoin mining is the Anubis puzzle at planetary scale: find a nonce with $\mathsf{SHA256}(\mathsf{SHA256}(\text{header}))$ below a target. Pure SHA-256, about $10^{21}$ times per second.
		\item Hardware followed the money: CPUs $\rightarrow$ GPUs $\rightarrow$ FPGAs $\rightarrow$ ASICs in four years (2009--2013), chips that do only SHA-256, orders of magnitude more efficient than a CPU.
		\item It burns well over 100 TWh a year (Cambridge, 2025: 140--190 TWh).
		\item \textbf{Lesson}: PBKDF2-HMAC-SHA256 is just more SHA-256, so the attacker's edge is the miner's. Remove it by making each guess need what ASICs lack: \textbf{memory}.
		\item Litecoin (2011) picked Scrypt against ASICs, but with only 128 KB per hash: Scrypt ASICs arrived in 2014. Memory-hardness needs actual memory.
	\end{itemize}
\end{frame}

\begin{frame}{Scrypt: a memory-hard password hashing function}
	\definitionbox{Scrypt}{
		\textbf{Scrypt} is a password-based key derivation function designed so that each evaluation needs a large amount of memory, not just CPU time, making hardware acceleration expensive.
	}
	\begin{itemize}
		\item Created by Colin Percival in 2009 (for his Tarsnap backup service) specifically to make brute force costly.
		\item Key components:
		      \begin{itemize}
			      \item \textbf{PBKDF2-HMAC-SHA256}: Initial and final mixing of password and salt.
			      \item \textbf{ROMix}: Core memory-hard function.
			      \item \textbf{BlockMix}: Based on the Salsa20/8 core function.
		      \end{itemize}
		\item \textbf{Parameters}: \texttt{N} (CPU/memory cost), \texttt{r} (block size), \texttt{p} (parallelization). Memory used: $128 \cdot N \cdot r$ bytes, e.g.\ 16 MiB for $N = 2^{14}, r = 8$. OWASP recommends $N = 2^{17}$ (128 MiB).
	\end{itemize}
\end{frame}

\begin{frame}{Why Scrypt is memory-hard}
	\begin{itemize}
		\item ROMix, on $N$ blocks of $128 r$ bytes:
		      \begin{itemize}
			      \item \textbf{Fill}: $V_i = \mathsf{BlockMix}^{i}(X)$ for $i = 0, \ldots, N-1$.
			      \item \textbf{Mix}: repeat $N$ times: $j \leftarrow \mathsf{Integerify}(X) \bmod N$, then $X \leftarrow \mathsf{BlockMix}(X \oplus V_j)$.
		      \end{itemize}
		\item The index $j$ depends on the data, so you can't predict which block you'll need next: either keep all $N$ blocks in memory, or recompute them on demand.
		\item That tradeoff is what hurts attackers: cut the memory by a factor $k$ and the time goes up by about a factor $k$. Memory $\times$ time stays large, and that product is what an ASIC has to pay for. RAM costs the same for everyone.
		\item Proven \textbf{maximally memory-hard} by Alwen, Chen, Pietrzak, Reyzin and Tessaro (2017).\footnote{\url{https://appliedcryptography.page/paper/\#scrypt-memory}}
		\item Weak spot: with small $N$ (Litecoin's 128 KB) the guarantee is worth little. Pick $N$ large.
	\end{itemize}
\end{frame}

\begin{frame}{Password cracking speeds}{The importance of slow hashing}
	\begin{columns}[c]
		\begin{column}{0.55\textwidth}
			\begin{itemize}
				\item One RTX 4090 GPU (Hashcat benchmark):\footnote{\url{https://gist.github.com/Chick3nman/32e662a5bb63bc4f51b847bb422222fd}}
				      \begin{itemize}
					      \item MD5: 164 billion hashes/second
					      \item SHA-1: 51 billion hashes/second
					      \item SHA-256: 22 billion hashes/second
					      \item bcrypt (cost 12): $\approx$ 1,400/second (cost 5: 184,000)
					      \item Scrypt ($N = 2^{14}$, 16 MiB): $\approx$ 7,100/second
				      \end{itemize}
				\item $10^{11}$ vs.\ $10^{3}$ per second: each guess becomes $10^8$ times more expensive.
			\end{itemize}
		\end{column}
		\begin{column}{0.45\textwidth}
			\alertbox{Worked example}{A leaked list of $10^8$ common passwords, tried against one account:\\[0.3em]
				SHA-256: under a second.\\
				Scrypt (16 MiB): about four hours.\\
				bcrypt (cost 12): about a day.\\[0.3em]
				And thanks to the salt, this is \emph{per account}.}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Insights from Hashcat 7.0.0}{August 2025: reading the attackers' changelog}
	\definitionbox{Hashcat}{
		\textbf{Hashcat} is the de-facto standard open-source password cracker: hundreds of hash types, GPU-accelerated, with dictionary, rule-based, mask and hybrid attack modes.
	}
	\begin{itemize}
		\item Version 7.0.0 (1 August 2025) landed after two years of development, 900,000+ changed lines of code and 105 contributors.\footnote{\url{https://hashcat.net/forum/thread-13330.html}}
		\item It now runs on anything: NVIDIA (CUDA), AMD (HIP), Apple Silicon (Metal) and, through the new ``Assimilation Bridge'', CPUs, FPGAs and even Python plugins. Assume your attacker uses every chip they own.
		\item Two trends in the release notes matter for this lecture.
	\end{itemize}
\end{frame}

\begin{frame}{Trend 1: memory-hard functions are winning}{But attackers are adapting}
	\begin{columns}[c]
		\begin{column}{0.55\textwidth}
			\begin{itemize}
				\item \textbf{Argon2 support added}: defenders adopt memory-hard functions, attackers follow.
				\item \textbf{Scrypt up to $+320\%$ faster}: nobody is giving up on memory-hard targets.
				\item \textbf{4 GB allocation cap removed}: cracking rigs can now use all of their GPU memory.
				\item Memory-hardness raises the price per guess; it doesn't make guessing impossible.
			\end{itemize}
		\end{column}
		\begin{column}{0.45\textwidth}
			\begin{tcolorbox}[colback=black!5!white,colframe=ciphergray]
				\textbf{Performance gains (7.0.0 release notes):}\\
				\texttt{Scrypt: up to +320\%}\\
				\texttt{RAR3: up to +54\%}\\
				\texttt{NetNTLMv2: +223\% (Intel CPUs)}\\[0.5em]
				\textcolor{cipherprimary}{\textbf{Note:} \small Even memory-hard functions face continuous optimization pressure!}
			\end{tcolorbox}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Trend 2: real-world targets dominate}
	\begin{columns}[c]
		\begin{column}{0.55\textwidth}
			\begin{itemize}
				\item \textbf{58 new application-specific hash types}:
				      \begin{itemize}
					      \item MetaMask (cryptocurrency wallets)
					      \item Microsoft Online Accounts
					      \item LUKS2 (disk encryption)
					      \item GPG, OpenSSH (security tools)
				      \end{itemize}
				\item \textbf{20 new extraction tools} for BitLocker, APFS, VirtualBox, wallet files.
				\item \textbf{Hash-mode autodetection}: paste a hash, Hashcat guesses the algorithm.
				\item Password hashing is not just about web logins: wherever a password protects a key, the same cracking pipeline applies.
			\end{itemize}
		\end{column}
		\begin{column}{0.45\textwidth}
			\begin{tcolorbox}[colback=black!5!white,colframe=ciphergray]
				\textbf{Attack Evolution:}
				\begin{itemize}
					\item Past: Break MD5/SHA-1
					\item Today: Extract from:
					      \begin{itemize}
						      \small
						      \item Encrypted disks
						      \item Password managers
						      \item Cryptocurrency wallets
						      \item Cloud accounts
					      \end{itemize}
				\end{itemize}
				\textcolor{cipherprimary}{\textbf{Lesson:} \small Security depends on implementation, not just algorithm choice!}
			\end{tcolorbox}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Best practices for password hashing}{OWASP Password Storage Cheat Sheet, 2025}
	\begin{itemize}
		\item Use a dedicated password hashing function, never raw SHA-256. In order of preference (and tune upward as your login latency budget allows):\footnote{\url{https://cheatsheetseries.owasp.org/cheatsheets/Password_Storage_Cheat_Sheet.html}}
		      \begin{itemize}
			      \item \textbf{Argon2id}: winner of the Password Hashing Competition (2015), RFC 9106. Minimum: 19 MiB of memory, 2 iterations, parallelism 1.
			      \item \textbf{Scrypt} if Argon2 is unavailable: $N = 2^{17}$ (128 MiB), $r = 8$, $p = 1$.
			      \item \textbf{bcrypt} for legacy systems: work factor $\geq 10$ (and mind its 72-byte input limit).
			      \item \textbf{PBKDF2} only where FIPS-140 compliance forces it: 600,000 iterations of HMAC-SHA256.
		      \end{itemize}
		\item Salts are handled for you: good libraries generate a random 16-byte salt per password and store it inside the output string.
		\item Optional extra: a \textbf{pepper}, a secret key kept outside the database (e.g.\ in an HSM), applied as $\mathsf{HMAC}(\mathsf{pepper}, \mathsf{hash})$. A database dump alone is then useless to the attacker.
	\end{itemize}
\end{frame}

\begin{frame}{``Migrate to stronger algorithms?''}
	\begin{itemize}
		\item Cryptographic algorithms have limited lifespans: new attacks, more computing power, and (for some of them) quantum computers.
		\item \textbf{Protocol agility} refers to a system's ability to:
		      \begin{itemize}
			      \item Support multiple cryptographic algorithms.
			      \item Negotiate which algorithm to use.
			      \item Transition smoothly as algorithms are deprecated.
		      \end{itemize}
		\item Challenges with protocol agility:
		      \begin{itemize}
			      \item Complexity leads to implementation errors.
			      \item Downgrade attacks target the weakest supported option.
			      \item Backward compatibility vs. security tradeoffs.
		      \end{itemize}
		\item Password hashes are a special case: the server can't rehash what it doesn't have. Either upgrade at the user's next login, or wrap the legacy hash right now, $\mathsf{bcrypt}(\mathsf{md5}(P))$, and replace it with a direct hash when the user next logs in.
		\item We'll talk a lot more about all of this in Part 2 of the course.
	\end{itemize}
\end{frame}

\section{The Random Oracle Model}

\begin{frame}{Why can't we just say ``$H$ is pseudorandom''?}
	\begin{itemize}
		\item A PRF $F_k$ looks random only to someone who doesn't know $k$. A hash function has \textbf{no key}: everyone, including the adversary, can compute $H$.
		\item So ``$H$ is indistinguishable from a random function'' is false for every fixed $H$: the distinguisher just computes $H(x)$ itself and compares.
		\item Yet many schemes (signatures, key derivation, Fiat--Shamir, RSA-OAEP) are only secure if $H$ behaves like ``random junk''. Collision resistance alone is not enough to prove them.
		\item The way out: \textbf{pretend}. Analyze the scheme as if $H$ were a truly random function that everyone can query, then instantiate it with SHA-256 and hope the analysis carries over.
		\item That ``pretend'' has a name: the \textbf{random oracle model}.
	\end{itemize}
\end{frame}

\begin{frame}{The Random Oracle Model}
	\begin{columns}[c]
		\begin{column}{0.55\textwidth}
			\definitionbox{Random Oracle Model}{
				The \textbf{Random Oracle Model} treats the hash function as a truly random function that every party can query.
			}
			\begin{itemize}
				\item A random oracle is a black box that:
				      \begin{itemize}
					      \item Answers new queries with fresh random outputs.
					      \item Repeats itself: same input, same output.
					      \item Is available to everyone, honest or not.
				      \end{itemize}
				\item Far easier to analyze than 64 rounds of SHA-256.
			\end{itemize}
		\end{column}
		\begin{column}{0.45\textwidth}
			\begin{center}
				\sslibrary{}{ro}{
					$T := $ empty associative array\\[0.3em]
					\sslibrarysubroutine{h}{x}{
						if $T[x]$ undefined:\\
						\quad $T[x] \twoheadleftarrow \bits^n$\\
						return $T[x]$
					}{1}
				}{0.9}
			\end{center}
			\small In library terms, the oracle is \emph{lazily sampled}: no output exists until somebody asks.
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Using the Random Oracle Model}
	\begin{columns}[c]
		\begin{column}{0.55\textwidth}
			\begin{itemize}
				\item Used to prove security where standard assumptions don't suffice. Examples: RSA-OAEP, Schnorr and Ed25519 signatures, Fiat--Shamir (see 2.8).
				\item Two ``superpowers'' the proof gets, which no real hash function offers:
				      \begin{itemize}
					      \item \textbf{Observability}: it sees every query the adversary makes.
					      \item \textbf{Programmability}: it may choose the answers, as long as they look uniform.
				      \end{itemize}
			\end{itemize}
		\end{column}
		\begin{column}{0.45\textwidth}
			\begin{itemize}
				\item \textbf{Proving in the ROM:}
				      \begin{enumerate}
					      \item Replace the hash function with a random oracle.
					      \item Show that the adversary has negligible advantage.
					      \item Conclude the scheme is secure (assuming the hash function behaves like a random oracle).
				      \end{enumerate}
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Limitations of the Random Oracle Model}
	\begin{itemize}
		\item \textbf{Gap between theory and practice}: no real hash function is a random oracle, so a ROM proof says nothing about attacks that exploit the \emph{structure} of SHA-256 specifically.
		\item \textbf{Artificial counterexamples exist}: Canetti, Goldreich and Halevi (1998) built schemes that are provably secure in the ROM, yet provably insecure once the oracle is replaced by \emph{any} concrete hash function.\footnote{\url{https://appliedcryptography.page/paper/\#rom-methodology}}
		      \begin{itemize}
			      \item The trick: the scheme misbehaves only when fed a short program that computes its own hash function. No such program exists for a random oracle; one exists for every real $H$.
		      \end{itemize}
		\item \textbf{In practice}: no \emph{natural} scheme has fallen this way, and much of deployed cryptography (RSA-OAEP, Ed25519, every Fiat--Shamir signature) has only ROM proofs. The model is a heuristic, but a remarkably reliable one.
		\item \ldots mostly. In 2.8 we'll see a 2025 attack on Fiat--Shamir where the gap between a random oracle and a real hash function is exploited on a real proof system.\footnote{\url{https://appliedcryptography.page/paper/\#prove-false}}
	\end{itemize}
\end{frame}

\begin{frame}{Key takeaways}
	\begin{columns}[c]
		\begin{column}{0.5\textwidth}
			\begin{itemize}
				\item \textbf{Collisions always exist}; collision resistance means nobody can find one. An $n$-bit hash gives at most $n/2$ bits of it (birthday bound).
				\item \textbf{Real breaks beat the birthday bound}: MD5 (2004) and SHA-1 (2017) fell to structural attacks. Never use them for anything security-related.
				\item \textbf{Merkle--Damg\aa rd leaks its state}: $H(k \| m)$ is not a MAC. Use HMAC (or a sponge).
			\end{itemize}
		\end{column}
		\begin{column}{0.5\textwidth}
			\begin{itemize}
				\item \textbf{Passwords}: salt (kills precomputation), slow down (kills brute force), use memory (kills ASICs). Today that means Argon2id.
				\item \textbf{Random oracle model}: a useful lie. Proofs in it rule out every attack except those that exploit the specific hash function, and so far that has been enough. Mostly.
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}
