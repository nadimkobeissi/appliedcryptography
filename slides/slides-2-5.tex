\section{The Case for Computer-Aided Cryptography}

\begin{frame}{Securely implementing cryptography}
	\begin{itemize}
		\item Designing, implementing, and deploying cryptographic mechanisms is notoriously hard
		\item Regular discoveries of high-profile design flaws, devastating implementation bugs, and side-channel vulnerabilities
		\item Even in \textbf{widely deployed} mechanisms!
	\end{itemize}
	\vspace{0.5em}
	\begin{center}
		\Large
		Three levels of complexity, three chances to fail:
	\end{center}
	\begin{enumerate}
		\item \textbf{Design level}: Is the scheme itself secure? (proofs, adversarial models)
		\item \textbf{Implementation level}: Does the code compute it correctly? (memory safety, speed)
		\item \textbf{Deployment level}: Does the running code keep its secrets? (side channels)
	\end{enumerate}
	\vspace{0.5em}
	\begin{center}
		\textit{Part 1 of this lecture takes each level in turn}
	\end{center}
\end{frame}

\begin{frame}{The problem}
	\begin{center}
		\Large
		Attackers can break designs, exploit bugs, recover secrets via side channels, or \textbf{any combination}
	\end{center}
	\vspace{0.7em}
	\begin{itemize}
		\item \textbf{Pen-and-paper proofs}: the only way to rule out entire classes of attacks, and required for any new standard. But they are tedious, cover only the ``core,'' and their errors go unnoticed for years
		\item \textbf{Simplified models}: a proof about an idealized protocol misses attacks on the real one
		\item \textbf{Aggressive optimization}: every hand-tuned line is a chance to get it wrong
		\item \textbf{Current modus operandi}: a select few experts with rudimentary tools
	\end{itemize}
	\vspace{0.7em}
	\begin{center}
		This approach \textbf{cannot keep pace} with the rate of innovation in cryptography
	\end{center}
\end{frame}

\begin{frame}{Enter: computer-aided cryptography (CAC)}
	\begin{itemize}
		\item Active area of research: \textbf{formal, machine-checkable} approaches to designing, analyzing, and implementing cryptography
		\item \textbf{Design level}: Tools manage the complexity of security proofs, can reveal subtle flaws
		\item \textbf{Implementation level}: Guarantee correct behavior, even for highly optimized code
		\item \textbf{Deployment level}: Check side-channel protections
	\end{itemize}
	\vspace{1em}
	\begin{center}
		Individual tools may only address \textit{part} of the problem, but when \textbf{combined}, they provide a \textbf{high degree of assurance}
	\end{center}
\end{frame}

\begin{frame}{Where verified crypto already ships}
	\begin{itemize}
		\item Computer-aided security analyses were influential in \textbf{TLS 1.3} standardization
		\item \textbf{\haclstar}: Verified crypto integrated into Mozilla Firefox's NSS
		\item \textbf{Fiat Cryptography}: Verified elliptic curve code in Google's BoringSSL, used by Chrome and Android
		\item \textbf{Linux kernel}: WireGuard's Curve25519 is built from verified \haclstar and Fiat code
		\item \textbf{Python}: CPython's bundled hash implementations (SHA-2, SHA-3, Blake2, HMAC) come from \haclstar
	\end{itemize}
	\vspace{0.5em}
	\begin{center}
		\textbf{Formally verified code at Internet scale!}
	\end{center}
\end{frame}

\begin{frame}{Proofs cover algorithms; attackers target systems}
	\begin{itemize}
		\item Pen-and-paper proofs consider \textbf{isolated algorithms}
		\item Real deployments involve massive software stacks:
		      \begin{itemize}
			      \item Operating system layers
			      \item Network protocol implementations
			      \item Library dependencies
			      \item Hardware abstraction layers
		      \end{itemize}
		\item Each layer introduces new attack surfaces and unexpected interactions
		\item \textbf{Result}: Security proofs may not hold in practice!
	\end{itemize}
	\vspace{0.5em}
	\begin{center}
		\textit{The SoK's own warning: CAC risks becoming a victim of its own success, its guarantees hedged by fine print that nobody reads. Hold that thought until Part 2}
	\end{center}
\end{frame}

\section{Design-Level Security}

\begin{frame}{Two flavors of design-level security}
	\begin{itemize}
		\item Developed over years in \textbf{two separate communities}:
		      \vspace{0.5em}
		\item \textbf{Symbolic security} (formal methods community)
		      \begin{itemize}
			      \item Messages are abstract \textbf{symbols}, not bits
			      \item Cryptographic primitives are \textbf{perfect black boxes}
			      \item Adversary has \textbf{full network control}: the \textbf{Dolev-Yao model}
		      \end{itemize}
		      \vspace{0.5em}
		\item \textbf{Computational security} (cryptography community)
		      \begin{itemize}
			      \item Messages are \textbf{bitstrings} of specific length
			      \item Encryption is a \textbf{probabilistic algorithm}
			      \item Adversaries have \textbf{bounded computational power}
		      \end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{Symbolic security: the Dolev-Yao model}
	\textbf{Symmetric encryption modeled by:}
	\begin{itemize}
		\item Functions: \texttt{Enc} and \texttt{Dec}
		\item Equational theory:
	\end{itemize}
	\begin{center}
		\Large
		\texttt{Dec(Enc(m,k),k) = m}
	\end{center}
	\begin{itemize}
		\item The \textbf{Dolev-Yao adversary} \textit{is} the network: it intercepts, modifies, injects, and replays any message it likes
		\item But it can only use the specified primitives and equations: no brute force, no partial guessing
		\item It can decrypt \texttt{Enc(m,k)} only if it knows the \textbf{entire} key \texttt{k}: all-or-nothing
	\end{itemize}
\end{frame}

\begin{frame}{Symbolic vs.\ computational: key differences}
	\begin{center}
		\begin{tabular}{|l|l|l|}
			\hline
			\textbf{Aspect} & \textbf{Symbolic} & \textbf{Computational}         \\
			\hline
			Messages        & Abstract terms    & Bitstrings                     \\
			\hline
			Adversary       & Rule-based        & Probabilistic algorithm        \\
			\hline
			Security        & All-or-nothing    & Probabilistic bounds           \\
			\hline
			Automation      & Mostly push-button & Often hand-guided             \\
			\hline
			Guarantees      & ``Cannot break''  & ``Hard to break''              \\
			\hline
			Tools           & ProVerif, Tamarin & CryptoVerif, EasyCrypt, \fstar \\
			\hline
		\end{tabular}
	\end{center}
	\vspace{1em}
	\begin{itemize}
		\item \textbf{Symbolic}: Quick protocol flaw detection, automated analysis
		\item \textbf{Computational}: Concrete security guarantees, cryptographic proofs
		\item \textbf{Both}: Different insights, complementary verification
	\end{itemize}
\end{frame}

\begin{frame}{Symbolic security properties}
	Two main flavors:
	\vspace{0.5em}
	\begin{enumerate}
		\item \textbf{Trace properties}
		      \begin{itemize}
			      \item Bad events never occur on any execution trace
			      \item Example: Secret data never in adversarial knowledge
		      \end{itemize}
		      \vspace{0.5em}
		\item \textbf{Equivalence properties}
		      \begin{itemize}
			      \item Adversary cannot distinguish between two protocols
			      \item Examples: Real secret vs.\ random value, vote privacy, unlinkability
		      \end{itemize}
	\end{enumerate}
	\vspace{0.5em}
	\begin{center}
		\textit{Equivalence properties cannot be naturally expressed as trace properties}
	\end{center}
\end{frame}

\begin{frame}{Computational security properties}
	Characterized along two \textbf{independent} axes:
	\vspace{0.5em}
	\begin{enumerate}
		\item \textbf{How is security stated?}
		      \begin{itemize}
			      \item \textbf{Game-based}: No adversary can win a specific game (e.g.\ IND-CPA)
			      \item \textbf{Simulation-based}: The real protocol is indistinguishable from an ideal one; supports \textbf{composition}, but proofs are more complex
		      \end{itemize}
		      \vspace{0.5em}
		\item \textbf{How is adversarial power quantified?}
		      \begin{itemize}
			      \item \textbf{Concrete}: ``Any adversary running in time $t$ succeeds with probability at most $\epsilon$''
			      \item \textbf{Asymptotic}: ``Any polynomial-time adversary succeeds with negligible probability''
		      \end{itemize}
	\end{enumerate}
	\vspace{0.5em}
	\begin{center}
		CAC tools primarily target \textbf{game-based, concrete} security---exactly the style of reasoning we practiced in Part 1 of this course
	\end{center}
\end{frame}

\begin{frame}{Key features of symbolic analysis tools}
	\begin{itemize}
		\item \textbf{Unbounded sessions}: Can analyze an arbitrary number of concurrent sessions, proving \textbf{absence of attacks} in the model
		\item \textbf{Trace properties}: Bad events never occur on any execution trace
		\item \textbf{Equivalence properties}: Adversary cannot distinguish between two protocols
		\item \textbf{Equational theories}: Model algebraic properties of crypto operations (XOR, DH, etc.)
		\item \textbf{Global mutable state}: Support for shared databases, key servers
	\end{itemize}
	\vspace{0.5em}
	\begin{center}
		Modern unbounded tools often \textbf{outperform} bounded tools, even for small session counts!
	\end{center}
\end{frame}

\begin{frame}{Switching sides: game-based security, mechanized}
	\begin{itemize}
		\item Probabilistic experiment: ``game'' between challenger and adversary
		\item Security statement:
		      \begin{center}
			      \textit{``For all adversaries, probability of achieving goal $\leq$\ threshold''}
		      \end{center}
		\item Core methodology: \textbf{Game hopping}
		      \begin{enumerate}
			      \item Start with original game
			      \item Transform step-by-step
			      \item Reach a game where the success probability is known
			      \item Bound the probability increase at each step
		      \end{enumerate}
	\end{itemize}
	\vspace{0.5em}
	\begin{center}
		\textit{This is the hybrid argument from Part 1 of this course, with a tool checking every hop}
	\end{center}
\end{frame}

\begin{frame}{Key features of computational analysis tools}
	\textbf{Automation vs. Expressivity:} Every tool makes this tradeoff
	\begin{itemize}
		\item Automated tools: Push-button proofs, limited patterns
		\item Expressive tools: Arbitrary arguments, more manual effort
	\end{itemize}
	\vspace{0.5em}
	\textbf{Other key dimensions:}
	\begin{itemize}
		\item \textbf{Composition support}: Can we reason about pieces separately?
		\item \textbf{Concrete security bounds}: Not just ``secure'' but ``how secure''
		\item \textbf{Link to implementation}: Can guarantees reach real code?
		\item \textbf{Trusted computing base}: Theorem prover? Tool itself? SMT solver?
	\end{itemize}
	\vspace{0.5em}
	\begin{center}
		Smaller TCB = Higher confidence
	\end{center}
\end{frame}

\begin{frame}{CryptoVerif: automated protocol analysis}
	\begin{center}
		\Large
		Best for highly automated protocol analysis
	\end{center}
	\vspace{0.5em}
	\begin{itemize}
		\item Both proof-finding and proof-checking
		\item Excellent for key exchange protocols
		\item Applied $\pi$-calculus input language
	\end{itemize}
	\vspace{0.5em}
	\textbf{Success stories:}
	\begin{itemize}
		\item TLS 1.3 and the Signal Protocol (same Ph.D. thesis!)
		\item WireGuard VPN
	\end{itemize}
	\vspace{0.5em}
	\begin{center}
		\small
		\textit{Both models take the peers' long-term public keys as authentic inputs. Where those keys come from is a separate problem.}
	\end{center}
\end{frame}

\begin{frame}{\fstar: beyond the cryptographic core}
	\begin{center}
		\Large
		Best for analyzing full protocols and systems
	\end{center}
	\vspace{0.5em}
	\begin{itemize}
		\item General-purpose verification language with \textbf{dependent types}
		\item A well-typed program is a \textbf{proof of its specification}
		\item Type checking = Proof checking, with Z3 doing the heavy lifting!
	\end{itemize}
	\vspace{0.5em}
	\textbf{Example:}
	\begin{itemize}
		\item Type: \texttt{point : \{(x,y) : int * int | y*y = x*x*x + a*x + b\}}
		\item This type \textbf{only} accepts valid elliptic curve points
		\item Your code literally cannot process invalid points if it type-checks
	\end{itemize}
\end{frame}

\begin{frame}{\fstar: attacks found, code verified}
	\textbf{How \fstar handles computational proofs:}
	\begin{enumerate}
		\item Start with detailed protocol description
		\item Transform into an ideal program with ideal crypto functionalities
		\item Validate the transformation manually, verify the final program automatically
	\end{enumerate}
	\vspace{0.5em}
	\textbf{Success stories:}
	\begin{itemize}
		\item Discovered new attacks on TLS
		\item miTLS: Verified reference implementation of TLS 1.2
		\item Implemented and verified TLS 1.3 record layer
	\end{itemize}
	\vspace{0.5em}
	\begin{center}
		Meanwhile, \textbf{EasyCrypt} verified the AWS Key Management System and the SHA-3 standard
	\end{center}
\end{frame}

\begin{frame}{The fine print}
	\begin{center}
		\Large
		Computer-aided security proofs are only as good as the statements being proven
	\end{center}
	\vspace{1em}
	\begin{itemize}
		\item Understanding these statements can be \textbf{challenging}
		\item Most proofs rely on \textbf{implicit assumptions}
		\item Reconstructing top-level statements is hard, even for experts
		\item Tools may have bugs (like any software)
	\end{itemize}
	\vspace{0.5em}
	\begin{center}
		\textit{One of those assumptions sits in a protocol most of you have on your phone\ldots}
	\end{center}
\end{frame}

\begin{frame}{Where does Bob's identity key come from?}
	The ProVerif and CryptoVerif models of the Signal Protocol prove what they claim. Both start with Alice already holding the right long-term identity key for Bob.
	\vspace{0.5em}
	\begin{itemize}
		\item In deployment that key reaches Alice inside a \textbf{prekey bundle}, served to her by Signal's servers
		\item A compromised or coerced server can serve a key it controls. Every theorem still holds, for a session with the wrong Bob
		\item Signal's long-standing answer: \textbf{safety numbers}, compared out of band, in person or over another channel
		\item Correct, and almost nobody did it. The assumption was discharged by a human action the model never sees
	\end{itemize}
	\vspace{0.5em}
	\begin{center}
		\textit{Key distribution sat outside the boundary. Closing it was the user's job}
	\end{center}
\end{frame}

\begin{frame}{Signal's automatic key verification (August 2026)}
	\footnotesize
	Since August 11, 2026, the mapping from a phone number or username to a public key is \textbf{globally consistent and auditable}.
	\vspace{0.3em}
	\begin{itemize}
		\item A \textbf{log tree} appends every registration, number change, username change and account re-creation
		\item \textbf{Prefix trees} index it, so a lookup is a deterministic search the server cannot steer
		\item A \textbf{verifiable random function} hides identifiers, a \textbf{keyed hash} hides values: auditors see no plaintext
		\item \textbf{Cloudflare} and \textbf{Trail of Bits} sign entries, so Alice and Bob cannot be shown different ledgers
		\item Clients re-check their own entry, and keys they already accepted, so later substitution is caught too
	\end{itemize}
	\vspace{0.3em}
	\begin{center}
		\scriptsize
		Built on an earlier draft of the IETF key transparency protocol. \url{https://signal.org/blog/automatic-key-verification/}
	\end{center}
\end{frame}

\begin{frame}{The limits of automatic key verification}
	\small
	The check runs on its own now, and ``View Safety Number'' reports \textbf{``Encryption verified''} when it passes. The limits:
	\vspace{0.4em}
	\begin{itemize}
		\item Verification needs the contact's \textbf{phone number}. Meet someone by username alone and there is nothing to check against
		\item Everyone sees the same key for an identifier. Nobody proves that the identifier belongs to the person you have in mind
		\item The server and its auditors still have to not collude, and clients still have to monitor
	\end{itemize}
	\vspace{0.4em}
	\begin{center}
		\textit{Software checks the assumption on a schedule now, which is more than the user ever did. We will return to this theme in Part 2\ldots}
	\end{center}
\end{frame}

\section{Deployment-Level Security: Side Channels}

\begin{frame}{Side channels: the hidden attack surface}
	\begin{center}
		\Large
		Design-level security assumes attackers only see inputs and outputs.

		\normalsize
		Reality: implementations leak much more!
	\end{center}
	\vspace{0.5em}
	\textbf{Information leaks through:}
	\begin{itemize}
		\item \textbf{Timing}: How long operations take
		\item \textbf{Memory access}: Which cache lines are touched
		\item \textbf{Power consumption}, \textbf{EM radiation}, \textbf{acoustics}
	\end{itemize}
	\vspace{0.5em}
	\textbf{Real attacks:} RSA/AES key recovery, Lucky 13, cross-VM extraction
\end{frame}

\begin{frame}{The constant-time solution}
	\begin{center}
		\Large
		Make execution independent of secrets
	\end{center}
	\vspace{0.5em}
	\textbf{Rules:} No secret-dependent branches, no secret-dependent memory access, no variable-time operations on secrets
	\vspace{0.5em}

	\textbf{Example --- constant-time selection:}
	\begin{flushleft}
		\textit{Insecure:} \texttt{if (secret\_bit) \{ result = a; \} else \{ result = b; \}}
	\end{flushleft}
	\begin{flushleft}
		\textit{Secure:} \texttt{mask = -secret\_bit; result = (a \& mask) | (b \& \textasciitilde mask);}
	\end{flushleft}
	\vspace{0.5em}
	\begin{center}
		Same result, no branches, \textit{much harder to understand}
	\end{center}
\end{frame}

\begin{frame}{Compilers and hardware betray you}
	\textbf{The compiler problem:}
	\begin{itemize}
		\item Compiler ``optimizations'' can reintroduce branches, use lookup tables, short-circuit operations
		\item The compiler doesn't know about your security requirements
	\end{itemize}
	\vspace{0.5em}
	\textbf{The hardware problem:}
	\begin{itemize}
		\item Branch prediction, speculative execution, cache hierarchies
		\item Variable-latency instructions
		\item Spectre/Meltdown-style attacks
	\end{itemize}
	\vspace{0.5em}
	\begin{center}
		\textit{Even perfect constant-time source code may not survive to execution!}
	\end{center}
\end{frame}

\begin{frame}{The pessimist case}
	\textbf{BearSSL's constant-time conditional move, written around 2015:}\footnote{\url{https://appliedcryptography.page/paper/\#pessimist-case}}
	\begin{flushleft}
		\texttt{mask1 = -(uint64\_t)(((x | -x) >> 31) \textasciicircum{} 1); mask2 = \textasciitilde mask1;}\\
		\texttt{a[i] = (mask1 \& a[i]) | (mask2 \& b[i]);}
	\end{flushleft}
	{\scriptsize\textit{Builds an all-ones or all-zeros mask from the secret \texttt{x}, then blends \texttt{a} and \texttt{b} word by word: every word is touched, whatever \texttt{x} is.}}
	\vspace{0.4em}

	\textbf{What Clang 18.1.3 emits for it today, at \texttt{-O3}:}
	\begin{flushleft}
		\texttt{test ecx, ecx} \quad \texttt{je .LBB0\_2} \quad \texttt{jmp \textbf{memcpy@PLT}}
	\end{flushleft}
	{\scriptsize\textit{Clang saw through the arithmetic: it now \textbf{branches on the secret} and calls \texttt{memcpy}, so timing and cache behavior reveal \texttt{x}: exactly the leak the mask existed to prevent.}}
	\vspace{0.4em}
	\begin{center}
		\textit{Not a compiler defect. Finding that shortcut is exactly its job}
	\end{center}
\end{frame}

\begin{frame}{Why constant-time is hard}
	\begin{center}
		\Large
		Natural code patterns become forbidden
	\end{center}
	\begin{columns}[t]
		\begin{column}{0.5\textwidth}
			\textbf{You can't write:}
			\begin{enumerate}
				\item \texttt{if (secret\_bit) \{ ... \}}
				\item \texttt{array[secret\_index]}
				\item Return early if \texttt{secret == 0}
				\item Use division only when \texttt{secret != 0}\footnote{\url{https://kyberslash.cr.yp.to}}
			\end{enumerate}
		\end{column}
		\begin{column}{0.5\textwidth}
			\textbf{Instead, you must:}
			\begin{itemize}
				\item Use bit manipulation tricks
				\item Process all data uniformly
				\item Avoid natural optimizations
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{The TLS patching cycle}
	\begin{center}
		\Large
		Even experts get constant-time wrong
	\end{center}
	\vspace{0.5em}
	\textbf{The vicious cycle:}
	\begin{enumerate}
		\item 2002: Vaudenay's padding oracle attack against CBC mode in TLS
		\item Countermeasure: Decrypt anyway, compute the MAC over dummy data
		\item 2013: Lucky 13 exploits the tiny timing skew that countermeasure left behind (remember topic 2.1?)
		\item OpenSSL patches Lucky 13
		\item 2016: The patch introduces a \textit{new} padding oracle (CVE-2016-2107)!
	\end{enumerate}
	\vspace{0.5em}
	\begin{center}
		\textit{Each ``fix'' introduced a new vulnerability---for over a decade}
	\end{center}
\end{frame}

\begin{frame}{Formal leakage models}
	\textbf{Three levels of strictness. Each fixes what the attacker is assumed to observe:}
	\begin{enumerate}
		\item \textbf{Program counter (PC) model}: control flow only. Covers simple timing attacks
		\item \textbf{Constant-time model}: PC + memory addresses. Covers cache attacks; the usual target
		\item \textbf{Size-respecting model}: PC + memory + operand sizes. Covers variable-time instructions
	\end{enumerate}
	\vspace{0.4em}
	\textbf{Security = leakage independent of secrets} (\textbf{observational non-interference}), and a tool can check it for \textit{all} inputs, not just the ones a test suite tried
	\vspace{0.4em}
	\begin{itemize}
		\item \textbf{Caveat 1}: the proof is relative to the model. Hardware may leak more (Spectre)
		\item \textbf{Caveat 2}: physical attacks bypass all software protections (next slides)
	\end{itemize}
\end{frame}

			\textbf{Physical interposition on DDR5 bus:}
			\begin{itemize}
				\item Built with cheap, off-the-shelf electronics
				\item Inspects all memory traffic in real time
			\end{itemize}

\section{Case Study: Lessons from TLS}

\begin{frame}{TLS before 1.3: reactive patching}
	\begin{center}
		\Large
		One of the most important deployments of cryptography

		\normalsize
		Powers HTTPS, secures billions of connections daily
	\end{center}
	\vspace{0.5em}
	\textbf{The endless loop (pre-1.3):}
	\begin{enumerate}
		\item Attack found $\rightarrow$ Quick patch $\rightarrow$ Long-term fix $\rightarrow$ New attack
	\end{enumerate}
	\vspace{0.2em}
	\textit{The Lucky 13 patching cycle from the last section is one lap of this loop}

	\vspace{0.3em}
	\textbf{When academics looked at real TLS:} BEAST, CRIME, Lucky 13, POODLE, Logjam\ldots
	\vspace{0.3em}
	\begin{center}
		\textit{No substantial academic analysis during design}
	\end{center}
\end{frame}

\begin{frame}{TLS 1.3: designed and analyzed together}
	\begin{center}
		\Large
		From reactive patching to proactive design
	\end{center}
	\vspace{0.5em}
	\begin{itemize}
		\item The academic community was consulted \textbf{during} design: 28 drafts over 4+ years
		\item Multiple CAC tools applied: ProVerif, Tamarin, CryptoVerif, \fstar
		\item Real attacks found and fixed \textbf{before} standardization:
		      \begin{itemize}
			      \item Draft 10: Client impersonation in PSK-resumption
			      \item Draft 13: Attack on 0-RTT client authentication
			      \item Draft 21: Unexpected authentication behavior
		      \end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{Three lessons from TLS 1.3}
	\textbf{Lesson 1: Formal methods find real flaws}
	\begin{itemize}
		\item Just \textit{trying} to write formal specs uncovers ambiguities and corner cases
		\item \textit{``If you can't specify it clearly, it's probably broken''}
	\end{itemize}
	\vspace{0.3em}
	\textbf{Lesson 2: Protocols are moving targets}
	\begin{itemize}
		\item 28 drafts = 28 different protocols to verify
		\item Machine-checked proofs can evolve with the protocol; manual proofs can't keep pace
	\end{itemize}
	\vspace{0.3em}
	\textbf{Lesson 3: Design for verifiability}
	\begin{itemize}
		\item Small changes (key separation, consistent tagging) dramatically simplify analysis
		\item Negligible performance impact, huge verification benefit
	\end{itemize}
	\vspace{0.3em}
	\begin{center}
		\textit{From ``ship and patch'' to ``verify and ship''}
	\end{center}
\end{frame}

\begin{frame}{Part 1 recap: the promise}
	\begin{itemize}
		\item \textbf{Design level}: Symbolic (Verifpal, ProVerif, Tamarin) and computational (CryptoVerif, \fstar, EasyCrypt) tools find real flaws and prove real theorems
		\item \textbf{Implementation level}: Verified code now matches or beats hand-optimized code (\haclstar, Fiat, EverCrypt)
		\item \textbf{Deployment level}: Constant-time discipline can be checked against explicit leakage models
		\item \textbf{TLS 1.3}: Verification during design became the new playbook
	\end{itemize}
	\vspace{1em}
	\begin{center}
		\Large
		So: Is cryptographic software a solved problem?

		\vspace{0.3em}

		\normalsize
		\textit{Not quite\ldots}
	\end{center}
\end{frame}

\section{The Verification Boundary}

\begin{frame}{What is a verification boundary?}
	\definitionbox{Verification Boundary}{The interface separating code whose properties carry \textbf{machine-checked proofs} from code that is \textbf{trusted without proof}.}
	\vspace{0.5em}
	\begin{itemize}
		\item Every formally verified system embeds one. It can be drawn tightly or loosely, but it always exists
		\item Outside the boundary: compilers, operating system, hardware, platform intrinsics, ``glue code''
		\item Experts understand that the boundary is there. Users often do not
	\end{itemize}
	\vspace{0.5em}
	\begin{center}
		\textit{The question is whether a project tells you where its boundary runs}
	\end{center}
\end{frame}

\begin{frame}{Boundaries documented, and boundaries assumed}
	\textbf{Projects that treat the boundary as first-class documentation:}
	\begin{itemize}
		\item \textbf{CompCert}: documents which compiler passes are verified and which (parser, assembler, linker) are not
		\item \textbf{seL4}: maintains a dedicated page listing every trust assumption about hardware behavior
		\item \textbf{EverCrypt}: names the compiler, the Vale embedding semantics, and the hardware as trusted
		\item \textbf{Jasmin}: documents which compiler passes carry machine-checked proofs
	\end{itemize}
	\vspace{0.5em}
	\textbf{What happens when a library says ``formally verified'' with no qualification:}
	\begin{itemize}
		\item Users read it as a guarantee over the whole library
		\item The gap between the claim and the reality produces false assurance
		\item Call that gap \textbf{verification theatre}
	\end{itemize}
\end{frame}

\begin{frame}{Case study: Cryspen's libcrux}
	\begin{itemize}
		\item Cryspen: ``services and software for high assurance software''
		\item libcrux repository: ``the formally verified crypto library''
		\item Documentation: ``every implementation in libcrux is formally verified''
	\end{itemize}
	\vspace{0.5em}
	\textbf{Where it ships:}
	\begin{itemize}
		\item ML-KEM implementation used internally at Google
		\item hpke-rs is a dependency of Signal's \texttt{signal-crypto} crate
		\item hpke-rs is used by OpenMLS
	\end{itemize}

	\end{center}
\end{frame}

\section{A Taxonomy of Verification Failures}

\begin{frame}{Five types of verification boundary failure}
	\begin{center}

		\item Types I--IV: bugs at or \textbf{beyond} the verification boundary

	\end{itemize}

\begin{frame}{Type I: platform abstractions the verifier takes on faith}
	\small
	Verified code calls platform intrinsics marked \texttt{\#[hax\_lib::opaque]}: the verifier takes their specification as an \textbf{axiom} and never looks at the implementation.

	\textbf{V1: Incorrect ARM64 SHA-3 intrinsic}
	\begin{itemize}
		\item The \texttt{\_vxarq\_u64} wrapper falls back to software on ARM64 chips without SHA-3 instructions, and that fallback was wrong
		\item The axiom was correct; the implementation behind it was not. Found independently by Filippo Valsorda
	\end{itemize}
	\textbf{V2: Cross-backend endianness bug}
	\begin{itemize}
		\item The portable backend serialized ML-KEM state in a different byte order than the SIMD backends
		\item Each backend was proved correct on its own. Nobody specified that they had to agree
		\item Caused \textbf{decryption failures in Signal's post-quantum ratchet}; reported by GuuJiang, December 2025
	\end{itemize}
\end{frame}

\begin{frame}{Type II: unverified protocol logic}
	\begin{center}
		\Large
		The primitives are used correctly. The protocol around them has bugs.
	\end{center}
	\vspace{0.7em}
	\textbf{V3: Missing X25519 zero check}
	\begin{itemize}
		\item RFC 9180 requires rejecting an all-zero Diffie-Hellman output. hpke-rs returns it unchecked
		\item A small-subgroup attack can force the shared secret to zero
	\end{itemize}
	\vspace{0.3em}
	\textbf{V4: Nonce reuse via integer overflow}
	\begin{itemize}
		\item The HPKE sequence counter is a \texttt{u32}. Release builds wrap silently back to 0
		\item After $2^{32}$ messages, the nonce repeats
	\end{itemize}
	\vspace{0.5em}
	\begin{center}
		\textit{Standard-mandated checks live in protocol code, which nothing here verifies}
	\end{center}
\end{frame}

\begin{frame}{Type III: specification incompleteness}
	\begin{center}
		\Large
		The code is provably correct against its spec. The spec omits a real requirement.
	\end{center}
	\vspace{1em}
	\textbf{V5: ECDSA missing low-$S$ normalization}
	\begin{itemize}
		\item ECDSA signatures are malleable: both $(r, s)$ and $(r, n-s)$ verify
		\item Bitcoin, Ethereum and others require $s \leq n/2$, the ``low-$S$'' form
		\item libcrux's \fstar specification never states that requirement
		\item The implementation is correct per spec, and rejected by the systems that consume it
	\end{itemize}
	\vspace{0.5em}
	\begin{center}
		\textit{Nothing is wrong with the proof. The thing being proved is too weak.}
	\end{center}
\end{frame}

\begin{frame}{Type IV: unverified wrapper logic}
	\begin{center}
		\Large
		The verified core is correct. The API layer around it introduces defects.
	\end{center}
	\vspace{1em}
	\textbf{V6: Ed25519 double clamping}
	\begin{itemize}
		\item The key generation wrapper clamps before hashing; the standard hashes first, then clamps
		\item Reduces the entropy of generated keys
	\end{itemize}
	\vspace{0.5em}
	\textbf{V7: AES-GCM decryption panic}
	\begin{itemize}
		\item The decryption wrapper calls \texttt{.unwrap()} on its result
		\item Any tampered ciphertext panics the process, and the correct error handling sits right above it, commented out
	\end{itemize}
\end{frame}

\begin{frame}{V8 and V9: two FIPS 204 checks that never fire}
	\small
	\textbf{V8: the norm check is dead code}
	\begin{flushleft}
		\texttt{if vector\_infinity\_norm\_exceeds(response, \textbf{(2 << GAMMA1\_EXPONENT)} - BETA)}
	\end{flushleft}
	\begin{itemize}
		\item \texttt{2 << GAMMA1\_EXPONENT} is $2\gamma_1$, not $\gamma_1$, so the bound is doubled. Any $z$ the signature encoding can express has norm at most $\gamma_1$, below the doubled bound, so the check can never fire
	\end{itemize}
	\vspace{0.4em}
	\textbf{V9: the hint bound is checked one iteration late}
	\begin{flushleft}
		\texttt{if (\textbf{previous\_true\_hints\_seen} > max\_ones\_in\_hint) \{ malformed\_hint = true; \}}
	\end{flushleft}
	\begin{itemize}
		\item Should compare \texttt{current\_true\_hints\_seen}; the final iteration is never validated
	\end{itemize}
	\vspace{0.4em}
	\begin{center}
		\textit{Both are FIPS 204 requirements, and libcrux accepts non-conforming ML-DSA signatures}
	\end{center}
\end{frame}

\begin{frame}{Type V: bugs inside the verified code}
	\begin{center}
		\Large
		Everything so far lived outside the boundary. These four live inside it.
	\end{center}
	\vspace{0.4em}
	\textbf{Three mechanisms let a false proof survive:}
	\begin{enumerate}
		\item \textbf{Wrong specifications.} A formal spec is a mathematical statement. Proving a wrong statement buys nothing
		\item \textbf{Admitted proofs.} The build system can skip proof checking and still report success
		\item \textbf{False axioms.} An unproved assumption can contradict reality, and every proof above it becomes unsound
	\end{enumerate}
	\vspace{0.5em}
	\begin{center}
		\textit{Four examples follow: two wrong specs, one admitted proof, one false axiom}
	\end{center}
\end{frame}

\begin{frame}{V10 and V11: two wrong ML-KEM specifications}
	\textbf{V10: the decompression constant does not depend on $d$}
	\begin{flushleft}
		FIPS 203: $\mathtt{decompress}_d(x) = \lceil x \cdot q / 2^d \rfloor = (x \cdot 3329 + 2^{d-1}) / 2^d$\\
		libcrux: \texttt{let decompress\_d d x = (x * v\_FIELD\_MODULUS + \textbf{1664}) / pow2 d}
	\end{flushleft}
	\begin{itemize}
		\item $1664 = (q-1)/2$, a fixed value, where the rounding constant should be $2^{d-1}$
		\item ML-KEM uses $d \in \{1, 4, 5, 10, 11\}$, and all five come out wrong
	\end{itemize}
	\vspace{0.4em}
	\textbf{V11: the inverse NTT is missing}
	\begin{flushleft}
		\texttt{let v = poly\_add (poly\_add (\textbf{vector\_dot\_product\_ntt} t\_as\_ntt r\_as\_ntt) error\_2) mu}
	\end{flushleft}
	\begin{itemize}
		\item The dot product lands in the NTT domain; \texttt{error\_2} and \texttt{mu} are coefficients
		\item Adding them is meaningless, and it type-checks anyway, because both domains share one type
	\end{itemize}
\end{frame}

\begin{frame}{V12: a false serialization proof}
	\textbf{The claim:}
	\begin{flushleft}
		\texttt{let deserialize\_12\_bit\_vec\_lemma\_bounded (v: t\_Array u8 (sz 24))}\\
		\texttt{~~: squash (forall (i: nat \{i < 16\}).}\\
		\texttt{~~~~~~bounded (Seq.index result.f\_elements i) \textbf{1})}
	\end{flushleft}
	\vspace{0.5em}
	\begin{itemize}
		\item The lemma says 12-bit deserialized values are bounded by $2^{\textbf{1}} = 2$
		\item The correct bound is $2^{12} = 4096$
		\item A 12-bit value reaches 4095, so the statement is false on its face
	\end{itemize}
	\vspace{0.5em}
	\begin{center}
		\Large
		So how did it get accepted?
	\end{center}
\end{frame}

\begin{frame}{The lax-mode escape hatch}
	\textbf{The proof tactic:}
	\begin{flushleft}
		\texttt{let prove\_bit\_vector\_equality' () : Tac unit =}\\
		\texttt{~~if \textbf{lax\_on ()}}\\
		\texttt{~~then \textbf{iterAll tadmit}}\\
		\texttt{~~else prove\_bit\_vector\_equality'' ()}
	\end{flushleft}
	\vspace{0.5em}
	\textbf{Three layers, each individually defensible:}
	\begin{enumerate}
		\item The Makefile admits this module by default (\texttt{ADMIT\_MODULES})
		\item When a module is admitted, \texttt{lax\_on()} returns true
		\item The tactic then calls \texttt{tadmit}, which accepts every goal unconditionally
	\end{enumerate}
	\vspace{0.5em}
	\begin{center}
		\textit{The false proof becomes a ``verified'' proof, and full verification never runs in any automated pipeline}
	\end{center}
\end{frame}

\begin{frame}{V13: an axiom that computes the wrong thing}
	\textbf{ML-DSA AVX2 intrinsics specification:}
	\begin{flushleft}
		\texttt{let i16\_mul\_32extended (x y: i16): i32 =}\\
		\texttt{~~(cast x : i32) *! (cast \textbf{x} : i32) ~~// should be: cast \textbf{y}}
	\end{flushleft}
	\vspace{0.5em}
	\begin{itemize}
		\item The second parameter \texttt{y} is ignored, and the body computes $x^2$ instead of $x \cdot y$
		\item The specification is axiomatized with \texttt{[@@ "opaque\_to\_smt"]} and never proved against a reference model
		\item Every proof resting on this axiom is unsound
	\end{itemize}
	\vspace{0.5em}
	\begin{center}
		\textit{The verified ML-DSA AVX2 proofs establish properties of $x^2$, not of $x \cdot y$}
	\end{center}
\end{frame}

\section{Getting the Boundary Right}

\begin{frame}{How AWS does it differently}
	\footnotesize
	\begin{center}
		\small
		\begin{tabular}{|l|c|c|}
			\hline
			\textbf{Practice}               & \textbf{Cryspen} & \textbf{AWS} \\
			\hline
			Soundness documents             & Manual/stale     & Structured   \\
			\hline
			Proofs in CI                    & Not triggered    & Every commit \\
			\hline
			Admitted modules documented     & In Makefile only & Explicit     \\
			\hline
			Verification scope communicated & Unqualified      & Qualified    \\
			\hline
			Axioms tested/proved            & Neither          & Tested       \\
			\hline
		\end{tabular}
	\end{center}
	\vspace{0.4em}
	\begin{itemize}
		\item aws-lc-verification (SAW/Cryptol), s2n-bignum (HOL Light) and mlkem-native (CBMC) face the same boundary problems, and their soundness documents enumerate verified properties, assumptions, residual risks and open gaps
		\item seL4 remains the high-water mark: one page listing every trust assumption
	\end{itemize}
	\vspace{0.2em}
	\begin{center}
		\scriptsize
		\textit{Thompson's question from 1984 has not gone away: \fstar dispatches its proofs through Z3, and Z3 has bugs like anything else. The trusted computing base is never zero, but the proofs can at least be run.}
	\end{center}
\end{frame}

\begin{frame}{Five recommendations}
	\begin{enumerate}
		\item \textbf{Machine-readable verification manifests.} Every release ships a manifest of which code paths are verified, against which properties, for which platforms, generated from the build system rather than maintained by hand
		\item \textbf{Proofs in CI on every commit.} Discharge all proof obligations. Flag proofs too slow to run as incomplete instead of admitting them silently
		\item \textbf{No lax-mode escape hatches.} A proof tactic should fail loudly in lax mode, not admit every goal
		\item \textbf{Scope the claim.} ``Formally verified'' should name the properties, the code paths and the platforms. Reserve the unqualified phrase for a minimal TCB, at CompCert or seL4 level
		\item \textbf{Defense in depth.} Code review, specification compliance auditing, cross-platform testing and fuzzing catch what formal verification does not
	\end{enumerate}
\end{frame}

\begin{frame}{Verification theatre, defined}
	\definitionbox{Verification Theatre}{The practice of deploying formal methods in a manner that:
		\begin{enumerate}
			\item Establishes machine-checked proofs for a \textbf{subset} of code and properties, where some proofs may themselves be \textbf{unchecked, vacuous, or built on false axioms},
			\item Communicates these proofs using language that implies \textbf{comprehensive coverage},
			\item Fails to adequately disclose the verification boundary, unverified components, unchecked proofs, or properties not covered.
		\end{enumerate}}
	\vspace{0.3em}
	\begin{center}
		\small
		\textbf{A spectrum of honest claims:} \textit{lax-checked} (syntax and types) $\rightarrow$ \textit{partially verified} (some modules admitted) $\rightarrow$ \textit{verified with caveats} (all proofs run, trust assumptions documented) $\rightarrow$ \textit{fully verified} (minimal TCB, proofs in CI, boundary documented).

		Most projects sit at the middle two. Their claims should say so.
	\end{center}
\end{frame}

\section{Verifpal: A Symbolic Verifier for the Rest of Us}

\begin{frame}{Where Verifpal sits}
	Part 1 gave you the tool table. Verifpal belongs in the symbolic column, next to ProVerif and Tamarin, and it aims at a different reader.
	\vspace{0.5em}
	\begin{center}
		\small
		\begin{tabular}{|l|l|l|l|}
			\hline
			                     & \textbf{ProVerif}  & \textbf{Tamarin}   & \textbf{Verifpal}      \\
			\hline
			Model                & Applied $\pi$      & Multiset rewriting & Principals \& messages \\
			\hline
			Primitives           & User-defined       & User-defined       & 25 built in, fixed     \\
			\hline
			Sessions             & Unbounded          & Unbounded          & Bounded (2 by default) \\
			\hline
			Termination          & Not guaranteed     & Not guaranteed     & Always                 \\
			\hline
			Observational equiv.\ & Yes               & Yes                & No                     \\
			\hline
			Output               & Clause derivations & Interactive proof  & Numbered causal steps  \\
			\hline
		\end{tabular}
	\end{center}
	\vspace{0.5em}
	\begin{center}
		Same Dolev-Yao attacker, same symbolic abstraction. Different audience.
	\end{center}
\end{frame}

\begin{frame}{Why a third symbolic tool?}
	\small
	\textbf{The barrier is rarely the theory. It is the modeling language.}
	\begin{itemize}
		\item ProVerif's applied $\pi$-calculus reads naturally if you already know the ML family behind it
		\item Its output is Horn-clause derivations, not Alice-and-Bob messages; recovering the attack takes an afternoon
		\item Tamarin is more expressive still; Part 1 told you what TLS 1.3 cost there: a week and 100GB of RAM
	\end{itemize}
	\vspace{0.3em}
	\textbf{Verifpal inverts the usual research order: user first, verification capability second.}
	\begin{itemize}
		\item Explicit principals with their own state, knowledge and computations
		\item Primitives ship with the tool, so you cannot define AEAD incorrectly
		\item Attacks print as numbered steps in your model's names, each \textbf{re-executed against real principal state}: a solver bug costs a missed attack, never an invented one
		\item Two sessions per principal by default, so replays come for free
	\end{itemize}
\end{frame}

\section{The Verifpal Language}

\begin{frame}{Every model has four parts, in this order}
	\begin{enumerate}
		\item \textbf{The attacker.} \texttt{attacker[passive]} or \texttt{attacker[active]}: what you are up against
		\item \textbf{The principals.} \texttt{principal Alice[\ldots]}: who exists, what they know, what they compute
		\item \textbf{The messages.} \texttt{Alice -> Bob: ga}: what crosses the network, and therefore what the attacker sees
		\item \textbf{The queries.} \texttt{queries[\ldots]}: the questions you want answered
	\end{enumerate}
	\vspace{0.3em}
	\begin{itemize}
		\item The \texttt{queries} block is mandatory and must come last
		\item Anything written after it is an error, so a stray principal at the end of a file cannot go unanalyzed
		\item Between messages and queries, optionally: \texttt{phase[n]} and \texttt{scenarios} blocks
		\item Comments are \texttt{//} to end of line, or \texttt{/* \ldots */} across lines
	\end{itemize}
\end{frame}

\begin{frame}[fragile]{A complete model}
	\begin{columns}[t]
		\begin{column}{0.53\textwidth}
			\begin{vpmodel}{A pre-shared-key exchange}
				\begin{lstlisting}
attacker[active]
principal Alice[
  knows private k
  generates n1, m1
  e1 = AEAD_ENC(k, n1, m1, nil)
]
Alice -> Bob: n1, e1
principal Bob[
  knows private k
  d1 = AEAD_DEC(k, n1, e1, nil)?
]
queries[
  confidentiality? m1
  authentication? Alice -> Bob: e1
]
\end{lstlisting}
			\end{vpmodel}
		\end{column}
		\begin{column}{0.45\textwidth}
			\small
			All four parts, in order:
			\vspace{0.2em}
			\begin{itemize}
				\item An \textbf{active} attacker owns the network
				\item Alice and Bob share a long-term \texttt{k}. She seals a fresh \texttt{m1} under a fresh nonce \texttt{n1}
				\item Both cross the network, as in any real implementation
				\item Two questions: can the attacker read \texttt{m1}, and could Bob accept an \texttt{e1} Alice never sent?
			\end{itemize}
			\vspace{0.3em}
			The \texttt{?} makes that decryption \textbf{checked}: if it fails, Bob halts there.
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{The attacker}
	\begin{columns}[t]
		\begin{column}{0.48\textwidth}
			\textbf{\texttt{attacker[passive]}}
			\begin{itemize}
				\item A malicious observer
				\item Sees every value that crosses the network
				\item Reasons about what it can derive from them
				\item Injects nothing, modifies nothing
			\end{itemize}
		\end{column}
		\begin{column}{0.48\textwidth}
			\textbf{\texttt{attacker[active]}}
			\begin{itemize}
				\item Owns the network
				\item Drops, replays and replaces any unguarded value
				\item Injects values it has constructed itself
				\item Knows \texttt{nil}, hence the private half of \texttt{PUBKEY(nil)}
			\end{itemize}
		\end{column}
	\end{columns}
	\vspace{0.4em}
	\begin{center}
		The same model under both attackers answers two different questions.

		\vspace{0.1em}

		\textit{Under a passive attacker, the last slide's model passes both queries. Under an active one, confidentiality holds, but authentication fails: an \texttt{e1} from another run decrypts just as well under the shared \texttt{k}.}
	\end{center}
\end{frame}

\begin{frame}{Principals}
	\begin{itemize}
		\item Declared with a name and a block of operations, and each has independent state
		\item A principal may be declared \textbf{more than once}. Each later block continues where that principal left off
		\item That is how you write a protocol as turns: Alice acts, sends, Bob replies, Alice acts on what she received
		\item Names are title-cased, so \texttt{alice} and \texttt{Alice} are the same principal
		\item At most 128 principals per model
	\end{itemize}
	\vspace{0.5em}
	\begin{center}
		Every principal is analyzed as \textbf{two concurrent sessions} by default.

		\vspace{0.3em}

		\small Set the count with \texttt{verifpal verify model.vp --sessions k}, up to sixteen. This matters more than it sounds, and we will come back to it.
	\end{center}
\end{frame}

\begin{frame}{Four ways a constant enters a model}
	\begin{itemize}
		\item \texttt{knows public c0} \quad Everyone has it, the attacker included
		\item \texttt{knows private m1} \quad Prior knowledge, not shared with the attacker. \textbf{Long-term}: the same value in every session and every run
		\item \texttt{generates a} \quad \textbf{Fresh}: drawn anew on every run, and separately in every session
		\item \texttt{leaks a} \quad Hands a value the principal already knows to the attacker, at that point in the protocol
		\item \texttt{ga = PUBKEY(a)} \quad Assignment, always to a primitive expression
	\end{itemize}
	\vspace{0.3em}
	\begin{center}
		\small
		\textit{The split between \texttt{knows} and \texttt{generates} is what makes cross-session attacks possible: two sessions share the long-term key, so a message from one can verify in the other.}
	\end{center}
\end{frame}

\begin{frame}{Four rules that keep models readable}
	\begin{enumerate}
		\item \textbf{Immutability.} Once assigned, a constant cannot be reassigned
		\item \textbf{Global name-space.} A name denotes one value everywhere. If Bob generates or assigns \texttt{c}, Alice cannot introduce a \texttt{c}. Repeating the same \texttt{knows} line, same qualifier, is how two principals share one long-term value
		\item \textbf{No referencing.} Constants are assigned to primitives, never to other constants
		\item \textbf{Reserved names.} No keywords, no primitive names, nothing beginning with \texttt{attacker} or \texttt{unnamed}
	\end{enumerate}
	\vspace{0.3em}
	\begin{itemize}
		\item Together these guarantee that a name means the same value everywhere it appears, so any line reads on its own
		\item Identifiers are case-insensitive: \texttt{AEAD\_ENC} and \texttt{aead\_enc} are the same primitive
		\item Write \texttt{\_} on the left of an assignment when you will not refer to the result again, which is the usual way to write a checked primitive you only care about for its side effect
	\end{itemize}
\end{frame}

\begin{frame}[fragile]{Messages, and guarded constants}
	\begin{vpmodel}{Sending values}
		\begin{lstlisting}
Alice -> Bob: ga, e1     // the attacker may read and replace both
Bob -> Alice: [gb], e2   // gb is guarded: readable, not replaceable
\end{lstlisting}
	\end{vpmodel}
	\vspace{0.2em}
	\footnotesize
	\begin{itemize}
		\item Only constants may be sent, and messages live outside principal blocks
		\item Brackets \textbf{guard} a value: an active attacker still observes it, but cannot substitute its own
		\item A guard says the recipient already knows what that value should be: a certificate, a QR code, a trusted directory, a fingerprint checked out of band
	\end{itemize}
	\vspace{0.2em}
	\alertbox{A guard is a claim about the real world}{Guarding every public key models a deployment where every key was pre-authenticated, and few deployments provide that. One exception is deliberate: if the same constant \textit{also} reached its sender unguarded, Verifpal treats it as substitutable anyway.}
\end{frame}

\begin{frame}[fragile]{Checked primitives: the question mark}
	\begin{vpmodel}{Two decryptions, two behaviors}
		\begin{lstlisting}
d1 = AEAD_DEC(k, n1, e1, c0)   // failure produces a value and Bob carries on
d2 = AEAD_DEC(k, n1, e1, c0)?  // failure halts Bob here, permanently
\end{lstlisting}
	\end{vpmodel}
	\vspace{0.3em}
	\begin{itemize}
		\item Six primitives are checkable: \texttt{ASSERT}, \texttt{SPLIT}, \texttt{AEAD\_DEC}, \texttt{SIGNVERIF}, \texttt{RINGSIGNVERIF}, \texttt{KEM\_DECAP}. \texttt{DEC}, \texttt{PKE\_DEC} and \texttt{UNBLIND} rewrite too, but a mismatch there yields a different value rather than a detectable failure
		\item A check constrains the attacker: whatever it substitutes must still satisfy the check, or the principal stops and computes nothing further
		\item A verification whose outcome nothing depends on is not a verification. \texttt{SIGNVERIF} without \texttt{?} lets the principal sail past a signature that did not verify
	\end{itemize}
	\vspace{0.4em}
	\begin{center}
		\textit{What you check decides what counts as someone having been fooled, which is why authentication queries lean so heavily on this one character.}
	\end{center}
\end{frame}

\begin{frame}{The primitive catalog}
	\begin{center}
		\scriptsize
		\begin{tabular}{|l|l|}
			\hline
			\textbf{Group}    & \textbf{Primitives}                                                 \\
			\hline
			Core              & \texttt{ASSERT}, \texttt{CONCAT}, \texttt{SPLIT}                    \\
			\hline
			Hashing           & \texttt{HASH}, \texttt{MAC}, \texttt{HKDF}                         \\
			\hline
			Public key        & \texttt{PUBKEY}, \texttt{DH\_KEX}                                   \\
			\hline
			Encryption        & \texttt{ENC}, \texttt{DEC}, \texttt{AEAD\_ENC}, \texttt{AEAD\_DEC},           \\
			                  & \texttt{PKE\_ENC}, \texttt{PKE\_DEC}                                          \\
			\hline
			Signatures        & \texttt{SIGN}, \texttt{SIGNVERIF}, \texttt{RINGSIGN},                         \\
			                  & \texttt{RINGSIGNVERIF}, \texttt{BLIND}, \texttt{UNBLIND}                      \\
			\hline
			Key encapsulation & \texttt{KEM\_ENCAP}, \texttt{KEM\_DECAP}                                      \\
			\hline
			Threshold         & \texttt{THRESHOLD\_SPLIT}, \texttt{THRESHOLD\_SIGN}, \texttt{THRESHOLD\_JOIN} \\
			\hline
		\end{tabular}
	\end{center}
	\vspace{0.3em}
	\footnotesize
	\begin{itemize}
		\item Twenty-five, and you cannot add a twenty-sixth. That is the point
		\item \texttt{DH\_KEX(their\_public, my\_private)} is commutative, so Alice's and Bob's secrets are the same term
		\item A primitive may be annotated \textbf{broken}: \texttt{SIGN[forgeable](sk, m)}, also \texttt{weak} and \texttt{malleable}, once phases are on the table
	\end{itemize}
\end{frame}

\begin{frame}[fragile]{The nonce slot, and what fills it}
	\begin{columns}[t]
		\begin{column}{0.45\textwidth}
			\begin{vpmodel}[top=0.5mm, bottom=0.5mm]{A counter that never advances}
				\begin{lstlisting}[basicstyle=\vpcode]
attacker[passive]
principal Alice[
  knows private k
  knows public ctr
  generates m1, m2
  e1 = AEAD_ENC(k, ctr, m1, nil)
  e2 = AEAD_ENC(k, ctr, m2, nil)
]
Alice -> Bob: e1, e2
principal Bob[
  knows private k
  knows public ctr
  d1 = AEAD_DEC(k, ctr, e1, nil)?
  d2 = AEAD_DEC(k, ctr, e2, nil)?
]
queries[
  confidentiality? m1
  confidentiality? m2
]
\end{lstlisting}
			\end{vpmodel}
		\end{column}
		\begin{column}{0.53\textwidth}
			\begin{vpresult}[top=0.5mm, bottom=0.5mm]{Both messages, to an attacker that never touched the wire}
				\begin{lstlisting}[basicstyle=\vpcode]
FAIL x confidentiality? m1
 | 1. Attacker observes e1 on the wire.
 | 2. Attacker observes e2 on the wire.
 | 3. Attacker recovers m1 from e1: e2 shares its key
 |    and nonce.
 > m1 (m1) is obtained by Attacker.
\end{lstlisting}
			\end{vpresult}
			\scriptsize
			\begin{itemize}
				\item \texttt{AEAD\_ENC(key, nonce, plaintext, ad)}, and \texttt{AEAD\_DEC} takes the nonce in the same place
				\item Two such terms under one key and one nonce, differing in content, are a \textbf{reused pair}: both plaintexts fall out, and any ciphertext becomes forgeable under that pair without the key, which is how AES-GCM and ChaCha20-Poly1305 really fail
				\item \texttt{knows} gives every session the same nonce, \texttt{generates} gives each its own
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}[fragile]{Phases: modeling the passage of time}
	\begin{vpmodel}{Compromise, after the fact}
		\begin{lstlisting}
principal Alice[...]      // the protocol runs to completion
Bob -> Alice: b1
phase[1]
principal Alice[leaks a2] // only now does a2 escape
\end{lstlisting}
	\end{vpmodel}
	\vspace{0.3em}
	\small
	\begin{enumerate}
		\item \textbf{Knowledge.} The attacker learns a value at the earliest phase in which it is communicated: \texttt{b1} in phase 0, \texttt{a2} in phase 1
		\item \textbf{Manipulation.} A substitution is judged against what the attacker knew at that earliest phase, so it cannot use \texttt{a2} to forge \texttt{b1}
		\item \textbf{No retroaction.} Without this rule, any key ever leaked would forge any signature ever checked
	\end{enumerate}
	\begin{center}
		\small Phase numbers increase by exactly one. This is how forward secrecy and harvest-now-decrypt-later get expressed.
	\end{center}
\end{frame}

\begin{frame}[fragile]{Scenarios: one principal, several peers}
	\begin{columns}[t]
		\begin{column}{0.44\textwidth}
			\begin{vpmodel}{Binding a peer placeholder}
				\begin{lstlisting}
principal Alice[
  knows private a
  knows public gpeer
  generates ni
  ga = PUBKEY(a)
  m1 = PKE_ENC(gpeer, CONCAT(ni, ga))
]
Alice -> Bob: m1
...
scenarios[
  Alice[gpeer = gb]
  Alice[gpeer = gm]
]
\end{lstlisting}
			\end{vpmodel}
		\end{column}
		\begin{column}{0.54\textwidth}
			\scriptsize
			\begin{itemize}
				\item Sessions replicate runs with the \textit{same} peers. Scenarios replicate runs with \textit{different} ones, under one attacker, for properties that depend on \textit{who} the peer is: identity misbinding, unknown-key-share, impersonation
				\item Each entry binds constants of one principal. The model is cloned once per entry, with the binding substituted
				\item \texttt{generates} values differ per scenario, suffixed \texttt{@2} onward. \texttt{knows} values stay shared: Bob is the same Bob in both
				\item A binding that depends on a \texttt{leaks} value marks the scenario \textbf{corrupt}, and that run may halt at a failed check
				\item Queries refer to the first scenario; the others feed the attacker. Two scenarios at two sessions is four copies of every principal
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}

\section{Queries}

\begin{frame}{Five questions you can ask}
	\begin{center}
		\small
		\begin{tabular}{|l|l|}
			\hline
			\textbf{Query}                     & \textbf{Asks}                                          \\
			\hline
			\texttt{confidentiality? v}        & Can the attacker obtain \texttt{v}?                    \\
			\hline
			\texttt{authentication? A -> B: x} & If B uses \texttt{x}, did A really send it?            \\
			\hline
			\texttt{freshness? v}              & Could \texttt{v} recur across runs, enabling a replay? \\
			\hline
			\texttt{equivalence? x, y}         & Do these always resolve to the same value?             \\
			\hline
			\texttt{unlinkability? x, y}       & Can the attacker link these to one principal?          \\
			\hline
		\end{tabular}
	\end{center}
	\vspace{0.7em}
	\begin{itemize}
		\item All five are judged against the same executions: the honest one, plus every mutated execution the search validates
		\item A query \textbf{passes} when no explored execution contradicts it
		\item A single contradicting execution fails it, and Verifpal prints that execution
	\end{itemize}
\end{frame}

\begin{frame}{\texttt{confidentiality?}}
	\definitionbox{\texttt{confidentiality? v} is contradicted when}{In some explored execution, the attacker's knowledge contains a value \textbf{equivalent} to what \texttt{v} \textbf{resolves to} in some principal's state.}
	\vspace{0.5em}
	\begin{itemize}
		\item \textbf{Resolves to}: the query is about the value, not the name. If the model says \texttt{v = HASH(a, b)}, the attacker must come to hold that hash
		\item \textbf{Equivalent}: equality is judged under the equational theory, so an attacker holding \texttt{DH\_KEX(PUBKEY(b), a)} holds \texttt{DH\_KEX(PUBKEY(a), b)}
	\end{itemize}
	\vspace{0.5em}
	\begin{center}
		A passive attacker gets \texttt{m1} only if the key protecting it, or every ingredient of that key, crosses the network. An active attacker has a second route: replace the public keys before they are used.
	\end{center}
\end{frame}

\begin{frame}{\texttt{authentication?}}
	\definitionbox{Contradicted when all three hold in Bob's state}{\textbf{(1)} the sender recorded for \texttt{e1} is not Alice, \textbf{(2)} the value is not just this session's own \texttt{e1} handed back, and \textbf{(3)} Bob \textit{successfully uses} \texttt{e1} inside at least one primitive.}
	\vspace{0.1em}
	\small
	\begin{itemize}
		\item Evaluated in the recipient's state and nowhere else. What Alice believes she sent has no bearing on whether Bob was fooled
		\item This is \textbf{injective agreement} in Lowe's hierarchy: each acceptance by Bob needs its own sending by Alice behind it
		\item Condition 3 is why checks matter: failing a check merely halts Bob, and halting is not impersonation. A query on a value Bob never uses is refused at load time
	\end{itemize}
	\begin{center}
		\footnotesize
		\textit{Two shapes of failure, and traces name them apart: the attacker \textbf{replaces} a value Alice never produced, or it \textbf{replays} one she really sent, from another session.}
	\end{center}
\end{frame}

\begin{frame}{The other three}
	\small
	\textbf{\texttt{freshness? v}} --- contradicted when \texttt{v} resolves to a value containing no \texttt{generates} constant at any depth, and a principal uses it inside a primitive.
	\begin{itemize}
		\item About how a value is \textit{derived}, not who knows it: leaking a generated value does not make it stale
		\item Catches values that \textit{could} be replayed, where \texttt{authentication?} catches the replays actually carried out
	\end{itemize}
	\vspace{0.3em}
	\textbf{\texttt{equivalence? c1, c2}} --- contradicted when a principal holds both and they do not resolve to equivalent values, after mutation and rewriting.
	\begin{itemize}
		\item Answered by a \textit{divergence} in principals' states, not by a deduction in attacker knowledge
		\item Halting a principal is not the same as making two diverge, and Verifpal will not report it as such
	\end{itemize}
	\vspace{0.3em}
	\textbf{\texttt{unlinkability? c1, c2}} --- contradicted when the attacker holds a concrete \textit{link witness} tying two observed values to the same principal. Built for contact tracing, voting and RFID.
\end{frame}

\begin{frame}[fragile]{Three labels that say how to read a failure}
	\footnotesize
	Not every \texttt{FAIL} is a forgery. A verdict may carry a qualifier, and the report explains it once per query:
	\vspace{0.2em}
	\begin{vpresult}[top=0.5mm, bottom=0.5mm]{The pre-shared-key model from earlier, under an active attacker}
		\begin{lstlisting}[basicstyle=\vpcode]
Info . authentication? Alice -> Bob: e1 reports a duplicate that Bob cannot rule out on its own:
       it contributes nothing to e1 before accepting it, so any run of it takes the same message
       twice. Read it as a replay-protection question about this flight rather than as a forgery.
FAIL x authentication? Alice -> Bob: e1  [duplicate acceptance: no recipient-generated context]
\end{lstlisting}
	\end{vpresult}
	\vspace{0.3em}
	\begin{itemize}
		\item \texttt{[duplicate acceptance]} --- a replay the recipient \textit{could} have caught, because the value it accepted carries a nonce or key that recipient generated
		\item \texttt{[duplicate acceptance: no recipient-generated context]} --- a replay of a flight the recipient contributed nothing to, so every run of it accepts the same message. The TLS 1.3 0-RTT replay that RFC 9846 names lands here
		\item \texttt{[attacker-supplied value]} --- a confidentiality failure over a value the attacker put there. The principal really does compute on it, but the honest value was not disclosed
	\end{itemize}
	\vspace{0.2em}
	\begin{center}
		\small \textit{All three are still failures. The label tells you which question to take back to the protocol, not that the result can be waved away.}
	\end{center}
\end{frame}

\begin{frame}[fragile]{A query option: precondition}
	\begin{columns}[t]
		\begin{column}{0.40\textwidth}
			\begin{vpmodel}[top=0.5mm, bottom=0.5mm]{Acting on an unauthenticated value}
				\begin{lstlisting}[basicstyle=\vpcode]
attacker[active]
principal Bob[
  knows private psk
  generates m
  e = ENC(psk, m)
  h = MAC(psk, e)
]
Bob -> Alice: e, h
principal Alice[
  knows private psk
  _ = ASSERT(MAC(psk, e), h)?
  m2 = DEC(psk, e)
]
Alice -> Carol: [m2]
principal Carol[
  _ = HASH(m2)
]
queries[
  authentication? Bob -> Alice:
    e[precondition[Alice -> Carol: m2]]
]
\end{lstlisting}
			\end{vpmodel}
		\end{column}
		\begin{column}{0.58\textwidth}
			\begin{vpresult}[top=0.5mm, bottom=0.5mm]{The trace ends with the send that makes it count}
				\begin{lstlisting}[basicstyle=\vpcode]
FAIL x authentication? Bob -> Alice:
       e[precondition[Alice -> Carol: m2]]
       [duplicate acceptance: no recipient-generated context]
 | 2. Attacker replays e (Bob to Alice) from another session,
 |    where it is ENC(psk, m#2).
 > e (ENC(psk, m#2)), which Bob sent in another session and not
 in this one, is successfully used in ASSERT(MAC(psk, e), h)?
 within Alice's state: Bob sent it once, Alice accepts it twice,
 so agreement is not injective.
 ! Alice still sends m2 to Carol, so the failure counts.
\end{lstlisting}
			\end{vpresult}
			\scriptsize
			\vspace{-0.5em}
			\begin{itemize}
				\item \texttt{precondition}, the only query option, restricts a query to executions where the named message is sent. Any query kind takes one
				\item Alice forwards \texttt{m2} only after authenticating \texttt{e}; had she halted first, that execution falls outside the query
				\item Secrecy \emph{on acceptance} has the same shape: \texttt{confidentiality? k[precondition[Client -> Server: req]]}
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}

\section{Worked Example: Challenge-Response}

\begin{frame}[fragile]{A challenge-response mechanism}
	\begin{columns}[t]
		\begin{column}{0.49\textwidth}
			\begin{vpmodel}{challengeresponse.vp}
				\begin{lstlisting}[basicstyle=\vpcode]
attacker[active]
principal Server[
  knows private s
  gs = PUBKEY(s)
]
principal Client[
  knows private c
  gc = PUBKEY(c)
  generates nonce
]
Client -> Server: nonce
principal Server[
  proof = SIGN(s, nonce)
]
Server -> Client: gs, proof
\end{lstlisting}
			\end{vpmodel}
		\end{column}
		\begin{column}{0.49\textwidth}
			\begin{vpmodel}{\ldots continued}
				\begin{lstlisting}[basicstyle=\vpcode]
principal Client[
  valid = SIGNVERIF(gs, nonce, proof)
  generates attestation
  signed = SIGN(c, attestation)
]
Client -> Server: [gc], attestation, signed
principal Server[
  storage = SIGNVERIF(gc, attestation, signed)?
]
queries[
  authentication? Server -> Client: proof
  authentication? Client -> Server: signed
]
\end{lstlisting}
			\end{vpmodel}
		\end{column}
	\end{columns}
	\vspace{0.4em}
	\begin{center}
		\small
		Client challenges Server to prove it holds the key behind \texttt{gs} by signing a nonce Client chose. Client then makes a signed attestation of its own, which Server verifies. Mechanisms like this sit inside many larger protocols. \textit{Does this one say what it means?}
	\end{center}
\end{frame}

\begin{frame}[fragile]{Two failures, of two different shapes}
	\begin{vpresult}[top=0.5mm, bottom=0.5mm]{Both queries fail, for unrelated reasons}
		\begin{lstlisting}[basicstyle=\vpcode]
FAIL x authentication? Server -> Client: proof
       | 1. Attacker observes nonce on the wire.
       | 2. Attacker constructs SIGN(nil, nonce).
       | 3. Attacker replaces proof (sent by Server to Client) with
       |    SIGN(nil, nonce). (proof was SIGN(s, nonce))
       > proof (SIGN(nil, nonce)), sent by Attacker and not by Server, is
       successfully used in SIGNVERIF(gs, nonce, proof) within Client's state.

FAIL x authentication? Client -> Server: signed
       [duplicate acceptance: no recipient-generated context]
       | 2. Attacker replays attestation (Client to Server) from another
       |    session, where it is attestation#2.
       | 4. Attacker replays signed (Client to Server) from another
       |    session, where it is SIGN(c, attestation#2).
       | 5. Server's SIGNVERIF(gc, attestation#2, SIGN(c, attestation#2))?
       |    passes --- the attacker controls one of its inputs.
       > signed (SIGN(c, attestation#2)), which Client sent in another
       session and not in this one, is successfully used in
       SIGNVERIF(gc, attestation, signed)? within Server's state: Client
       sent it once, Server accepts it twice, so agreement is not injective.
\end{lstlisting}
	\end{vpresult}
	\begin{center}
		\small The first is a \textbf{forgery}. The second is a \textbf{replay}, and its label says which kind: Server puts nothing of its own into \texttt{signed}. Different fixes.
	\end{center}
\end{frame}

\begin{frame}[fragile]{Fix one: guard the key, check the signature}
	\begin{columns}[t]
		\begin{column}{0.5\textwidth}
			\begin{vpmodel}{Two edits}
				\begin{lstlisting}
Server -> Client: [gs], proof
...
valid = SIGNVERIF(gs, nonce, proof)?
\end{lstlisting}
			\end{vpmodel}
			\vspace{0.4em}
			\begin{vpresult}{Re-run}
				\begin{lstlisting}
PASS + authentication? Server -> Client: proof
       [search exhausted at 2 sessions]
FAIL x authentication? Client -> Server: signed

FAIL x 1 of 2 queries failed.
\end{lstlisting}
			\end{vpresult}
		\end{column}
		\begin{column}{0.48\textwidth}
			\small
			The first trace never touched \texttt{gs}.
			\vspace{0.3em}
			\begin{itemize}
				\item \texttt{SIGNVERIF} had no \texttt{?}, so any signature at all counts as successfully used: Client computed \texttt{valid} and signed an attestation regardless
				\item Add the \texttt{?} alone and re-run. The next trace replaces \texttt{gs} with \texttt{PUBKEY(nil)} and \texttt{proof} with \texttt{SIGN(nil, nonce)}, and the check passes: \texttt{gs} was unguarded in the same message
			\end{itemize}
			\vspace{0.3em}
			Guard the key as well, and the forgery is gone.

			The replay is not.
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{The replay that remains}
	\begin{center}
		\Large
		Nothing in \texttt{signed} belongs to this run of the protocol.
	\end{center}
	\vspace{0.7em}
	\begin{itemize}
		\item \texttt{attestation} is fresh, so it differs between Client's two sessions. Good
		\item But it is fresh \textit{on Client's side}. Server contributed nothing to it
		\item So the attacker takes \texttt{attestation\#2} and \texttt{signed\#2} off session 2's wire and delivers both into session 1
		\item Server's \texttt{SIGNVERIF} passes: the signature is real, by the real Client, over a real attestation. Client signed once, Server accepted twice, and that is what injective agreement forbids
	\end{itemize}
	\vspace{0.5em}
	\begin{center}
		\textit{A signature proves who made a value. It says nothing about which conversation the value belongs to.}
	\end{center}
\end{frame}

\begin{frame}[fragile]{Why the default is two sessions}
	\begin{vpresult}{The model after fix one, run with one session per principal}
		\begin{lstlisting}
$ verifpal verify challengeresponse.vp --sessions 1

PASS + authentication? Server -> Client: proof  [search exhausted at 1 session]
PASS + authentication? Client -> Server: signed  [search exhausted at 1 session]

PASS + All 2 queries pass.
\end{lstlisting}
	\end{vpresult}
	\vspace{0.4em}
	\small
	\begin{itemize}
		\item Under one session the replay cannot be \textbf{stated}, let alone found: \texttt{attestation\#2} does not exist as a term
		\item Before analysis, Verifpal rewrites the model so each principal and message appears once per session, values renamed apart. \texttt{knows} material stays shared; \texttt{generates} values do not
	\end{itemize}
	\begin{center}
		\footnotesize
		\textit{Two is not unbounded: an attack needing three sessions is out of reach at the default, as this one was at one. \texttt{--saturate} raises the count until the verdicts stop moving.}
	\end{center}
\end{frame}

\begin{frame}[fragile]{Fix two: let the Server choose the challenge}
	\begin{columns}[t]
		\begin{column}{0.52\textwidth}
			\begin{vpmodel}{The real fix}
				\begin{lstlisting}[basicstyle=\vpcode]
principal Server[
  knows private s
  gs = PUBKEY(s)
  generates schallenge
]
...
Server -> Client: [gs], proof, schallenge
principal Client[
  valid = SIGNVERIF(gs, nonce, proof)?
  signed = SIGN(c, schallenge)
]
Client -> Server: [gc], signed
principal Server[
  storage = SIGNVERIF(gc, schallenge, signed)?
]
\end{lstlisting}
			\end{vpmodel}
		\end{column}
		\begin{column}{0.46\textwidth}
			\vspace{0.2em}
			\begin{vpresult}{Re-run}
				\begin{lstlisting}[basicstyle=\vpcode]
PASS + authentication?
       Server -> Client: proof
       [search exhausted at 2 sessions]
PASS + authentication?
       Client -> Server: signed
       [search exhausted at 2 sessions]

PASS + All 2 queries pass.
\end{lstlisting}
			\end{vpresult}
			\vspace{0.2em}
			\footnotesize
			Now each side signs something the \textit{other} side drew fresh. A signature from session 2 is over \texttt{schallenge\#2}, which session 1's Server never sent and will not accept.
			\vspace{0.3em}

			Binding \texttt{attestation} to Client's own \texttt{nonce} does not work: the attacker replays the nonce too.
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{The modeling loop}
	\begin{center}
		\Large
		Model. Query. Read the trace. Adjust. Re-run.
	\end{center}
	\vspace{0.7em}
	\begin{itemize}
		\item Three edits took that challenge-response from two failures to two passes, and each one came from reading a trace rather than from guessing
		\item Two levers decided most of it: \textbf{what is guarded}, and \textbf{what is checked}
		\item The third was structural, and the tool found it because it analyzes two sessions rather than one
	\end{itemize}
	\vspace{0.7em}
	\begin{center}
		\textit{The value of the exercise is not that Verifpal fixed the protocol. It is that writing the model down forced three questions that reading the protocol description never would have.}
	\end{center}
\end{frame}

\section{Worked Example: A 3-of-5 Threshold Signature}

\begin{frame}[fragile]{Three primitives for splitting a key}
	\begin{columns}[t]
		\begin{column}{0.5\textwidth}
			\begin{vpmodel}[top=0.5mm, bottom=0.5mm]{A 3-of-5 signing group, abridged}
				\begin{lstlisting}[basicstyle=\vpcode]
principal Dealer[
  generates k
  pk = PUBKEY(k)
  s1, s2, s3, s4, s5 = THRESHOLD_SPLIT[3](k)
  ea = AEAD_ENC(psk_a, nd_a, s1, pk)  // eb, ec too
  leaks s4
]
Dealer -> Alice: [pk], nd_a, ea       // Bob, Carol too
principal Alice[
  sa = AEAD_DEC(psk_a, nd_a, ea, pk)?
  generates na
  ca = PUBKEY(na)      // ca, cb, cc -> Coordinator
]
Coordinator -> Alice: m, cb, cc
Coordinator -> Bob:   m, ca, cc
Coordinator -> Carol: [m], [ca], [cb]
principal Alice[
  cl_a = CONCAT(ca, cb, cc)
  pa = THRESHOLD_SIGN(sa, na, cl_a, m)
]
principal Coordinator[
  sig = THRESHOLD_JOIN(pa, pb, pc)
]
Coordinator -> Verifier: m, sig

\end{lstlisting}
			\end{vpmodel}
		\end{column}
		\begin{column}{0.48\textwidth}
			\scriptsize
			A FROST-shaped signing group, sitting on its own threshold.
			\vspace{-0.8em}
			\begin{itemize}
				\item \texttt{THRESHOLD\_SPLIT[t](k)}: the bracket carries the threshold, the names on the left give the share count. Five names and a \texttt{[3]} is 3-of-5
				\item The threshold is part of the term's identity: shares of \texttt{[2]} and \texttt{[3]} do not mix, even for one \texttt{k}
				\item \texttt{THRESHOLD\_SIGN(share, nonce, commitments, message)} is round two: partials combine only over \textit{distinct} shares of one split, agreeing on commitments and message
				\item \texttt{THRESHOLD\_JOIN} is symbolic Lagrange interpolation, three ways: \texttt{t} shares give \texttt{k}, \texttt{t} values \texttt{PUBKEY(share)} give \texttt{PUBKEY(k)}, and \texttt{t} partials give \texttt{SIGN(k, message)}
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}[fragile]{Two failures, neither of them enough}
	\begin{vpresult}[top=0.5mm, bottom=0.5mm]{One leaked share, two signing oracles, and a signature the group never made}
		\begin{lstlisting}[basicstyle=\vpcode]
PASS + confidentiality? k   [search exhausted at 2 sessions]
PASS + confidentiality? s1  [search exhausted at 2 sessions]
FAIL x authentication? Coordinator -> Verifier: sig
       | 1. Attacker observes THRESHOLD_SIGN(s1, na, cl_a, nil) on the wire.
       | 2. Attacker observes THRESHOLD_SIGN(s2, nb, cl_a, nil) on the wire.
       | 3. Attacker is handed s4 by a leaks declaration.
       | 7. During Dealer's run (phase 0), attacker constructs cl_a.
       | 8. Attacker combines SIGN(k, nil) out of the partial signatures THRESHOLD_SIGN(s1, na, cl_a, nil),
       |    THRESHOLD_SIGN(s2, nb, cl_a, nil), THRESHOLD_SIGN(s4, nil, cl_a, nil).
       | 9. Attacker replaces m, sig (sent by Coordinator to Verifier) with nil, SIGN(k, nil).
       |    (sig was THRESHOLD_JOIN(pa, pb, pc))
       | 10. Verifier's SIGNVERIF(pk, nil, SIGN(k, nil))? passes --- the attacker controls one of its inputs.
       > sig (SIGN(k, nil)), sent by Attacker and not by Coordinator, is
       successfully used in SIGNVERIF(pk, m, sig)? within Verifier's state.
\end{lstlisting}
	\end{vpresult}
	\scriptsize
	\begin{itemize}
		\item Carol's round-two inputs are guarded, so her partial stays bound to the real \texttt{m}. Alice's and Bob's are not, so the attacker picks what they sign: they become \textbf{signing oracles}, which is the input validation RFC 9591 tells applications to do and this model declines to
		\item The leaked \texttt{s4} supplies the third contribution. One leaked share is under the threshold, two oracle partials are under it, and together they are three
		\item The group key and Alice's share stay confidential. Nothing was stolen or broken: three distinct contributions to one signature were collected
	\end{itemize}
\end{frame}

\begin{frame}{The boundary, from both sides}
	\begin{center}
		\scriptsize
		\begin{tabular}{|l|l|l|}
			\hline
			\textbf{Model}                 & \textbf{What the attacker has}          & \textbf{Result}         \\
			\hline
			\texttt{..three\_of\_five}      & Leaked \texttt{s4}, Alice as an oracle  & 2 partials, no forgery  \\
			\hline
			\texttt{..two\_oracles\_hold}   & Alice and Bob as oracles, no leak       & 2 partials, no forgery  \\
			\hline
			\texttt{..two\_oracles}         & Leaked \texttt{s4}, \textit{and} both oracles & 3 partials, forgery \\
			\hline
		\end{tabular}
	\end{center}
	\vspace{0.2em}
	\scriptsize
	\begin{itemize}
		\item The first two rows are \texttt{PASS} at the default two sessions: the search found nothing, which as always is weaker than proof
		\item Repeating an oracle's partial does not help: \texttt{THRESHOLD\_JOIN} counts \textit{distinct} shares, and partials over another split, list or message do not join
		\item Reuse a nonce under one share for two partials and Verifpal hands that share to the attacker: RFC 9591's nonce warning, shaped like the AEAD rule
		\item \textit{Not} modeled: curve arithmetic, ciphersuites, or a distributed key-generation ceremony. \texttt{THRESHOLD\_SPLIT} is a trusted dealer, one symbolic nonce standing for FROST's hiding-and-binding pair
	\end{itemize}
	\begin{center}
		\small \textit{A leaked share and an exposed signing oracle are different failures, filed in different places. Counting them together is what the tool does.}
	\end{center}
\end{frame}

\begin{frame}[fragile]{Forward secrecy, measured}

Alice -> Bob: e1

 |    obtaining m1.

			\end{vpresult}

principal Bob[

Bob -> Alice: [gb], gy, sig

\end{lstlisting}
			\end{vpresult}
		\end{column}
	\end{columns}
	\begin{center}
		\scriptsize Both leak Bob's long-term key \texttt{b} in phase 1. On the left it decrypts everything he ever received. On the right, the leaked key could sign new things, but the phase-0 run is over, and \texttt{k} never depended on \texttt{b}.
	\end{center}
\end{frame}

\begin{frame}[fragile]{When the primitive itself is the assumption}
	\footnotesize
	Everything so far assumed each primitive is perfect. A call site may declare otherwise:
	\begin{vpmodel}{Three capabilities}
		\begin{lstlisting}
s  = SIGN[forgeable](sk, m)         // authenticity is lost
c  = ENC[malleable](k, m)           // this ciphertext can be reshaped
ga = PUBKEY[weak from phase 1](a)   // confidentiality lost, from phase 1 on
\end{lstlisting}
	\end{vpmodel}
	\vspace{0.3em}
	\begin{itemize}
		\item \texttt{weak}: holding the term recovers what it protects. On \texttt{PUBKEY} it recovers the private key, which is the discrete logarithm problem falling
		\item \texttt{forgeable}: the term becomes constructible without its secret argument. Declared for \texttt{SIGN}, \texttt{MAC}, \texttt{RINGSIGN} and \texttt{AEAD\_ENC}
		\item \texttt{malleable}: \texttt{ENC} only. Reshape a ciphertext you already hold, without ever learning the key
		\item \texttt{from phase N} delays a capability. Cryptanalysis does not un-happen, so it reads \textit{from} a phase rather than \textit{in} one
	\end{itemize}
	\begin{center}
		Each keyword grants one named capability and nothing else. An unannotated model behaves exactly as before.
	\end{center}
\end{frame}

\begin{frame}[fragile]{Forgeability, demonstrated}
	\begin{columns}[t]
		\begin{column}{0.42\textwidth}
			\begin{vpmodel}[top=0.5mm, bottom=0.5mm]{A signed ephemeral, bound to a nonce}
				\begin{lstlisting}[basicstyle=\vpcode]
principal Client[
  generates n, x, m1
  gx = PUBKEY(x)
]
Client -> Server: n, gx
principal Server[
  knows private s
  gs = PUBKEY(s)
  generates y
  gy = PUBKEY(y)
  sig = SIGN[forgeable](s, HASH(gy, n))
]
Server -> Client: [gs], gy, sig
principal Client[
  _ = SIGNVERIF(gs, HASH(gy, n), sig)?
  k = DH_KEX(gy, x)
  e1 = AEAD_ENC(k, n, m1, gx)
]
Client -> Server: e1
\end{lstlisting}
			\end{vpmodel}
		\end{column}
		\begin{column}{0.56\textwidth}
			\begin{vpresult}[top=0.5mm, bottom=0.5mm]{With plain \texttt{SIGN}}
				\begin{lstlisting}[basicstyle=\vpcode]
PASS + confidentiality? m1
PASS + authentication? Server -> Client: sig
\end{lstlisting}
			\end{vpresult}
			\vspace{0.1em}
			\begin{vpresult}[top=0.5mm, bottom=0.5mm]{With \texttt{SIGN[forgeable]}}
				\begin{lstlisting}[basicstyle=\vpcode]
FAIL x confidentiality? m1
 | 4. Attacker breaks sig under the declared `forgeable` assumption
 |    using HASH(PUBKEY(nil), n), obtaining
 |    SIGN(s, HASH(PUBKEY(nil), n)).
 | 5. Attacker replaces gy, sig (sent by Server to Client) with
 |    PUBKEY(nil), SIGN(s, HASH(PUBKEY(nil), n)).
 | 6. Client's SIGNVERIF(gs, HASH(PUBKEY(nil), n),
 |    SIGN(s, HASH(PUBKEY(nil), n)))? passes --- the attacker
 |    controls one of its inputs.
 | 10. Attacker opens AEAD_ENC(DH_KEX(gx, nil), n, m1, gx) with
 |     DH_KEX(gx, nil), n, obtaining m1.
FAIL x authentication? Server -> Client: sig
\end{lstlisting}
			\end{vpresult}
			\vspace{0.1em}
			\scriptsize
			One bracketed word, and both properties go. The forgery is under Server's real \texttt{s}: forgeable means the secret is no longer needed, so guarding \texttt{gs} buys nothing.\vspace{-0.6em}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}[fragile]{Harvest now, decrypt later}
	\begin{columns}[t]

Bob -> Alice: [gb], [ekb]

\begin{frame}[fragile]{Reading a result under assumptions}
	\scriptsize
	\begin{columns}[t]
		\begin{column}{0.52\textwidth}
			\begin{vpresult}[top=0.5mm, bottom=0.5mm]{Printed before every result, pass or fail}
				\begin{lstlisting}[basicstyle=\vpcode]
Warning ! Analysis performed under 2 declared
          weakening assumptions:
Warning ! PUBKEY[weak from phase 1](b)
Warning ! PUBKEY[weak from phase 1](a)
\end{lstlisting}
			\end{vpresult}
			\vspace{0.15em}
			\begin{vpresult}[top=0.5mm, bottom=0.5mm]{When an annotation reaches further}
				\begin{lstlisting}[basicstyle=\vpcode]
Info . s2 is written without an annotation, but a
  `forgeable` assumption is declared on sk, so the
  attacker is analyzed as able to produce s2 too.
\end{lstlisting}
			\end{vpresult}
			\vspace{0.15em}
			\begin{vpresult}[top=0.5mm, bottom=0.5mm]{Rejected at load time}
				\begin{lstlisting}[basicstyle=\vpcode]
sanity error: SIGN provides authenticity, not
  confidentiality; did you mean `SIGN[forgeable]`?
sanity error: the secret `DH_KEX` derives is only
  as hidden as the key it is built on; did you mean
  `PUBKEY[weak]`?
\end{lstlisting}
			\end{vpresult}
		\end{column}
		\begin{column}{0.46\textwidth}
			\begin{itemize}
				\item An attack found under a declared assumption is genuine \textit{under that assumption}, and individual trace steps name the assumption they rest on
				\item Annotations reach past their own call site: \textbf{equivalent terms} share an annotation, and \texttt{forgeable} attaches to the \textbf{scheme under one secret}. A signature scheme that has lost authenticity has lost it for every message
				\item An assumption scheduled for a phase the model never declares is an error, not a no-op
			\end{itemize}
		\end{column}
	\end{columns}
	\begin{center}
		\scriptsize
		\textit{A clean result under a stated assumption is not a clean result. The summary line exists so that a trace pasted into an issue tracker does not read as an unconditional finding.}
	\end{center}
\end{frame}

\section{Limits, and Getting Started}

\begin{frame}{Sound, but incomplete}
	\scriptsize
	\begin{columns}[t]
		\begin{column}{0.48\textwidth}
			\textbf{What a \texttt{FAIL} means}
			\begin{itemize}
				\item Attacker actions the engine re-executed against real principal state and re-checked before printing
				\item Nothing in the search can record a result on its own, so a solver bug costs a missed attack, never an invented one
				\item That covers the solver, not the deduction around it: 1.4.3 fixed several false reports that came of combining runs which could not both have happened
			\end{itemize}
		\end{column}
		\begin{column}{0.48\textwidth}
			\textbf{What a \texttt{PASS} does not mean}
			\begin{itemize}
				\item Not a proof: the search did not reach an attack, which is weaker
				\item Bounded sessions and peers: an attack needing more of either than you analyzed is out of reach
				\item Symbolic model, so padding oracles, timing channels and weak randomness are not modeled
			\end{itemize}
		\end{column}
	\end{columns}
	\alertbox{The honest claim}{Verifpal is a bug-finding tool, not a proof-producing verifier. Its own manual says so, and it labels the traces whose witnesses rest on within-session sharing. Compare that to Part 2.}
\end{frame}

\begin{frame}{The command line, beyond a verdict}
	\scriptsize
	\begin{columns}[t]
		\begin{column}{0.48\textwidth}
			\textbf{Scripting \texttt{verify}}
			\begin{itemize}
				\item \texttt{--result-code} appends one letter-digit pair per query: \texttt{a0a1} is a passing authentication query, then a failing one. Pin it in a test
				\item \texttt{--fail-on-attack} exits with status 2 on any failure, for continuous integration
				\item \texttt{--saturate} climbs from the default two sessions up to four, stops at the first count whose result code matches the one below it, and reports that analysis
				\item \texttt{--auto-queries} discards your queries and asks the generic ones: every secret confidential, every used value authenticated and fresh. A first look, not a specification
			\end{itemize}
		\end{column}
		\begin{column}{0.48\textwidth}
			\textbf{Reports and other commands}
			\begin{itemize}
				\item \texttt{--format html}: one self-contained page, with a sequence diagram of the protocol and of each attack, source highlighted by verdict
				\item \texttt{--format tex}: a LaTeX document, terms typeset as mathematics, figures liftable into a paper. \texttt{json} for tooling
				\item \texttt{verifpal pretty}: canonical formatting; \texttt{--check} fails CI on drift
				\item \texttt{verifpal diagram}: a Mermaid sequence diagram of the model
				\item \texttt{VERIFPAL\_SOLVE\_DEBUG=1} logs every substitution the search proposed, including the rejected ones
			\end{itemize}
		\end{column}
	\end{columns}
	\vspace{0.4em}
	\begin{center}
		\small \textit{A model in a repository should carry a curated \texttt{queries} block and a pinned result code, not a generated one.}
	\end{center}
\end{frame}

\begin{frame}{Getting started}
	\footnotesize
	\begin{columns}[t]
		\begin{column}{0.48\textwidth}
			\textbf{Install}
			\begin{itemize}
				\item Homebrew on macOS and Linux, Scoop on Windows, both with automatic updates; otherwise a release binary, or \texttt{cargo build --release} from source
				\item Or install nothing: \url{https://verifpal.com/workbench} runs the analyzer in your browser as WebAssembly
			\end{itemize}
			\vspace{0.3em}
			\textbf{Editors}
			\begin{itemize}
				\item \texttt{verifpal lsp} drives extensions for VS Code, Neovim and Zed: live analysis, diagrams, hover docs
			\end{itemize}
		\end{column}
		\begin{column}{0.48\textwidth}
			\textbf{Read}
			\begin{itemize}
				\item Manual: \url{https://static.verifpal.com/manual.pdf}, with case studies on Signal, Scuttlebutt and a post-quantum handshake
				\item Paper: \textit{Verifpal Seven Years Later}, ePrint 2026/1654
			\end{itemize}
			\vspace{0.1em}
			\textbf{Try}
			\begin{itemize}
				\item Model something you already use: a password reset or an API login
				\item Ask for confidentiality and authentication first. Add \texttt{phase[1]} once those hold
				\item Shipping? Cross-check in ProVerif/Tamarin
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Where this leaves us}
	\begin{itemize}
		\item \textbf{Part 1}: machine-checked proofs of designs and implementations are real, and they are deployed at Internet scale
		\item \textbf{Part 2}: a claim of verification is only worth the boundary documentation and the CI behind it
		\item \textbf{Part 3}: the tools are within reach, and a model can state which hardness assumption it is betting on, then report what the protocol is worth if that bet loses
	\end{itemize}
	\vspace{0.7em}
	\begin{center}
		\Large
		Writing the model is most of the benefit.

		\vspace{0.3em}

		\normalsize
		\textit{You cannot specify a protocol precisely without discovering what you left unspecified. That was Lesson 1 from TLS 1.3, and it holds at every scale, including a twenty-line model you wrote this afternoon.}
	\end{center}
\end{frame}
